type unpackBlock struct {
 Arm string `json:"arm"`
 Calls int `json:"calls"`
 Nanoseconds int64 `json:"nanoseconds"`
 Bytes, Allocs uint64
 GC uint32
 PauseNs uint64
 HeapBefore, HeapAfter uint64
}

type unpackQuartet struct {
 Round, Index int
 Order string
 Blocks []unpackBlock
 Ratio float64
}

// TestUnpackInProcessExperiment 在同一模块/进程交错操作，保留全部GC、分配与时间块。
func TestUnpackInProcessExperiment(t *testing.T) {
 if runtime.GOMAXPROCS(0)!=1 { t.Fatal("GOMAXPROCS1 required") }
 name, mode := os.Getenv("GSR_CASE"), os.Getenv("GSR_MODE")
 var packed []byte
 var newMessage func() any
 var independent []byte
 if strings.Contains(name,"_trace_") || strings.HasPrefix(name,"mask4_") || strings.HasPrefix(name,"counts_") {packed,independent=mixedMaskFixture(t,name)}
 for _, sample := range representativeCases(t) {
  if sample.Name!=name { continue }
  _, wires := representativeOperations(t, sample)
  checkRepresentativeWire(t, sample, wires)
  packed = wires["sproto_pack"]
  _, newMessage = representativeMessages(sample.Type)
 }
 if packed==nil {
  var word [8]byte
  repeats:=512
  switch name {
  case "density1", "density2", "density3", "density4", "density5", "density6", "density7":
   count:=int(name[len(name)-1]-'0')
   for i:=0;i<count;i++ {word[i]=byte(i+1)}
  case "short1":word[0]=1;repeats=1
  case "short7":for i:=0;i<7;i++ {word[i]=byte(i+1)};repeats=1
  case "zero4096":
  case "ff4096":for i:=range word {word[i]=255}
  default:t.Fatal("unknown case",name)
  }
  packed=sp.Pack(bytes.Repeat(word[:],repeats))
 }
 operation:=os.Getenv("GSR_OP")
 if operation!="unpack" && operation!="pdecode" { t.Fatal("invalid operation") }
 if operation=="pdecode" && newMessage==nil {t.Fatal("no message constructor")}
 left:=sp.UnpackWithLimit
 right:=sp.ExperimentUnpackCandidate
 switch mode {
 case "self": right=left
 case "clone": right=sp.ExperimentUnpackClone
 case "pass1", "pass2":
 default: t.Fatal("unknown mode")
 }
 beforeInput:=append([]byte(nil),packed...)
 expected,err:=left(packed,8<<20)
 if err!=nil {t.Fatal(err)}
 if independent!=nil && !bytes.Equal(expected,independent) {t.Fatal("independent restored wire mismatch")}
 for _, fn:=range []func([]byte,int)([]byte,error){left,right} {
  got,err:=fn(packed,8<<20)
  if err!=nil || !bytes.Equal(got,expected) || cap(got)!=cap(expected) {t.Fatal("wire/cap mismatch",err)}
  if operation=="pdecode" {
   a,b:=newMessage(),newMessage()
   if _,err:=sp.Decode(expected,a);err!=nil {t.Fatal(err)}
   if _,err:=sp.Decode(got,b);err!=nil || !reflect.DeepEqual(a,b) {t.Fatal("decode mismatch",err)}
  }
 }
 if os.Getenv("GSR_PREFLIGHT")=="1" {
  chunks:=splitMaskChunks(t,packed)
  histogram:=make([]int,256)
  bag:=make([][]byte,len(chunks))
  for i,c:=range chunks {histogram[int(c.packed[0])]++;bag[i]=c.packed}
  sort.Slice(bag,func(i,j int)bool{return bytes.Compare(bag[i],bag[j])<0})
  hash:=func(b []byte)string{v:=sha256.Sum256(b);return hex.EncodeToString(v[:])}
  // 单个chunk长度由header和内容确定，排序后拼接可唯一恢复分段。
  raw,err:=json.MarshalIndent(struct{Name string;PackedBytes,OutputBytes int;PackedSHA256,OutputSHA256,BagSHA256 string;Histogram []int}{name,len(packed),len(expected),hash(packed),hash(expected),hash(bytes.Join(bag,nil)),histogram},"","  ")
  if err!=nil {t.Fatal(err)}
  if err:=os.WriteFile(os.Getenv("GSR_AUDIT"),append(raw,'\n'),0644);err!=nil {t.Fatal(err)}
  return
 }
 makeRun:=func(fn func([]byte,int)([]byte,error))func() {
  return func() {
   data,err:=fn(packed,8<<20)
   if err!=nil {t.Fatal(err)}
   if operation=="unpack" {encodedSink=data; return}
   message:=newMessage()
   if _,err:=sp.Decode(data,message);err!=nil {t.Fatal(err)}
   representativeSink=message
  }
 }
 runs:=map[string]func(){"A":makeRun(left),"B":makeRun(right)}
 runtime.GC()
 for _,arm:=range []string{"A","B"} {
  start:=time.Now()
  for time.Since(start)<50*time.Millisecond {for range 64 {runs[arm]()}}
 }
 start:=time.Now()
 for range 256 {runs["A"]()}
 calls:=int(float64(20*time.Millisecond)/float64(time.Since(start))*256)
 calls=max(1,min(calls,1<<20))
 blocks:=[]unpackQuartet{}
 for round:=0; round<6; round++ {
  for quartet:=0; quartet<4; quartet++ {
   order:="ABBA"
   if (round+quartet)%2!=0 {order="BAAB"}
   q:=unpackQuartet{Round:round,Index:quartet,Order:order}
   logRatio:=0.0
   for _,label:=range order {
    arm:=string(label)
    var before,after runtime.MemStats
    runtime.ReadMemStats(&before)
    run:=runs[arm]
    start:=time.Now()
    for range calls {run()}
    ns:=time.Since(start).Nanoseconds()
    runtime.ReadMemStats(&after)
    block:=unpackBlock{Arm:arm,Calls:calls,Nanoseconds:ns,Bytes:after.TotalAlloc-before.TotalAlloc,Allocs:after.Mallocs-before.Mallocs,GC:after.NumGC-before.NumGC,PauseNs:after.PauseTotalNs-before.PauseTotalNs,HeapBefore:before.HeapAlloc,HeapAfter:after.HeapAlloc}
    q.Blocks=append(q.Blocks,block)
    sign:=1.0
    if arm=="A" {sign=-1}
    logRatio+=sign*math.Log(float64(ns)/float64(calls))
   }
   q.Ratio=math.Exp(logRatio/2)
   blocks=append(blocks,q)
  }
 }
 if !bytes.Equal(packed,beforeInput) {t.Fatal("input modified")}
 raw,err:=json.MarshalIndent(struct{Name,Operation,Mode string;CallsPerBlock int;Quartets []unpackQuartet}{name,operation,mode,calls,blocks},"","  ")
 if err!=nil {t.Fatal(err)}
 if err:=os.WriteFile(os.Getenv("GSR_OUTPUT"),append(raw,'\n'),0644);err!=nil {t.Fatal(err)}
 runtime.KeepAlive(runs)
}


type maskChunk struct {packed,plain []byte}

// splitMaskChunks独立解释完整packed段，供重排诊断验证内容、直方图和输出长度。
func splitMaskChunks(t *testing.T,packed []byte) []maskChunk {
 t.Helper()
 var chunks []maskChunk
 for len(packed)>0 {
  header:=packed[0]
  n:=1+bits.OnesCount8(header)
  plain:=make([]byte,8)
  if header==255 {
   if len(packed)<2 {t.Fatal("truncated FF length")}
   n=2+(int(packed[1])+1)*8
   if len(packed)<n {t.Fatal("truncated FF contents")}
   plain=append([]byte(nil),packed[2:n]...)
  }else {
   if len(packed)<n {t.Fatal("truncated ordinary word")}
   cursor:=1
   for bit:=0;bit<8;bit++ {if header&(1<<bit)!=0 {plain[bit]=packed[cursor];cursor++}}
  }
  chunks=append(chunks,maskChunk{packed:append([]byte(nil),packed[:n]...),plain:plain})
  packed=packed[n:]
 }
 return chunks
}

// mixedMaskFixture只改变诊断输入的mask位置/顺序；保留原业务消息作为单独对照。
func mixedMaskFixture(t *testing.T,name string)([]byte,[]byte) {
 t.Helper()
 var chunks []maskChunk
 if strings.Contains(name,"_trace_") {
  source:="game_large"
  if strings.HasPrefix(name,"int_") {source="int32_edges"}
  for _,sample:=range representativeCases(t) {
   if sample.Name!=source {continue}
   _,wires:=representativeOperations(t,sample)
   checkRepresentativeWire(t,sample,wires)
   chunks=splitMaskChunks(t,wires["sproto_pack"])
  }
 }else {
  masks:=[]byte{15}
  if strings.HasPrefix(name,"mask4_") && name!="mask4_fixed" {
   masks=nil
   for mask:=1;mask<255;mask++ {if bits.OnesCount8(uint8(mask))==4 {masks=append(masks,byte(mask))}}
  }
  if strings.HasPrefix(name,"counts_") {masks=[]byte{1,3,7,15,31,63,127}}
  for i:=0;i<512;i++ {
   header:=masks[i%len(masks)]
   packed:=[]byte{header}
   for bit:=0;bit<8;bit++ {if header&(1<<bit)!=0 {packed=append(packed,byte(bit+1))}}
   chunks=append(chunks,splitMaskChunks(t,packed)...)
  }
 }
 if len(chunks)==0 {t.Fatal("missing trace")}
 if strings.HasSuffix(name,"grouped") {
  sort.SliceStable(chunks,func(i,j int)bool{return chunks[i].packed[0]<chunks[j].packed[0]})
 }else if strings.HasSuffix(name,"shuffled") {
  // 固定本地LCG，与计时无关；同一输入跨构建保持相同重排。
  seed:=uint32(0x12345678)
  for i:=len(chunks)-1;i>0;i-- {seed=seed*1664525+1013904223;j:=int(seed%uint32(i+1));chunks[i],chunks[j]=chunks[j],chunks[i]}
 }
 var packed,plain []byte
 for _,chunk:=range chunks {packed=append(packed,chunk.packed...);plain=append(plain,chunk.plain...)}
 return packed,plain
}
