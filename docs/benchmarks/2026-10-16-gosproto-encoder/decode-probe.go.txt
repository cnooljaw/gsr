package benchmarks

import (
	"bytes"
	"encoding/json"
	before "github.com/cnooljaw/gosproto-before"
	"os"
	"strings"
	"testing"
)

// 仅复核完整矩阵出现大幅波动的三个解码项，不改变生产实现。
func BenchmarkPairedDecode(b *testing.B) {
	for _, sample := range representativeCases(b) {
		if sample.Name != "addressbook" {
			continue
		}
		ops, wires := representativeOperations(b, sample)
		_, newSproto := representativeMessages(sample.Type)
		packed := before.Pack(wires["sproto_encode"])
		pair := []representativeOperation{{"before_pdecode", func() error {
			message := newSproto()
			err := before.DecodePacked(packed, message)
			representativeSink = message
			return err
		}}}
		for _, op := range ops {
			if op.name == "sproto_pdecode" || op.name == "protobuf_decode" {
				pair = append(pair, op)
			}
		}
		runPaired(b, sample.Name, pair)
	}
	content, err := os.ReadFile("cases.json")
	if err != nil {
		b.Fatal(err)
	}
	var cases []benchmarkCase
	if err = json.Unmarshal(content, &cases); err != nil {
		b.Fatal(err)
	}
	for _, sample := range cases {
		if sample.Name != "ascii4k" {
			continue
		}
		text := strings.Repeat(sample.Pattern, sample.Repeat)
		old, now := beforeEchoCodecs(text), echoCodecs(text)
		var ops []representativeOperation
		for _, shape := range []string{"payload", "frame"} {
			left, right, control := old["sproto_"+shape], now["sproto_"+shape], now["protobuf_"+shape]
			wire, err := left.encode()
			if err != nil {
				b.Fatal(err)
			}
			same, err := right.encode()
			if err != nil || !bytes.Equal(wire, same) {
				b.Fatal("wire differs")
			}
			pbwire, err := control.encode()
			if err != nil {
				b.Fatal(err)
			}
			for _, op := range []struct {
				name   string
				decode func([]byte) (string, error)
				wire   []byte
			}{{"before_" + shape, left.decode, wire}, {"sproto_" + shape, right.decode, wire}, {"protobuf_" + shape, control.decode, pbwire}} {
				ops = append(ops, representativeOperation{op.name, func() error { var err error; decodedSink, err = op.decode(op.wire); return err }})
			}
		}
		runPaired(b, sample.Name, ops)
	}
}
