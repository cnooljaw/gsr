package benchmarks
import (
 "strings"
"bytes"
before "github.com/cnooljaw/gosproto-before"
sproto "github.com/cnooljaw/gosproto"
 "testing"
)
func pairOperations(t *testing.T,mode string) []pairOperation {
 result:=originalPairOperations(t,mode)
 for _,op:=range growthPairOperations(t,mode) {
  if !strings.HasSuffix(op.name,"/request_cancel") {result=append(result,op)}
 }
 for _,shape:=range unpackShapes() {
  raw:=shape.raw
  leftWire,rightWire:=before.Pack(raw),sproto.Pack(raw)
  if !bytes.Equal(leftWire,rightWire)||cap(leftWire)!=cap(rightWire) {t.Fatal("shape wire/cap",shape.name)}
  left:=func()error {encodedSink=before.Pack(raw);return nil}
  right:=func()error {encodedSink=sproto.Pack(raw);return nil}
  if mode=="self" {left=right}
  result=append(result,pairOperation{"shape/"+shape.name+"/pack",left,right})
 }
 return result
}
