package sproto

import (
	"os"
	"reflect"
	"runtime"
	"testing"
)

var entryPointerSink decodePointer
var entryBoolSink bool

func entryRecord(schema *SprotoType, value, record reflect.Value) decodePointer {
	if !schema.nativeInt64Pointer || record.Type() != schema.typ {
		return decodePointer{}
	}
	return decodePointer{value: value.UnsafePointer()}
}

func entryElement(schema *SprotoType, value reflect.Value) decodePointer {
	if !schema.nativeInt64Pointer || value.Elem().Type() != schema.typ {
		return decodePointer{}
	}
	return decodePointer{value: value.UnsafePointer()}
}

func entryAddress(schema *SprotoType, record reflect.Value) decodePointer {
	if !schema.nativeInt64Pointer || record.Type() != schema.typ {
		return decodePointer{}
	}
	return decodePointer{value: record.Addr().UnsafePointer()}
}

// 地址取得保持封装；布局事实由调用点基于同一对象即时计算，不跨调用缓存。
func entryBorrow(value reflect.Value, exactNative bool) decodePointer {
	if !exactNative {
		return decodePointer{}
	}
	return decodePointer{value: value.UnsafePointer()}
}

type entryNative struct {
	Padding [64]byte
	First   *int64 `sproto:"integer,0"`
	Last    *int64 `sproto:"integer,1"`
}

type entryForeign entryNative

type entryText struct {
	Text string `sproto:"string,0"`
}

type entryCase struct {
	name   string
	schema *SprotoType
	value  reflect.Value
}

func entryCases(tb testing.TB) []entryCase {
	tb.Helper()
	native, err := GetSprotoType(reflect.TypeFor[entryNative]())
	if err != nil {
		tb.Fatal(err)
	}
	text, err := GetSprotoType(reflect.TypeFor[entryText]())
	if err != nil {
		tb.Fatal(err)
	}
	return []entryCase{
		{"native", native, reflect.ValueOf(new(entryNative))},
		{"no_native", text, reflect.ValueOf(new(entryText))},
		{"type_mismatch", native, reflect.ValueOf(new(entryForeign))},
	}
}

// 三种真实入口状态下，候选返回与当前实现相同的受控地址或空能力。
func TestEntryCapabilities(t *testing.T) {
	for _, sample := range entryCases(t) {
		record := sample.value.Elem()
		want := messageDecodePointer(sample.schema, sample.value)
		actual := []decodePointer{
			entryRecord(sample.schema, sample.value, record),
			entryElement(sample.schema, sample.value),
			entryAddress(sample.schema, record),
			entryBorrow(sample.value, sample.schema.nativeInt64Pointer && record.Type() == sample.schema.typ),
		}
		for index, pointer := range actual {
			if pointer.value != want.value {
				t.Fatalf("%s variant=%d capability mismatch", sample.name, index)
			}
		}
	}
}

// 不含解析和对象分配；每次循环都执行即时类型检查，不预先计算 exactNative。
func BenchmarkDecodeEntry(b *testing.B) {
	for _, sample := range entryCases(b) {
		record := sample.value.Elem()
		operations := []struct {
			name string
			run  func(*testing.B)
		}{
			{"baseline", func(b *testing.B) {
				b.ReportAllocs()
				for range b.N {
					entryPointerSink = messageDecodePointer(sample.schema, sample.value)
				}
			}},
			{"record", func(b *testing.B) {
				b.ReportAllocs()
				for range b.N {
					entryPointerSink = entryRecord(sample.schema, sample.value, record)
				}
			}},
			{"element", func(b *testing.B) {
				b.ReportAllocs()
				for range b.N {
					entryPointerSink = entryElement(sample.schema, sample.value)
				}
			}},
			{"address", func(b *testing.B) {
				b.ReportAllocs()
				for range b.N {
					entryPointerSink = entryAddress(sample.schema, record)
				}
			}},
			{"split", func(b *testing.B) {
				b.ReportAllocs()
				for range b.N {
					entryPointerSink = entryBorrow(sample.value, sample.schema.nativeInt64Pointer && record.Type() == sample.schema.typ)
				}
			}},
		}
		if os.Getenv("GSR_AB_REVERSE") == "1" {
			for i, j := 0, len(operations)-1; i < j; i, j = i+1, j-1 {
				operations[i], operations[j] = operations[j], operations[i]
			}
		}
		for _, operation := range operations {
			b.Run(sample.name+"/"+operation.name, operation.run)
		}
		runtime.KeepAlive(sample.value)
	}
}

// 分项只用于解释入口成本，不从整体时间中减去或相加推导收益。
func BenchmarkEntryComponents(b *testing.B) {
	sample := entryCases(b)[0]
	record := sample.value.Elem()
	b.Run("pointer_type", func(b *testing.B) {
		b.ReportAllocs()
		for range b.N {
			entryBoolSink = sample.schema.nativeInt64Pointer && sample.value.Type().Elem() == sample.schema.typ
		}
	})
	b.Run("record_type", func(b *testing.B) {
		b.ReportAllocs()
		for range b.N {
			entryBoolSink = sample.schema.nativeInt64Pointer && record.Type() == sample.schema.typ
		}
	})
	b.Run("address", func(b *testing.B) {
		b.ReportAllocs()
		for range b.N {
			entryPointerSink = decodePointer{value: sample.value.UnsafePointer()}
		}
	})
	runtime.KeepAlive(sample.value)
}
