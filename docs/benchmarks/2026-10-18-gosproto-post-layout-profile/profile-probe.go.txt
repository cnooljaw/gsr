package benchmarks

import (
	"encoding/json"
	"reflect"
	"strings"
	"testing"

	sproto "github.com/cnooljaw/gosproto"
	"github.com/cnooljaw/gsr/examples/websocket-gate/gen/sproto/c2s"
)

// 每个输入分别观察纯 Encode、组合 body 与带 session 登记/取消的真实 RPC，不把采样耗时作为配对基线。
func BenchmarkCurrentEncoding(b *testing.B) {
	for _, name := range []string{"ascii32", "game_large", "int32_edges", "echo_ascii4k"} {
		b.Run(name, func(b *testing.B) {
			var message any
			if name == "ascii32" {
				text := strings.Repeat("A", 32)
				message = &c2s.EchoRequest{Text: &text}
			} else {
				for _, sample := range representativeCases(b) {
					if sample.Name == name {
						_, factory := representativeMessages(sample.Type)
						message = factory()
						if err := json.Unmarshal(sample.Data, message); err != nil {
							b.Fatal(err)
						}
					}
				}
			}
			if message == nil {
				b.Fatal("missing fixture")
			}
			b.Run("raw_body", func(b *testing.B) {
				if _, err := sproto.Encode(message); err != nil {
					b.Fatal(err)
				}
				b.ReportAllocs()
				b.ResetTimer()
				for b.Loop() {
					encoded, err := sproto.Encode(message)
					if err != nil {
						b.Fatal(err)
					}
					encodedSink = encoded
				}
			})
			b.Run("packed_body", func(b *testing.B) {
				if _, err := sproto.EncodePacked(message); err != nil {
					b.Fatal(err)
				}
				b.ReportAllocs()
				b.ResetTimer()
				for b.Loop() {
					encoded, err := sproto.EncodePacked(message)
					if err != nil {
						b.Fatal(err)
					}
					encodedSink = encoded
				}
			})
			b.Run("request_cancel", func(b *testing.B) {
				typ := reflect.TypeOf(message)
				rpc, err := sproto.NewRpc([]*sproto.Protocol{{Type: 1, Name: "frame", Request: typ, Response: typ}})
				if err != nil {
					b.Fatal(err)
				}
				if _, err := rpc.RequestEncode("frame", 1, message); err != nil {
					b.Fatal(err)
				}
				rpc.CancelSession(1)
				b.ReportAllocs()
				b.ResetTimer()
				for b.Loop() {
					encoded, err := rpc.RequestEncode("frame", 1, message)
					if err != nil {
						b.Fatal(err)
					}
					rpc.CancelSession(1)
					encodedSink = encoded
				}
			})
		})
	}
}
