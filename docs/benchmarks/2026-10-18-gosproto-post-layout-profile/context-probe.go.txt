type contextSnapshot struct {
	GC          uint64
	AllocBytes  uint64
	HeapBytes   uint64
	HeapGoal    uint64
	ScanHeap    uint64
	GCAssistCPU float64
	GCTotalCPU  float64
	UserCPU     float64
}

func contextCounters() []metrics.Sample {
	var counters []metrics.Sample
	for _, name := range []string{"/gc/cycles/total:gc-cycles", "/gc/heap/allocs:bytes", "/memory/classes/heap/objects:bytes", "/gc/heap/goal:bytes", "/gc/scan/heap:bytes", "/cpu/classes/gc/mark/assist:cpu-seconds", "/cpu/classes/gc/total:cpu-seconds", "/cpu/classes/user:cpu-seconds"} {
		counters = append(counters, metrics.Sample{Name: name})
	}
	return counters
}

func contextRead(counters []metrics.Sample) contextSnapshot {
	metrics.Read(counters)
	return contextSnapshot{
		GC: counters[0].Value.Uint64(), AllocBytes: counters[1].Value.Uint64(),
		HeapBytes: counters[2].Value.Uint64(), HeapGoal: counters[3].Value.Uint64(), ScanHeap: counters[4].Value.Uint64(),
		GCAssistCPU: counters[5].Value.Float64(), GCTotalCPU: counters[6].Value.Float64(), UserCPU: counters[7].Value.Float64(),
	}
}

func contextOperation(tb testing.TB) ([]pairOperation, pairOperation) {
	tb.Helper()
	name, live := os.Getenv("GSR_CONTEXT_CASE"), os.Getenv("GSR_CONTEXT_LIVE")
	if live != "full" && live != "isolated" {
		tb.Fatal("explicit full/isolated context required")
	}
	all := pairOperations(tb, "self")
	if len(all) != 70 {
		tb.Fatal("incomplete matrix")
	}
	var chosen pairOperation
	for _, operation := range all {
		if operation.name == name {
			chosen = operation
		}
	}
	if chosen.right == nil || reflect.ValueOf(chosen.left).Pointer() != reflect.ValueOf(chosen.right).Pointer() {
		tb.Fatal("missing case or A/A differs")
	}
	if live == "isolated" {
		all = nil
	}
	encodedSink = nil
	runtime.GC()
	return all, chosen
}

type contextBlock struct {
	Arm         string `json:"arm"`
	Calls       int    `json:"calls"`
	Nanoseconds int64  `json:"nanoseconds"`
	Before      contextSnapshot
	After       contextSnapshot
}

type contextQuartet struct {
	Round  int            `json:"round"`
	Index  int            `json:"index"`
	Blocks []contextBlock `json:"blocks"`
	Ratio  float64        `json:"rightOverLeft"`
}

// TestPackContextBlocks 检查同函数在两种夹具存活环境下的固定调用与GC计数，保留所有块。
func TestPackContextBlocks(t *testing.T) {
	all, operation := contextOperation(t)
	if runtime.GOMAXPROCS(0) != 1 {
		t.Fatal("GOMAXPROCS=1 required")
	}
	counters := contextCounters()
	initial := contextRead(counters)
	warm := time.Now()
	for time.Since(warm) < 50*time.Millisecond {
		for range 64 {
			if err := operation.right(); err != nil {
				t.Fatal(err)
			}
		}
	}
	afterWarmup := contextRead(counters)
	const calls = 16384
	quartets := make([]contextQuartet, 0, 24)
	for round := 0; round < 6; round++ {
		for index := 0; index < 4; index++ {
			q := contextQuartet{Round: round, Index: index, Blocks: make([]contextBlock, 0, 4)}
			var base []pairBlock
			for _, arm := range quartetOrder(round, index) {
				before := contextRead(counters)
				started := time.Now()
				for range calls {
					if err := operation.right(); err != nil {
						t.Fatal(err)
					}
				}
				elapsed := time.Since(started).Nanoseconds()
				after := contextRead(counters)
				if after.GC < before.GC || after.AllocBytes < before.AllocBytes {
					t.Fatal("nonmonotonic counters")
				}
				q.Blocks = append(q.Blocks, contextBlock{Arm: string(arm), Calls: calls, Nanoseconds: elapsed, Before: before, After: after})
				base = append(base, pairBlock{Arm: string(arm), Calls: calls, Nanoseconds: elapsed})
			}
			var err error
			q.Ratio, err = quartetRatio(base)
			if err != nil {
				t.Fatal(err)
			}
			quartets = append(quartets, q)
		}
	}
	runtime.KeepAlive(all)
	result := struct {
		Case        string
		Context     string
		Function    string
		Initial     contextSnapshot
		AfterWarmup contextSnapshot
		Quartets    []contextQuartet
	}{operation.name, os.Getenv("GSR_CONTEXT_LIVE"), runtime.FuncForPC(reflect.ValueOf(operation.right).Pointer()).Name(), initial, afterWarmup, quartets}
	encoded, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("GSR_CONTEXT_OUTPUT"), append(encoded, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
	t.Logf("%s %s:96 blocks x16384 calls;exact same A/A target", result.Case, result.Context)
}

// BenchmarkPackContext 只采固定存活环境的CPU归因，不作为两种实现的性能对照。
func BenchmarkPackContext(b *testing.B) {
	all, operation := contextOperation(b)
	b.ReportAllocs()
	b.ResetTimer()
	for b.Loop() {
		if err := operation.right(); err != nil {
			b.Fatal(err)
		}
	}
	runtime.KeepAlive(all)
}
