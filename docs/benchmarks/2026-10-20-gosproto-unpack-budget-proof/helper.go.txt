// unpackedSizeWithinBudget只由len(src)<=maxBytes/8证明后的入口调用。
// 普通word/零mask/FF段均满足produced<=8*consumed，故有效前缀之和不会超预算或int。
// 此处仍完整验证所有输入段；不能用于未证明预算的外部输入。
func unpackedSizeWithinBudget(src []byte) (int, error) {
 size:=0
 for len(src)>0 {
  header:=src[0]
  consumed,produced:=1,8
  if header==0 {
   if len(src)>=2 && src[1]==0 {
    run:=2+zeroHeaderPrefix(src[2:])
    consumed,produced=run,run*8
   }
  }else if header==255 {
   if len(src)<2 {return 0,ErrUnpack}
   produced=(int(src[1])+1)*8
   consumed=2+produced
  }else {
   consumed+=bits.OnesCount8(header)
  }
  if consumed>len(src) {return 0,ErrUnpack}
  size+=produced
  src=src[consumed:]
 }
 return size,nil
}
