package sproto

import (
	"bytes"
	"errors"
	"testing"
)

// 验证全部 mask 在短长混合零段及精确边界下解包正确，结果独占且失败不泄露部分输出。
func checkWordBounds(t *testing.T, unpack func([]byte, int) ([]byte, error)) {
	for mask := 0; mask < 256; mask++ {
		var word [8]byte
		packedWord := []byte{byte(mask)}
		if mask == 255 {
			packedWord = append(packedWord, 0)
		}
		for index := range word {
			if mask&(1<<index) != 0 {
				word[index] = byte(index + 1)
				packedWord = append(packedWord, word[index])
			}
		}
		for _, count := range []int{1, 2, 7, 8, 9, 255, 256, 257} {
			for _, prefix := range []int{0, 1, 9} {
				packed := append(make([]byte, prefix), bytes.Repeat(packedWord, count)...)
				packed = append(packed, 0, 0)
				want := append(make([]byte, prefix*8), bytes.Repeat(word[:], count)...)
				want = append(want, make([]byte, 16)...)
				for _, spare := range []int{0, 32} {
					input := make([]byte, len(packed), len(packed)+spare)
					copy(input, packed)
					output, err := unpack(input, len(want))
					if err != nil || !bytes.Equal(output, want) || cap(output) != len(want) {
						t.Fatalf("mask=%d count=%d prefix=%d spare=%d err=%v", mask, count, prefix, spare, err)
					}
					if len(output) > 0 {
						output[0] ^= 255
					}
					if !bytes.Equal(input, packed) {
						t.Fatal("output aliases input")
					}
					again, err := unpack(input, len(want))
					if err != nil || !bytes.Equal(again, want) {
						t.Fatal("output aliases another result")
					}
					if len(output) > 0 && output[0] != want[0]^255 {
						t.Fatal("later result overwrote earlier output")
					}
					output, err = unpack(input, len(want)-1)
					if output != nil || !errors.Is(err, ErrLimit) {
						t.Fatalf("short budget: %v", err)
					}
					broken := append(append([]byte{}, input...), 128)
					output, err = unpack(broken, len(want)+8)
					if output != nil || !errors.Is(err, ErrUnpack) {
						t.Fatalf("truncated byte: %v", err)
					}
				}
			}
		}
	}
}

// 验证三个同进程实验入口均保持全部mask、预算、截断及独占结果契约。
func TestUnpackExperimentBounds(t *testing.T) {
	for name, fn := range map[string]func([]byte, int) ([]byte, error){"baseline": UnpackWithLimit, "clone": ExperimentUnpackClone, "candidate": ExperimentUnpackCandidate} {
		t.Run(name, func(t *testing.T) { checkWordBounds(t, fn) })
	}
}

// 验证输出上界证明门槛两侧保持完整字节、预算、错误顺序及最大int预算安全。
func checkBudgetProof(t *testing.T, unpack func([]byte, int) ([]byte, error)) {
	check := func(input, want []byte, limit int, wantErr error) {
		before := append([]byte(nil), input...)
		got, err := unpack(input, limit)
		if !errors.Is(err, wantErr) || (wantErr != nil && got != nil) {
			t.Fatalf("input %x limit %d: %x %v want %v", input, limit, got, err, wantErr)
		}
		if err == nil && (!bytes.Equal(got, want) || cap(got) != len(want)) {
			t.Fatalf("bytes/cap %x != %x", got, want)
		}
		if !bytes.Equal(input, before) {
			t.Fatal("input changed")
		}
	}
	maximum := int(^uint(0) >> 1)
	for _, count := range []int{1, 2, 7, 8, 9, 255, 256, 257, 4096} {
		input := make([]byte, count)
		want := make([]byte, count*8)
		check(input, nil, count*8-1, ErrLimit)
		for _, limit := range []int{count * 8, count*8 + 1, maximum} {
			check(input, want, limit, nil)
		}
	}
	for mask := 1; mask < 255; mask++ {
		packed := []byte{byte(mask)}
		word := make([]byte, 8)
		for i := range word {
			if mask&(1<<i) != 0 {
				word[i] = byte(i + 1)
				packed = append(packed, word[i])
			}
		}
		input := append(append([]byte(nil), packed...), packed...)
		want := append(append([]byte(nil), word...), word...)
		for _, limit := range []int{len(input)*8 - 1, len(input) * 8, len(input)*8 + 1, maximum} {
			check(input, want, limit, nil)
		}
		broken := append(append([]byte(nil), input...), 128)
		for _, limit := range []int{len(broken)*8 - 1, len(broken) * 8, len(broken)*8 + 1, maximum} {
			check(broken, nil, limit, ErrUnpack)
		}
		// 严格回退路径不能为看到后面的坏包而掩盖已经发生的预算失败。
		check(broken, nil, 7, ErrLimit)
		check(packed[:len(packed)-1], nil, 1, ErrUnpack)
	}
	for _, count := range []byte{0, 1, 255} {
		size := (int(count) + 1) * 8
		input := append([]byte{255, count}, bytes.Repeat([]byte{7}, size)...)
		want := bytes.Repeat([]byte{7}, size)
		check(input, nil, size-1, ErrLimit)
		for _, limit := range []int{size, len(input)*8 - 1, len(input) * 8, len(input)*8 + 1, maximum} {
			check(input, want, limit, nil)
		}
		for _, cut := range []int{1, 2, len(input) - 1} {
			broken := input[:cut]
			for _, limit := range []int{1, len(broken)*8 - 1, len(broken) * 8, len(broken)*8 + 1, maximum} {
				check(broken, nil, limit, ErrUnpack)
			}
		}
	}
	check(nil, nil, 1, nil)
	check(nil, nil, 0, ErrInvalidLimits)
}

// 验证三入口在上界证明门槛两侧均保持预算、边界和错误顺序。
func TestUnpackExperimentProof(t *testing.T) {
	for name, fn := range map[string]func([]byte, int) ([]byte, error){"baseline": UnpackWithLimit, "clone": ExperimentUnpackClone, "candidate": ExperimentUnpackCandidate} {
		t.Run(name, func(t *testing.T) { checkBudgetProof(t, fn) })
	}
}
