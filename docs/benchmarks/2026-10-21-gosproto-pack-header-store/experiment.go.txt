package benchmarks

import (
	"bytes"
	"encoding/json"
	"math"
	"os"
	"runtime"
	"testing"
	"time"

	sp "github.com/cnooljaw/gosproto"
)

type unpackBlock struct {
	Arm                   string `json:"arm"`
	Calls                 int    `json:"calls"`
	Nanoseconds           int64  `json:"nanoseconds"`
	Bytes, Allocs         uint64
	GC                    uint32
	PauseNs               uint64
	HeapBefore, HeapAfter uint64
}

type unpackQuartet struct {
	Round, Index int
	Order        string
	Blocks       []unpackBlock
	Ratio        float64
}

// TestUnpackInProcessExperiment 在同一模块/进程交错操作，保留全部GC、分配与时间块。
func TestUnpackInProcessExperiment(t *testing.T) {
	if runtime.GOMAXPROCS(0) != 1 {
		t.Fatal("GOMAXPROCS1 required")
	}
	name, mode := os.Getenv("GSR_CASE"), os.Getenv("GSR_MODE")
	var packed []byte
	var message any
	for _, sample := range representativeCases(t) {
		if sample.Name != name {
			continue
		}
		_, wires := representativeOperations(t, sample)
		checkRepresentativeWire(t, sample, wires)
		packed = wires["sproto_encode"]
		_, newMessage := representativeMessages(sample.Type)
		message = newMessage()
		if err := json.Unmarshal(sample.Data, message); err != nil {
			t.Fatal(err)
		}
	}
	if packed == nil {
		var word [8]byte
		density := 4
		if name == "density7" {
			density = 7
		} else if name != "density4" {
			t.Fatal(name)
		}
		for i := 0; i < density; i++ {
			word[i] = byte(i + 1)
		}
		packed = bytes.Repeat(word[:], 512)
	}
	operation := os.Getenv("GSR_OP")
	left, right := sp.ExperimentPackBaseline, sp.ExperimentPackCandidate
	le, re := sp.ExperimentPEncodeBaseline, sp.ExperimentPEncodeCandidate
	switch mode {
	case "self":
		right = left
		re = le
	case "clone":
		right = sp.ExperimentPackClone
		re = sp.ExperimentPEncodeClone
	case "pass1", "pass2":
	default:
		t.Fatal(mode)
	}
	beforeInput := append([]byte(nil), packed...)
	expected := sp.Pack(packed)
	for _, fn := range []func([]byte) []byte{left, right} {
		got := fn(packed)
		if !bytes.Equal(got, expected) || cap(got) != cap(expected) {
			t.Fatal("wire/cap")
		}
	}
	if operation == "pencode" {
		want, err := sp.EncodePacked(message)
		if err != nil {
			t.Fatal(err)
		}
		for _, fn := range []func(any) ([]byte, error){le, re} {
			got, err := fn(message)
			if err != nil || !bytes.Equal(got, want) || cap(got) != cap(want) {
				t.Fatal("pencode wire/cap", err)
			}
		}
	}
	makeRun := func(fn func([]byte) []byte, enc func(any) ([]byte, error)) func() {
		return func() {
			if operation == "pack" {
				encodedSink = fn(packed)
				return
			}
			if operation != "pencode" {
				t.Fatal(operation)
			}
			var err error
			encodedSink, err = enc(message)
			if err != nil {
				t.Fatal(err)
			}
		}
	}
	runs := map[string]func(){"A": makeRun(left, le), "B": makeRun(right, re)}
	runtime.GC()
	for _, arm := range []string{"A", "B"} {
		start := time.Now()
		for time.Since(start) < 50*time.Millisecond {
			for range 64 {
				runs[arm]()
			}
		}
	}
	start := time.Now()
	for range 256 {
		runs["A"]()
	}
	calls := int(float64(20*time.Millisecond) / float64(time.Since(start)) * 256)
	calls = max(1, min(calls, 1<<20))
	blocks := []unpackQuartet{}
	for round := 0; round < 6; round++ {
		for quartet := 0; quartet < 4; quartet++ {
			order := "ABBA"
			if (round+quartet)%2 != 0 {
				order = "BAAB"
			}
			q := unpackQuartet{Round: round, Index: quartet, Order: order}
			logRatio := 0.0
			for _, label := range order {
				arm := string(label)
				var before, after runtime.MemStats
				runtime.ReadMemStats(&before)
				run := runs[arm]
				start := time.Now()
				for range calls {
					run()
				}
				ns := time.Since(start).Nanoseconds()
				runtime.ReadMemStats(&after)
				block := unpackBlock{Arm: arm, Calls: calls, Nanoseconds: ns, Bytes: after.TotalAlloc - before.TotalAlloc, Allocs: after.Mallocs - before.Mallocs, GC: after.NumGC - before.NumGC, PauseNs: after.PauseTotalNs - before.PauseTotalNs, HeapBefore: before.HeapAlloc, HeapAfter: after.HeapAlloc}
				q.Blocks = append(q.Blocks, block)
				sign := 1.0
				if arm == "A" {
					sign = -1
				}
				logRatio += sign * math.Log(float64(ns)/float64(calls))
			}
			q.Ratio = math.Exp(logRatio / 2)
			blocks = append(blocks, q)
		}
	}
	if !bytes.Equal(packed, beforeInput) {
		t.Fatal("input modified")
	}
	raw, err := json.MarshalIndent(struct {
		Name, Operation, Mode string
		CallsPerBlock         int
		Quartets              []unpackQuartet
	}{name, operation, mode, calls, blocks}, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("GSR_OUTPUT"), append(raw, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
	runtime.KeepAlive(runs)
}
