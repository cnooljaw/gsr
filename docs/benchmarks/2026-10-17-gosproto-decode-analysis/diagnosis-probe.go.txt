package benchmarks

import (
	"encoding/json"
	baseline "github.com/cnooljaw/gosproto-diag-baseline"
	fixed "github.com/cnooljaw/gosproto-diag-fixed"
	typed "github.com/cnooljaw/gosproto-diag-typed"
	"os"
	"reflect"
	"testing"
)

type diagnosisVariant struct {
	name    string
	factory func() any
	decode  func([]byte, any) (int, error)
	pdecode func([]byte, any) error
}

func diagnosisOperations(tb testing.TB, sample representativeCase) []representativeOperation {
	original, wires := representativeOperations(tb, sample)
	checkRepresentativeWire(tb, sample, wires)
	var ops []representativeOperation
	for _, op := range original {
		if op.name == "sproto_decode" || op.name == "sproto_pdecode" || op.name == "protobuf_decode" {
			ops = append(ops, op)
		}
	}
	if sample.Type != "GameState" {
		return ops
	}
	variants := []diagnosisVariant{
		{"generic", func() any { return new(baseline.DiagnosticGameState) }, baseline.Decode, baseline.DecodePacked},
		{"fixed_reflect", func() any { return new(fixed.DiagnosticGameState) }, fixed.Decode, fixed.DecodePacked},
		{"fixed_typed", func() any { return new(typed.DiagnosticGameState) }, typed.Decode, typed.DecodePacked},
	}
	for _, v := range variants {
		want := v.factory()
		if err := json.Unmarshal(sample.Data, want); err != nil {
			tb.Fatal(err)
		}
		raw, err := baseline.Encode(want)
		if err != nil || !reflect.DeepEqual(raw, wires["sproto_encode"]) {
			tb.Fatal(v.name, "wire", err)
		}
		result := v.factory()
		used, err := v.decode(raw, result)
		if err != nil || used != len(raw) || !reflect.DeepEqual(result, want) {
			tb.Fatal(v.name, "decode", err)
		}
		result = v.factory()
		if err := v.pdecode(wires["sproto_pack"], result); err != nil || !reflect.DeepEqual(result, want) {
			tb.Fatal(v.name, "pdecode", err)
		}
		ops = append(ops, representativeOperation{v.name + "_decode", func() error {
			message := v.factory()
			_, err := v.decode(wires["sproto_encode"], message)
			representativeSink = message
			return err
		}}, representativeOperation{v.name + "_pdecode", func() error {
			message := v.factory()
			err := v.pdecode(wires["sproto_pack"], message)
			representativeSink = message
			return err
		}})
	}
	return ops
}

// 固定探针和通用入口消费同一 wire，并恢复相同的 optional 值、数组和所有内容。
func TestDecodeDiagnosis(t *testing.T) {
	for _, s := range representativeCases(t) {
		diagnosisOperations(t, s)
	}
}

// 仅比较构造后的完整调用，分段与微基准不相加。
func BenchmarkDecodeDiagnosis(b *testing.B) {
	for _, sample := range representativeCases(b) {
		if sample.Type == "EchoRequest" {
			continue
		}
		ops := diagnosisOperations(b, sample)
		if os.Getenv("GSR_AB_REVERSE") == "1" {
			for i, j := 0, len(ops)-1; i < j; i, j = i+1, j-1 {
				ops[i], ops[j] = ops[j], ops[i]
			}
		}
		for _, op := range ops {
			b.Run(sample.Name+"/"+op.name, func(b *testing.B) {
				b.ReportAllocs()
				b.ResetTimer()
				for range b.N {
					if err := op.run(); err != nil {
						b.Fatal(err)
					}
				}
			})
		}
	}
}

// 构造微基准只用于分解成本，不是完整 decoder，也不代表可发布优化。
func BenchmarkConstructDiagnosis(b *testing.B) {
	typ := reflect.TypeFor[baseline.DiagnosticItem]()
	sliceType := reflect.TypeFor[[]*baseline.DiagnosticItem]()
	ops := []struct {
		name string
		run  func() any
	}{
		{"reflect", func() any {
			result := reflect.MakeSlice(sliceType, 128, 128)
			for i := 0; i < 128; i++ {
				item := reflect.New(typ)
				for f := 0; f < 3; f++ {
					v := new(int64)
					*v = int64(i + f)
					item.Elem().Field(f).Set(reflect.ValueOf(v))
				}
				result.Index(i).Set(item)
			}
			return result.Interface()
		}},
		{"typed", func() any {
			result := make([]*baseline.DiagnosticItem, 128)
			for i := range result {
				a, c, p := new(int64), new(int64), new(int64)
				*a, *c, *p = int64(i), int64(i+1), int64(i+2)
				result[i] = &baseline.DiagnosticItem{Id: a, Count: c, Power: p}
			}
			return result
		}},
	}
	if os.Getenv("GSR_AB_REVERSE") == "1" {
		ops[0], ops[1] = ops[1], ops[0]
	}
	for _, op := range ops {
		b.Run(op.name, func(b *testing.B) {
			b.ReportAllocs()
			for range b.N {
				representativeSink = op.run()
			}
		})
	}
}
