package sproto

import (
	"math"
	"reflect"
	"unsafe"
)

func experimentStringBaseline(field *SprotoField, value reflect.Value, output []byte, layout *encodeLayout, trusted bool) int {
	if field.Array {
		if value.Len() == 0 {
			return 0
		}
		offset := 0
		switch field.Wire {
		case WireVarintName:
			width := layout.next()
			if field.int64Slice {
				return encodeInt64Slice(*value.Addr().Interface().(*[]int64), width, output)
			}
			output[0] = byte(width)
			for index := 0; index < value.Len(); index++ {
				number := integerBits(field, value.Index(index))
				if width == 4 {
					writeUint32(output[1+index*4:], uint32(number))
				} else {
					writeUint64(output[1+index*8:], number)
				}
			}
			offset = 1 + width*value.Len()
		case WireDoubleName:
			output[0] = 8
			offset = 1
			for index := 0; index < value.Len(); index++ {
				writeUint64(output[offset:], math.Float64bits(value.Index(index).Float()))
				offset += 8
			}
		case WireBooleanName:
			for index := 0; index < value.Len(); index++ {
				if value.Index(index).Bool() {
					output[index] = 1
				} else {
					output[index] = 0
				}
			}
			offset = value.Len()
		case WireBytesName:
			for index := 0; index < value.Len(); index++ {
				text := value.Index(index).String()
				writeUint32(output[offset:], uint32(len(text)))
				offset += 4 + copy(output[offset+4:], text)
			}
		case WireStructName:
			childBinding := bindEncodeChild(field, value, trusted)
			integers := childBinding.native && field.st.int64Record && field.st.nativeIntegerFields()
			var pointers []unsafe.Pointer
			if integers {
				pointers = integerRecordPointers(field.st, value)
			}
			for index := 0; index < value.Len(); index++ {
				var length int
				if integers {
					var pointer encodePointer
					if pointers != nil {
						pointer = encodePointer{value: pointers[index]}
					} else {
						pointer = messageEncodePointer(value.Index(index), childBinding)
					}
					length = encodeIntegerRecordInto(field.st, pointer, output[offset+4:], layout)
				} else {
					length = encodeMessageInto(field.st, value.Index(index), output[offset+4:], layout, childBinding)
				}
				writeUint32(output[offset:], uint32(length))
				offset += 4 + length
			}
		}
		return offset
	}
	return encodeMessageInto(field.st, value, output, layout, bindEncodeChild(field, value, trusted))
}

				pointers = integerRecordPointers(field.st, value)
			}
			if integers && value.Len() >= 8 {
				return experimentIntegerArray(field, value, output, layout, childBinding, pointers)
			}
			for index := 0; index < value.Len(); index++ {
				var length int
				if integers {
					var pointer encodePointer
					if pointers != nil {
						pointer = encodePointer{value: pointers[index]}

					}
					length = experimentIntegerRecord(field.st, pointer, output[offset+4:], layout.next())
				} else {
					length = encodeMessageInto(field.st, value.Index(index), output[offset+4:], layout, childBinding)
				}
				writeUint32(output[offset:], uint32(length))
				offset += 4 + length
			}
		}
		return offset
	}
	return encodeMessageInto(field.st, value, output, layout, bindEncodeChild(field, value, trusted))
}

func experimentIntegerRecord(schema *SprotoType, pointer encodePointer, output []byte, count int) int {
	writeUint16(output, uint16(count))
	if pointer.value == nil {
		return 2
	}
	headerOffset, bodyOffset, previous := 2, 2+count*2, -1
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		bits, present := pointer.integer(field.encOffset)
		if !present {
			continue
		}
		if skip := skipTag(previous, field.Tag); skip > 0 {
			writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], skip)
			headerOffset += 2
		}
		// inline 整数只写头部，避免提前构造未使用的内容切片。
		var header uint16
		var length int
		var body []byte
		if bits <= MaxEmbeddedInt {
			header = uint16((bits + 1) * 2)
		} else {
			body = output[bodyOffset:]
			header, length = encodeIntegerInto(bits, body)
		}
		writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], header)
		headerOffset += 2
		previous = field.Tag
		if header == 0 {
			writeUint32(body, uint32(length))
			bodyOffset += 4 + length
		}
	}
	return bodyOffset
}

// experimentIntegerArray 只消费本次预检形成的连续整数记录布局，不包含嵌套布局项。
func experimentIntegerArray(field *SprotoField, value reflect.Value, output []byte, layout *encodeLayout, binding encodeBinding, pointers []unsafe.Pointer) int {
	count := value.Len()
	index, offset := 0, 0
	for index < count {
		cursor := layout.cursor
		var headers []uint32
		if cursor < len(layout.local) {
			end := min(len(layout.local), cursor+count-index)
			headers = layout.local[cursor:end:end]
		} else {
			start := cursor - len(layout.local)
			end := start + count - index
			headers = layout.extra[start:end:end]
		}
		layout.cursor += len(headers)
		for _, header := range headers {
			var pointer encodePointer
			if pointers != nil {
				pointer = encodePointer{value: pointers[index]}
			} else {
				pointer = messageEncodePointer(value.Index(index), binding)
			}
			length := experimentIntegerRecord(field.st, pointer, output[offset+4:], int(header))
			writeUint32(output[offset:], uint32(length))
			offset += 4 + length
			index++
		}
	}
	return offset
}

var experimentStringSelected func(*SprotoField, reflect.Value, []byte, *encodeLayout, bool) int

func init() { experimentStringSelected = experimentStringBaseline }

// ExperimentStringMode 仅在实验归档中选择编码数组路径。
func ExperimentStringMode(mode string) {
	switch mode {
	case "baseline":
		experimentStringSelected = experimentStringBaseline
	case "clone":
		experimentStringSelected = experimentStringClone
	case "candidate":
		experimentStringSelected = experimentStringCandidate
	default:
		panic("invalid experiment mode")
	}
}
