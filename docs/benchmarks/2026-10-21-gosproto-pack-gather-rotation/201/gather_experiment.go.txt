package sproto

import (
	"encoding/binary"
	"math/bits"
)

func packGatherCandidate(packed, src []byte) []byte {
	offset := 0
	for len(src) >= 8 {
		word := binary.LittleEndian.Uint64(src)
		nonzero := (((word & 0x7f7f7f7f7f7f7f7f) + 0x7f7f7f7f7f7f7f7f) | word) & 0x8080808080808080
		header := uint8((nonzero * 0x0002040810204081) >> 56)
		if header == 0xff {
			consumed := ffRunBytes(src)
			writeFF(packed[offset:], src, consumed)
			offset += 2 + ((consumed + 7) &^ 7)
			src = src[consumed:]
			continue
		}
		// 去掉至多两个置位后是否为零，比普通路径完整计数更直接。
		remaining := header & (header - 1)
		if remaining&(remaining-1) == 0 {
			// 长度与容量均固定且非空，避免每个 word 重算动态切片的指针掩码。
			dst := packed[offset : offset+3 : offset+3]
			// 已确认至多两个非零字节，直接写固定位置，避免通用循环和可变目标索引。
			_ = dst[2]
			dst[0] = header
			index := 1
			if header != 0 {
				shift := bits.TrailingZeros8(header) * 8
				dst[1] = byte(word >> shift)
				index = 2
				if remaining != 0 {
					shift = bits.TrailingZeros8(remaining) * 8
					dst[2] = byte(word >> shift)
					index = 3
				}
			}
			offset += index
		} else {
			dst := packed[offset : offset+9 : offset+9]
			// Pack 预留八字节尾空间，零字节暂写的位置会被下一个有效字节覆盖，不进入返回切片。
			_ = dst[8]
			// 固定八字节视图越过 mask，非零字节位置直接使用 0～7 的前缀计数。
			payload := dst[1:9]
			// 0/1 字节标志相乘得到前缀和，每个和至多为 8，不跨字节进位。
			// FF 已在前面处理；各目标位置只依赖对应前缀，不串行累加 index。
			ranks := (nonzero >> 7) * 0x0101010101010101
			payload[0] = byte(word)
			payload[int(ranks&7)] = byte(word >> 8)
			payload[int((ranks>>8)&7)] = byte(word >> 16)
			payload[int((ranks>>16)&7)] = byte(word >> 24)
			payload[int((ranks>>24)&7)] = byte(word >> 32)
			payload[int((ranks>>32)&7)] = byte(word >> 40)
			payload[int((ranks>>40)&7)] = byte(word >> 48)
			payload[int((ranks>>48)&7)] = byte(word >> 56)
			dst[0] = header
			offset += 1 + int(ranks>>56)
		}
		src = src[8:]
	}
	if len(src) > 0 {
		offset += packSegment(packed[offset:], src, 0)
	}
	return packed[:offset]
}

func ExperimentPackBaseline(src []byte) []byte {
	return packInto(make([]byte, packedBufferSize(len(src))), src)
}
func ExperimentPEncodeBaseline(m any) ([]byte, error) {
	out, body, err := encodeBufferWithLimits(m, Limits{}, true, nil)
	if err != nil {
		return nil, err
	}
	return packInto(out, body), nil
}
