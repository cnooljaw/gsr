package sproto_test

import (
 "testing"
 sp "github.com/cnooljaw/gosproto"
)

// 验证公开解码入口在每个parser模式下都保持指针独占、预算和描述替换契约。
func TestRecordScalarExternalContracts(t *testing.T) {
 defer sp.ExperimentStringMode("baseline")
 for _,mode:=range []string{"baseline","clone","candidate"} {
  sp.ExperimentStringMode(mode)
  t.Run(mode,func(t *testing.T) {
   t.Run("nativeOwnership",TestNativePointerDecodeOwnership)
   t.Run("nativePaddingGC",TestNativePointerDecodePaddingAndGC)
   t.Run("budgetsTags",TestIntegerRecordArrayBudgetsAndTags)
   t.Run("replacement",TestDecodeIntegerRecordFieldsReplacement)
  })
 }
}
