package benchmarks

import (
	"bytes"
	"reflect"
	"testing"

	sproto "github.com/cnooljaw/gosproto"
	before "github.com/cnooljaw/gosproto-before"
)

type rpcFuzzPayload struct {
	Binary []byte `sproto:"string,0"`
}

// FuzzRPCHeader 比较两版本真实三种 RPC 输出，覆盖任意 int32 type、正 session 与字节 body。
func FuzzRPCHeader(f *testing.F) {
	for _, seed := range []struct {
		code, session int32
		body          []byte
	}{
		{0, 1, nil}, {-2147483648, 2147483647, bytes.Repeat([]byte{255}, 2049)},
		{-1, 32767, []byte{0, 1, 2, 3, 4, 5, 6, 7}}, {32766, 32766, bytes.Repeat([]byte{0}, 4096)},
		{32767, 1, bytes.Repeat([]byte{1}, 2048)}, {2147483647, 2147483647, []byte{255}},
	} {
		f.Add(seed.code, seed.session, seed.body)
	}
	// 固定全 mask、尾块与 FF 段邻域，种子模式也能完成足量可重复的差分。
	for mask := 0; mask < 256; mask++ {
		for _, size := range []int{0, 1, 7, 8, 15, 16, 31, 32, 2047, 2048, 2049, 4096} {
			body := make([]byte, size)
			for index := range body {
				if mask&(1<<(index%8)) != 0 {
					body[index] = byte((index*31+mask)%255 + 1)
				}
			}
			code := int32(uint32(mask)*0x9e3779b9 ^ uint32(size))
			session := int32((uint32(mask+1)*0x85ebca6b ^ uint32(size)) & 0x7fffffff)
			f.Add(code, session, body)
		}
	}
	f.Fuzz(func(t *testing.T, code, session int32, data []byte) {
		if len(data) > 4096 {
			return
		}
		session &= 2147483647
		if session == 0 {
			session = 1
		}
		message := &rpcFuzzPayload{Binary: data}
		snapshot := bytes.Clone(data)
		typ := reflect.TypeOf(message)
		for _, call := range []bool{false, true} {
			var response reflect.Type
			if call {
				response = typ
			}
			old, err := before.NewRpc([]*before.Protocol{{Type: code, Name: "frame", Request: typ, Response: response}})
			if err != nil {
				t.Fatal(err)
			}
			current, err := sproto.NewRpc([]*sproto.Protocol{{Type: code, Name: "frame", Request: typ, Response: response}})
			if err != nil {
				t.Fatal(err)
			}
			left, err := old.RequestEncode("frame", session, message)
			if err != nil {
				t.Fatal(err)
			}
			right, err := current.RequestEncode("frame", session, message)
			if err != nil || !bytes.Equal(left, right) || cap(left) != cap(right) {
				t.Fatalf("request code=%d session=%d call=%v err=%v", code, session, call, err)
			}
			request := right
			saved := bytes.Clone(right)
			left, err = old.ResponseEncode("frame", session, message)
			if err != nil {
				t.Fatal(err)
			}
			right, err = current.ResponseEncode("frame", session, message)
			if err != nil || !bytes.Equal(left, right) || cap(left) != cap(right) || !bytes.Equal(request, saved) {
				t.Fatalf("response code=%d session=%d call=%v err=%v", code, session, call, err)
			}
			old.CancelSession(session)
			current.CancelSession(session)
		}
		if !bytes.Equal(data, snapshot) {
			t.Fatal("input modified")
		}
	})
}
