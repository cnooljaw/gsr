package benchmarks

import (
	"encoding/json"
	"os"
	"runtime"
	"runtime/pprof"
	"testing"
	"time"
)

func currentProfileOperation(t testing.TB, name string) pairOperation {
	t.Helper()
	operations, err := selectPairs(pairOperations(t, "self"), []string{name})
	if err != nil || len(operations) != 1 {
		t.Fatalf("profile selection %q: %v", name, err)
	}
	return operations[0]
}

// BenchmarkCurrentOperation 精确采一项；CPU仅覆盖预热后的循环，内存另用固定次数进程。
func BenchmarkCurrentOperation(b *testing.B) {
	name := os.Getenv("GSR_PROFILE_CASE")
	output := os.Getenv("GSR_PROFILE_META")
	kind := os.Getenv("GSR_PROFILE_KIND")
	if name == "" || output == "" || (kind != "cpu" && kind != "allocation") {
		b.Fatal("explicit profile case/meta/kind required")
	}
	if runtime.GOMAXPROCS(0) != 1 {
		b.Fatal("profile requires GOMAXPROCS=1")
	}
	operation := currentProfileOperation(b, name)
	encodedSink = nil
	runtime.GC()
	stages := make([]isolatedSnapshot, 0, 3)
	stages = append(stages, isolatedMetrics("after_gc"))
	start := time.Now()
	for time.Since(start) < 50*time.Millisecond {
		for range 64 {
			if err := operation.right(); err != nil {
				b.Fatal(err)
			}
		}
	}
	stages = append(stages, isolatedMetrics("before_loop"))
	var cpu *os.File
	if kind == "cpu" {
		var err error
		cpu, err = os.Create(os.Getenv("GSR_PROFILE_CPU"))
		if err != nil {
			b.Fatal(err)
		}
		if err := pprof.StartCPUProfile(cpu); err != nil {
			b.Fatal(err)
		}
	}
	b.ReportAllocs()
	b.ResetTimer()
	for b.Loop() {
		if err := operation.right(); err != nil {
			b.Fatal(err)
		}
	}
	b.StopTimer()
	if cpu != nil {
		pprof.StopCPUProfile()
		if err := cpu.Close(); err != nil {
			b.Fatal(err)
		}
	}
	stages = append(stages, isolatedMetrics("after_profile"))
	runtime.KeepAlive(operation)
	contents, err := json.MarshalIndent(struct {
		Case   string
		Kind   string
		Calls  int
		Stages []isolatedSnapshot
	}{name, kind, b.N, stages}, "", "  ")
	if err != nil {
		b.Fatal(err)
	}
	if err := os.WriteFile(output, append(contents, '\n'), 0644); err != nil {
		b.Fatal(err)
	}
}
