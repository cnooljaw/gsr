package sproto

import (
 "encoding/binary"
 "errors"
 "testing"
)

type headerCursorEmpty struct { Marker int }

// 验证头部顺序读取保持完整消费、未知字段、预算、截断和纯跳跃链的原有语义。
func TestDecodeHeaderCursorBoundaries(t *testing.T) {
 for _,count:=range []int{0,1,2,7,8,9,16,255,256,257,65535} {
  data:=make([]byte,2+count*2)
  binary.LittleEndian.PutUint16(data,uint16(count))
  for i:=0;i<count;i++ {binary.LittleEndian.PutUint16(data[2+i*2:],2)}
  value:=headerCursorEmpty{Marker:123}
  used,err:=DecodeWithLimits(data,&value,Limits{MaxElements:max(1,count)})
  if err!=nil || used!=len(data) || value.Marker!=0 {t.Fatalf("count=%d used=%d err=%v value=%v",count,used,err,value)}
  value.Marker=123
  if _,err:=Decode(data[:len(data)-1],&value);!errors.Is(err,ErrDecode)||value.Marker!=123 {t.Fatalf("truncated header count=%d: %v",count,err)}
  if count>1 {
   if _,err:=DecodeWithLimits(data,&value,Limits{MaxElements:count-1});!errors.Is(err,ErrLimit)||value.Marker!=123 {t.Fatalf("budget count=%d: %v",count,err)}
  }
  if count>0 {
   // 未知字段也必须消费并验证其非内联内容，不得提前跳过坏载荷。
   binary.LittleEndian.PutUint16(data[len(data)-2:],0)
   if _,err:=Decode(data,&value);!errors.Is(err,ErrDecode)||value.Marker!=123 {t.Fatalf("missing unknown payload count=%d: %v",count,err)}
  }
 }
 for _,sample:=range []struct{entries []uint16;invalid bool}{
  {[]uint16{0xffff,0xffff},false},
  {[]uint16{0xffff,0xffff,2},true},
  {[]uint16{0xffff,0xfffd,2},false},
 } {
  data:=make([]byte,2+len(sample.entries)*2)
  binary.LittleEndian.PutUint16(data,uint16(len(sample.entries)))
  for i,v:=range sample.entries {binary.LittleEndian.PutUint16(data[2+2*i:],v)}
  value:=headerCursorEmpty{Marker:123}
  used,err:=Decode(data,&value)
  if sample.invalid {
   if !errors.Is(err,ErrDecode)||value.Marker!=123 {t.Fatalf("invalid jump: %v",err)}
  }else if err!=nil || used!=len(data) || value.Marker!=0 {t.Fatalf("valid jump: used=%d err=%v",used,err)}
 }
}
