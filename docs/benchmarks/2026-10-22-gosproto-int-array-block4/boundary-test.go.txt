package sproto
import("bytes";"encoding/binary";"math";"reflect";"testing")
type experimentArrayNamed int64
type experimentArrayDocument struct{Native []int64 `sproto:"integer,0,array"`;Named []experimentArrayNamed `sproto:"integer,1,array"`;Narrow []int16 `sproto:"integer,2,array"`}
// 分块/余数边界须保留signed值、精确容量、命名/窄整数回退、错误预算和输入不变。
func TestIntArrayBlockDifferential(t *testing.T){
 schema,err:=GetSprotoType(reflect.TypeFor[experimentArrayDocument]());if err!=nil{t.Fatal(err)}
 values:=[]int64{0,1,-1,math.MinInt32,math.MaxInt32,math.MinInt64,math.MaxInt64,1<<33}
 for f,typ:=range []reflect.Type{reflect.TypeFor[[]int64](),reflect.TypeFor[[]experimentArrayNamed](),reflect.TypeFor[[]int16]()} {field:=schema.Fields[f]
 check:=func(wire []byte){
  snapshot:=append([]byte(nil),wire...)
  for _,limit:=range []int{0,1,3,4,7,128,1000}{for _,initial:=range []int{0,1}{
   fresh:=func()reflect.Value{v:=reflect.New(typ).Elem();v.Set(reflect.MakeSlice(typ,2,2));v.Index(0).SetInt(17);v.Index(1).SetInt(23);return v}
   a:=fresh();left:=decodeBudget{limits:Limits{MaxElements:limit},elements:initial};wantErr:=experimentStringBaseline(0,false,wire,field,a,&left)
   for _,fn:=range []func(uint16,bool,[]byte,*SprotoField,reflect.Value,*decodeBudget)error{experimentStringClone,experimentStringCandidate}{b:=fresh();right:=decodeBudget{limits:Limits{MaxElements:limit},elements:initial};gotErr:=fn(0,false,wire,field,b,&right)
    if wantErr!=gotErr||left!=right||!reflect.DeepEqual(a.Interface(),b.Interface())||a.Cap()!=b.Cap(){t.Fatalf("field=%d wire=%x limit=%d initial=%d error=%v/%v budgets=%+v/%+v",f,wire,limit,initial,wantErr,gotErr,left,right)}
    if gotErr==nil&&a.Len()>0&&a.Pointer()==b.Pointer(){t.Fatal("outputs alias")}
   }
  }}
  if !bytes.Equal(wire,snapshot){t.Fatal("input modified")}
 }
 for _,width:=range []int{4,8}{for _,count:=range []int{0,1,2,3,4,5,7,8,9,31,32,33,129}{wire:=make([]byte,1+width*count);wire[0]=byte(width);for i:=0;i<count;i++{v:=uint64(values[i%len(values)]);if width==4{binary.LittleEndian.PutUint32(wire[1+i*width:],uint32(v))}else{binary.LittleEndian.PutUint64(wire[1+i*width:],v)}};check(wire);if len(wire)>1{check(wire[:len(wire)-1])}}}
 for _,wire:=range [][]byte{nil,{}, {0},{3},{3,1,2,3},{4,1},{8,1}}{check(wire)}
 }
}
