package benchmarks

import (
	"bytes"
	"testing"

	sproto "github.com/cnooljaw/gosproto"
	before "github.com/cnooljaw/gosproto-before"
	benchsp "github.com/cnooljaw/gsr/examples/websocket-gate/benchmarks/gen/sproto"
)

func pairOperations(t *testing.T, mode string) []pairOperation {
	t.Helper()
	body := bodyPairOperations(t, mode)
	for index := range body {
		body[index].name = "body/" + body[index].name
	}
	rpc := rpcPairOperations(t, mode)
	for index := range rpc {
		rpc[index].name = "rpc/" + rpc[index].name
	}
	pairs := append(body, rpc...)
	message := &benchsp.IntArrays{Values: make([]int64, 896)}
	pattern := []int64{-9223372036854775808, 9223372036854775807, -2147483649, 2147483648, -1, 0, 1}
	for index := range message.Values {
		message.Values[index] = pattern[index%len(pattern)]
	}
	old, err := before.Encode(message)
	if err != nil {
		t.Fatal(err)
	}
	current, err := sproto.Encode(message)
	if err != nil || !bytes.Equal(old, current) || cap(old) != cap(current) {
		t.Fatal("wide64 raw wire/cap changed")
	}
	for _, packed := range []bool{false, true} {
		name := "body/wide64/sproto_encode"
		if packed {
			name = "body/wide64/sproto_pencode"
		}
		left := func() error {
			var err error
			if packed {
				encodedSink, err = before.EncodePacked(message)
			} else {
				encodedSink, err = before.Encode(message)
			}
			return err
		}
		right := func() error {
			var err error
			if packed {
				encodedSink, err = sproto.EncodePacked(message)
			} else {
				encodedSink, err = sproto.Encode(message)
			}
			return err
		}
		if err := left(); err != nil {
			t.Fatal(err)
		}
		want := bytes.Clone(encodedSink)
		capacity := cap(encodedSink)
		if err := right(); err != nil || !bytes.Equal(want, encodedSink) || cap(encodedSink) != capacity {
			t.Fatal("wide64 composed wire/cap changed")
		}
		if mode == "self" {
			left = right
		}
		pairs = append(pairs, pairOperation{name, left, right})
	}
	for _, probe := range rpcProbePairs(t, "wide64", message, false) {
		left, err := probe.left()
		if err != nil {
			t.Fatal(err)
		}
		right, err := probe.right()
		if err != nil || !bytes.Equal(left, right) || cap(left) != cap(right) {
			t.Fatal("wide64 RPC wire/cap changed")
		}
		runLeft := func() error {
			var err error
			encodedSink, err = probe.left()
			return err
		}
		runRight := func() error {
			var err error
			encodedSink, err = probe.right()
			return err
		}
		if mode == "self" {
			runLeft = runRight
		}
		pairs = append(pairs, pairOperation{"rpc/" + probe.name, runLeft, runRight})
	}
	for _, operation := range growthPairOperations(t, mode) {
		operation.name = "growth/" + operation.name
		pairs = append(pairs, operation)
	}
	return pairs
}
