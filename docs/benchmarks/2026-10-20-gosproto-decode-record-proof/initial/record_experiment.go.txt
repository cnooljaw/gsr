package sproto

import (
	"reflect"
	"unsafe"
)

func experimentRecordClone(_ uint16, _ bool, data []byte, field *SprotoField, value reflect.Value, budget *decodeBudget) error {
	count, err := countChunks(data, budget)
	if err != nil {
		return err
	}
	result := reflect.MakeSlice(value.Type(), count, count)
	var slots []unsafe.Pointer
	if count > 0 && field.Array && field.Wire == WireStructName && field.st != nil &&
		value.Type().Elem() == reflect.PointerTo(field.st.typ) {
		// 同步借用新切片的指针槽；实际元素类型已核对，不缓存地址。
		slots = unsafe.Slice((*unsafe.Pointer)(result.UnsafePointer()), count)
	}
	for index := 0; index < count; index++ {
		used, content, err := readChunk(data)
		if err != nil {
			return err
		}
		current := reflect.New(field.st.typ)
		consumed, err := decodeMessage(content, field.st, current, budget)
		if err != nil {
			return err
		}
		if consumed != len(content) {
			return ErrDecode
		}
		if slots != nil {
			// Go 指针赋值保留 GC 写屏障，不能改成字节复制或 uintptr 存储。
			slots[index] = current.UnsafePointer()
		} else {
			result.Index(index).Set(current)
		}
		data = data[used:]
	}
	value.Set(result)
	return nil
}

func experimentRecordCandidate(_ uint16, _ bool, data []byte, field *SprotoField, value reflect.Value, budget *decodeBudget) error {
	count, err := countChunks(data, budget)
	if err != nil {
		return err
	}
	result := reflect.MakeSlice(value.Type(), count, count)
	var slots []unsafe.Pointer
	nativeRecords := false
	if count > 0 && field.Array && field.Wire == WireStructName && field.st != nil &&
		value.Type().Elem() == reflect.PointerTo(field.st.typ) {
		// 同步借用新切片的指针槽；实际元素类型已核对，不缓存地址。
		slots = unsafe.Slice((*unsafe.Pointer)(result.UnsafePointer()), count)
		nativeRecords = canDecodeIntegerRecords(field.st)
	}
	for index := 0; index < count; index++ {
		used, content, err := readChunk(data)
		if err != nil {
			return err
		}
		current := reflect.New(field.st.typ)
		var consumed int
		if nativeRecords {
			consumed, err = decodeIntegerRecord(content, field.st, decodePointer{value: current.UnsafePointer()}, budget)
		} else {
			consumed, err = decodeMessage(content, field.st, current, budget)
		}
		if err != nil {
			return err
		}
		if consumed != len(content) {
			return ErrDecode
		}
		if slots != nil {
			// Go 指针赋值保留 GC 写屏障，不能改成字节复制或 uintptr 存储。
			slots[index] = current.UnsafePointer()
		} else {
			result.Index(index).Set(current)
		}
		data = data[used:]
	}
	value.Set(result)
	return nil
}

// canDecodeIntegerRecords 只证明当前Decoder路由可达的非nil描述；不依赖当前公开Tag。
// 与Encode不同，负Tag、nil和短Fields不能被提前扫描改变语义，不能跨调用缓存此结果。
func canDecodeIntegerRecords(schema *SprotoType) bool {
	if !schema.int64Record {
		return false
	}
	for _, index := range schema.tagMap {
		if uint(index) >= uint(len(schema.Fields)) {
			return false
		}
		field := schema.Fields[index]
		if field != nil && !field.nativeInteger(schema) {
			return false
		}
	}
	return true
}

// decodeIntegerRecord只接收本次数组证明覆盖的字段与reflect.New(schema.typ)创建的独占对象。
func decodeIntegerRecord(chunk []byte, schema *SprotoType, pointer decodePointer, budget *decodeBudget) (int, error) {
	budget.depth++
	defer func() { budget.depth-- }()
	if budget.depth > budget.limits.MaxDepth {
		return 0, ErrLimit
	}
	if len(chunk) < 2 {
		return 0, ErrDecode
	}
	count := int(readUint16(chunk))
	total := 2 + count*2
	if total > len(chunk) {
		return 0, ErrDecode
	}
	if err := budget.add(count); err != nil {
		return 0, err
	}
	tag := 0
	for index := 0; index < count; index++ {
		entry := int(readUint16(chunk[2+index*2 : 4+index*2 : 4+index*2]))
		if entry%2 == 1 {
			tag += (entry + 1) / 2
			continue
		}
		if tag > 65535 {
			return 0, ErrDecode
		}
		hasInline := entry != 0
		inline := uint16(entry/2 - 1)
		var content []byte
		if !hasInline {
			used, data, err := readChunk(chunk[total:])
			if err != nil {
				return 0, err
			}
			total += used
			content = data
		}
		field := schema.FieldByTag(tag)
		tag++
		if field == nil {
			continue
		}
		bits, err := scalarBits(inline, hasInline, content)
		if err != nil {
			return 0, err
		}
		result := new(int64)
		*result = int64(bits)
		// 数组入口已核对当前可达字段与实际子类型；Go指针写入保留屏障。
		*(**int64)(unsafe.Add(pointer.value, field.encOffset)) = result
	}
	return total, nil
}

// ExperimentRecordSelector仅在实验归档内替换数组decoder，选择发生在计时外。
func ExperimentRecordSelector(message any) func(string) {
	schema, err := GetSprotoType(reflect.TypeOf(message).Elem())
	if err != nil {
		panic(err)
	}
	seen := make(map[*SprotoType]bool)
	fields := []*SprotoField{}
	var visit func(*SprotoType)
	visit = func(st *SprotoType) {
		if st == nil || seen[st] {
			return
		}
		seen[st] = true
		for _, f := range st.Fields {
			if f == nil {
				continue
			}
			if f.valueType != nil && f.valueType.Kind() == reflect.Slice && f.valueType.Elem().Kind() == reflect.Ptr && f.valueType.Elem().Elem().Kind() == reflect.Struct {
				fields = append(fields, f)
			}
			visit(f.st)
		}
	}
	visit(schema)
	return func(mode string) {
		fn := decoder(decodeStructSlice)
		switch mode {
		case "baseline":
		case "clone":
			fn = experimentRecordClone
		case "candidate":
			fn = experimentRecordCandidate
		default:
			panic("invalid record experiment decoder")
		}
		for _, f := range fields {
			f.dec = fn
		}
	}
}
