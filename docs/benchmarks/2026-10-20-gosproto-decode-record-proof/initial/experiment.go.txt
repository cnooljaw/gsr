type unpackBlock struct {
 Arm string `json:"arm"`
 Calls int `json:"calls"`
 Nanoseconds int64 `json:"nanoseconds"`
 Bytes, Allocs uint64
 GC uint32
 PauseNs uint64
 HeapBefore, HeapAfter uint64
}

type unpackQuartet struct {
 Round, Index int
 Order string
 Blocks []unpackBlock
 Ratio float64
}

// TestRecordProofInProcessExperiment 在同一模块/进程交错结构数组decoder，保留全部GC、分配与时间块。
func TestRecordProofInProcessExperiment(t *testing.T) {
 if runtime.GOMAXPROCS(0)!=1 { t.Fatal("GOMAXPROCS1 required") }
 name, mode := os.Getenv("GSR_CASE"), os.Getenv("GSR_MODE")
 var wire,packed []byte
 var newMessage func() any
 for _, sample := range representativeCases(t) {
  if sample.Name!=name {continue}
  _,wires:=representativeOperations(t,sample)
  checkRepresentativeWire(t,sample,wires)
  wire,packed=wires["sproto_encode"],wires["sproto_pack"]
  _,newMessage=representativeMessages(sample.Type)
 }
 if wire==nil {
  count:=128
  if len(name)>8 && name[:8]=="records_" {
   var err error
   count,err=strconv.Atoi(name[8:])
   if err!=nil {t.Fatal(err)}
  }
  items:=make([]*experimentRecord,count)
  for i:=range items {
   item:=new(experimentRecord)
   value:=reflect.ValueOf(item).Elem()
   for j:=0;j<8;j++ {
    if i%4==0 && j==5 {continue}
    v:=int64(i*8+j+1)
    if j%3==0 {v=-v}else if j%3==1 {v=1<<40+v}
    value.Field(j).Set(reflect.ValueOf(&v))
   }
   items[i]=item
  }
  var message any
  switch name {
  case "outer128":message=&experimentOuter{Items:experimentRows(items)};newMessage=func()any{return new(experimentOuter)}
  case "pointer128":
   values:=make([]experimentPointer,count)
   for i,v:=range items {values[i]=experimentPointer(v)}
   message=&experimentPointerDocument{Items:values};newMessage=func()any{return new(experimentPointerDocument)}
  case "mixed128":
   values:=make([]*experimentMixedRecord,count)
   for i:=range values {n:=int64(i);label:="mixed/"+strconv.Itoa(i);values[i]=&experimentMixedRecord{Number:&n,Label:&label}}
   message=&experimentMixedDocument{Items:values};newMessage=func()any{return new(experimentMixedDocument)}
  default:message=&experimentDocument{Items:items};newMessage=func()any{return new(experimentDocument)}
  }
  var err error
  wire,err=sp.Encode(message)
  if err!=nil {t.Fatal(err)}
  packed=sp.Pack(wire)
 }
 operation:=os.Getenv("GSR_OP")
 if operation!="decode" && operation!="pdecode" {t.Fatal("invalid operation")}
 selectMode:=sp.ExperimentRecordSelector(newMessage())
 defer selectMode("baseline")
 right:="candidate"
 switch mode {
 case "self": right="baseline"
 case "clone": right="clone"
 case "pass1","pass2":
 default:t.Fatal("unknown mode")
 }
 selections:=map[string]string{"A":"baseline","B":right}
 beforeInput:=append([]byte(nil),packed...)
 expected:=newMessage()
 selectMode("baseline")
 if _,err:=sp.Decode(wire,expected);err!=nil {t.Fatal(err)}
 for _,variant:=range []string{"baseline",right} {
  selectMode(variant)
  decoded:=newMessage()
  if _,err:=sp.Decode(wire,decoded);err!=nil || !reflect.DeepEqual(decoded,expected) {t.Fatal("body mismatch",err)}
  decoded=newMessage()
  if err:=sp.DecodePacked(packed,decoded);err!=nil || !reflect.DeepEqual(decoded,expected) {t.Fatal("packed mismatch",err)}
 }
 if os.Getenv("GSR_PREFLIGHT")=="1" {return}
 run:=func() {
  message:=newMessage()
  if operation=="decode" {
   if _,err:=sp.Decode(wire,message);err!=nil {t.Fatal(err)}
  }else if err:=sp.DecodePacked(packed,message);err!=nil {t.Fatal(err)}
  representativeSink=message
 }
 runs:=map[string]func(){"A":run,"B":run}
 runtime.GC()
 for _,arm:=range []string{"A","B"} {
  selectMode(selections[arm])
  start:=time.Now()
  for time.Since(start)<50*time.Millisecond {for range 64 {runs[arm]()}}
 }
 selectMode("baseline")
 start:=time.Now()
 for range 256 {runs["A"]()}
 calls:=int(float64(20*time.Millisecond)/float64(time.Since(start))*256)
 calls=max(1,min(calls,1<<20))
 blocks:=[]unpackQuartet{}
 for round:=0; round<6; round++ {
  for quartet:=0; quartet<4; quartet++ {
   order:="ABBA"
   if (round+quartet)%2!=0 {order="BAAB"}
   q:=unpackQuartet{Round:round,Index:quartet,Order:order}
   logRatio:=0.0
   for _,label:=range order {
    arm:=string(label)
    var before,after runtime.MemStats
    selectMode(selections[arm])
    runtime.ReadMemStats(&before)
    run:=runs[arm]
    start:=time.Now()
    for range calls {run()}
    ns:=time.Since(start).Nanoseconds()
    runtime.ReadMemStats(&after)
    block:=unpackBlock{Arm:arm,Calls:calls,Nanoseconds:ns,Bytes:after.TotalAlloc-before.TotalAlloc,Allocs:after.Mallocs-before.Mallocs,GC:after.NumGC-before.NumGC,PauseNs:after.PauseTotalNs-before.PauseTotalNs,HeapBefore:before.HeapAlloc,HeapAfter:after.HeapAlloc}
    q.Blocks=append(q.Blocks,block)
    sign:=1.0
    if arm=="A" {sign=-1}
    logRatio+=sign*math.Log(float64(ns)/float64(calls))
   }
   q.Ratio=math.Exp(logRatio/2)
   blocks=append(blocks,q)
  }
 }
 if !bytes.Equal(packed,beforeInput) {t.Fatal("input modified")}
 raw,err:=json.MarshalIndent(struct{Name,Operation,Mode string;CallsPerBlock int;Quartets []unpackQuartet}{name,operation,mode,calls,blocks},"","  ")
 if err!=nil {t.Fatal(err)}
 if err:=os.WriteFile(os.Getenv("GSR_OUTPUT"),append(raw,'\n'),0644);err!=nil {t.Fatal(err)}
 runtime.KeepAlive(runs)
}

type experimentRecord struct {
 F0 *int64 `sproto:"integer,0"`
 F1 *int64 `sproto:"integer,1"`
 F2 *int64 `sproto:"integer,2"`
 F3 *int64 `sproto:"integer,3"`
 F4 *int64 `sproto:"integer,4"`
 F5 *int64 `sproto:"integer,5"`
 F6 *int64 `sproto:"integer,6"`
 F7 *int64 `sproto:"integer,7"`
}
type experimentRows []*experimentRecord
type experimentPointer *experimentRecord
type experimentDocument struct {Items []*experimentRecord `sproto:"struct,0,array"`}
type experimentOuter struct {Items experimentRows `sproto:"struct,0,array"`}
type experimentPointerDocument struct {Items []experimentPointer `sproto:"struct,0,array"`}
type experimentMixedRecord struct {Number *int64 `sproto:"integer,0"`;Label *string `sproto:"string,1"`}
type experimentMixedDocument struct {Items []*experimentMixedRecord `sproto:"struct,0,array"`}
