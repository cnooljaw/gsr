package benchmarks

import (
	"encoding/json"
	"fmt"
	before "github.com/cnooljaw/gosproto-before"
	"strings"
	"testing"
)

type layoutRecord struct {
	A *int64 `sproto:"integer,0"`
	B *int64 `sproto:"integer,1"`
	C *int64 `sproto:"integer,2"`
}

type layoutRows struct {
	Rows []*layoutRecord `sproto:"struct,0,array"`
}

type layoutGroups struct {
	Groups []*layoutRows `sproto:"struct,0,array"`
}

type layoutArrayRecord struct {
	Values []int64 `sproto:"integer,0,array"`
}

type layoutArrayRows struct {
	Rows []*layoutArrayRecord `sproto:"struct,0,array"`
}

type layoutChain struct {
	Child *layoutChain `sproto:"struct,0"`
}

type layoutText struct {
	Text *string `sproto:"string,0"`
}

type layoutSample struct {
	Name          string
	Message       any
	Limits        before.Limits
	ExpectedCount int
}

type layoutPlan struct {
	Lengths      []int    `json:"lengths"`
	FlatModes    []string `json:"flatModes"`
	NestedGroups []int    `json:"nestedGroups"`
	NestedRows   []int    `json:"nestedRows"`
	ChainDepths  []int    `json:"chainDepths"`
	Initial      []int    `json:"initialCapacitiesToSimulate"`
	Allocation   struct {
		Calls int `json:"callsPerEntry"`
	} `json:"allocation"`
}

func layoutRecords(length int, mode string) *layoutRows {
	rows := &layoutRows{Rows: make([]*layoutRecord, length)}
	for index := range rows.Rows {
		if mode == "nil" || (mode == "half" && index%2 == 0) {
			continue
		}
		a, b, c := int64(index), int64(-index-1), int64(1<<33)
		rows.Rows[index] = &layoutRecord{A: &a, B: &b, C: &c}
	}
	return rows
}

func layoutSamples(t *testing.T, plan layoutPlan) []layoutSample {
	t.Helper()
	var samples []layoutSample
	for _, sample := range representativeCases(t) {
		_, factory := representativeMessages(sample.Type)
		message := factory()
		if err := json.Unmarshal(sample.Data, message); err != nil {
			t.Fatal(err)
		}
		expected := map[string]int{"addressbook": 6, "game_small": 11, "game_large": 131, "int32_edges": 2, "echo_ascii4k": 1}[sample.Name]
		samples = append(samples, layoutSample{Name: sample.Name, Message: message, ExpectedCount: expected})
	}
	text := strings.Repeat("A", 32)
	samples = append(samples, layoutSample{Name: "ascii32", Message: &layoutText{Text: &text}, ExpectedCount: 1})
	for _, length := range plan.Lengths {
		for _, mode := range plan.FlatModes {
			samples = append(samples, layoutSample{Name: fmt.Sprintf("flat/%s/%d", mode, length), Message: layoutRecords(length, mode), ExpectedCount: 1 + length})
		}
		rows := &layoutArrayRows{Rows: make([]*layoutArrayRecord, length)}
		for index := range rows.Rows {
			rows.Rows[index] = &layoutArrayRecord{Values: []int64{int64(index), -1}}
		}
		samples = append(samples, layoutSample{Name: fmt.Sprintf("array_record/%d", length), Message: rows, ExpectedCount: 1 + length*2})
	}
	for _, groups := range plan.NestedGroups {
		for _, rows := range plan.NestedRows {
			message := &layoutGroups{Groups: make([]*layoutRows, groups)}
			for index := range message.Groups {
				message.Groups[index] = layoutRecords(rows, "full")
			}
			samples = append(samples, layoutSample{Name: fmt.Sprintf("nested/%dx%d", groups, rows), Message: message, ExpectedCount: 1 + groups*(1+rows)})
		}
	}
	for _, depth := range plan.ChainDepths {
		message := &layoutChain{}
		current := message
		for index := 1; index < depth; index++ {
			current.Child = &layoutChain{}
			current = current.Child
		}
		expected := depth
		if depth > 64 {
			expected = -1
		}
		samples = append(samples, layoutSample{Name: fmt.Sprintf("chain/%d", depth), Message: message, ExpectedCount: expected})
	}
	for _, failure := range []struct {
		name   string
		limits before.Limits
	}{
		{"elements", before.Limits{MaxElements: 1}},
		{"bytes_early", before.Limits{MaxBytes: 2}},
		{"depth", before.Limits{MaxDepth: 1}},
		{"bytes_after_spill", before.Limits{MaxBytes: 2500}},
	} {
		samples = append(samples, layoutSample{Name: "failure/" + failure.name, Message: layoutRecords(4096, "full"), Limits: failure.limits, ExpectedCount: -1})
	}
	return samples
}
