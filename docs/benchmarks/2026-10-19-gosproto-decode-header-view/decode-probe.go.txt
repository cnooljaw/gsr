package benchmarks

import (
	"bytes"
	"encoding/json"
	"os"
	"reflect"
	"strings"
	"testing"

	sproto "github.com/cnooljaw/gosproto"
	before "github.com/cnooljaw/gosproto-before"
)

func appendDecodePair(t testing.TB, pairs *[]pairOperation, mode, name string, left, right func() error) {
	t.Helper()
	if err := left(); err != nil {
		t.Fatal(name, err)
	}
	if err := right(); err != nil {
		t.Fatal(name, err)
	}
	if mode == "self" {
		left = right
	}
	*pairs = append(*pairs, pairOperation{name, left, right})
}

func decodeBodyOperations(t testing.TB, mode, name string, message any, factory func() any, controls bool) []pairOperation {
	t.Helper()
	wire, err := before.Encode(message)
	if err != nil {
		t.Fatal(err)
	}
	current, err := sproto.Encode(message)
	if err != nil || !bytes.Equal(wire, current) {
		t.Fatalf("%s encoded wire differs: %v", name, err)
	}
	packed := before.Pack(wire)
	if !bytes.Equal(packed, sproto.Pack(wire)) {
		t.Fatal(name, "packed wire differs")
	}
	oldResult, result := factory(), factory()
	oldUsed, oldErr := before.Decode(wire, oldResult)
	used, err := sproto.Decode(wire, result)
	if oldErr != nil || err != nil || oldUsed != len(wire) || used != len(wire) || !reflect.DeepEqual(oldResult, result) {
		t.Fatalf("%s decode semantics: %d/%d %v/%v", name, oldUsed, used, oldErr, err)
	}
	// nil 结构元素的既有 wire 语义会生成空结构；以两库结果对照，不误判为输入不相等。
	var pairs []pairOperation
	appendDecodePair(t, &pairs, mode, name+"/decode", func() error {
		value := factory()
		_, err := before.Decode(wire, value)
		representativeSink = value
		return err
	}, func() error {
		value := factory()
		_, err := sproto.Decode(wire, value)
		representativeSink = value
		return err
	})
	if controls {
		appendDecodePair(t, &pairs, mode, name+"/unpack", func() error {
			var err error
			encodedSink, err = before.Unpack(packed)
			return err
		}, func() error {
			var err error
			encodedSink, err = sproto.Unpack(packed)
			return err
		})
	}
	appendDecodePair(t, &pairs, mode, name+"/pdecode", func() error {
		value := factory()
		err := before.DecodePacked(packed, value)
		representativeSink = value
		return err
	}, func() error {
		value := factory()
		err := sproto.DecodePacked(packed, value)
		representativeSink = value
		return err
	})
	return pairs
}

func decodeDispatchPair(t testing.TB, mode, name string, message any) pairOperation {
	t.Helper()
	typ := reflect.TypeOf(message)
	oldRPC, err := before.NewRpc([]*before.Protocol{{Type: 1, Name: "frame", Request: typ}})
	if err != nil {
		t.Fatal(err)
	}
	rpc, err := sproto.NewRpc([]*sproto.Protocol{{Type: 1, Name: "frame", Request: typ}})
	if err != nil {
		t.Fatal(err)
	}
	wire, err := oldRPC.RequestEncode("frame", 0, message)
	if err != nil {
		t.Fatal(err)
	}
	current, err := rpc.RequestEncode("frame", 0, message)
	if err != nil || !bytes.Equal(wire, current) {
		t.Fatal("dispatch wire differs", err)
	}
	oldMode, oldName, oldSession, oldMessage, oldErr := oldRPC.Dispatch(wire)
	currentMode, currentName, session, currentMessage, err := rpc.Dispatch(wire)
	if oldErr != nil || err != nil || int(oldMode) != int(currentMode) || oldName != "frame" || currentName != oldName || oldSession != 0 || session != 0 || !reflect.DeepEqual(oldMessage, currentMessage) || !reflect.DeepEqual(currentMessage, message) {
		t.Fatal("dispatch result differs", oldErr, err)
	}
	left := func() error {
		_, _, _, value, err := oldRPC.Dispatch(wire)
		representativeSink = value
		return err
	}
	right := func() error {
		_, _, _, value, err := rpc.Dispatch(wire)
		representativeSink = value
		return err
	}
	if mode == "self" {
		left = right
	}
	return pairOperation{"rpc/" + name + "/dispatch", left, right}
}

func pairOperations(t testing.TB, mode string) []pairOperation {
	t.Helper()
	var pairs []pairOperation
	var encodeControls []pairOperation
	for _, sample := range representativeCases(t) {
		operations, wires := representativeOperations(t, sample)
		checkRepresentativeWire(t, sample, wires)
		_, factory := representativeMessages(sample.Type)
		message := factory()
		if err := json.Unmarshal(sample.Data, message); err != nil {
			t.Fatal(err)
		}
		pairs = append(pairs, decodeBodyOperations(t, mode, "body/"+sample.Name, message, factory, true)...)
		pairs = append(pairs, decodeDispatchPair(t, mode, sample.Name, message))
		if sample.Name == "game_large" {
			old := map[string]func() error{
				"sproto_encode": func() error {
					var err error
					encodedSink, err = before.Encode(message)
					return err
				},
				"sproto_pencode": func() error {
					var err error
					encodedSink, err = before.EncodePacked(message)
					return err
				},
			}
			for _, operation := range operations {
				if left, ok := old[operation.name]; ok {
					appendDecodePair(t, &encodeControls, mode, "control/game_large/"+strings.TrimPrefix(operation.name, "sproto_"), left, operation.run)
				}
			}
		}
	}
	content, err := os.ReadFile("cases.json")
	if err != nil {
		t.Fatal(err)
	}
	var samples []benchmarkCase
	if err := json.Unmarshal(content, &samples); err != nil {
		t.Fatal(err)
	}
	for _, sample := range samples {
		text := strings.Repeat(sample.Pattern, sample.Repeat)
		old, current := beforeEchoCodecs(text), echoCodecs(text)
		for _, shape := range []string{"payload", "frame"} {
			left, right := old["sproto_"+shape], current["sproto_"+shape]
			wire, err := left.encode()
			encoded, currentErr := right.encode()
			if err != nil || currentErr != nil || !bytes.Equal(wire, encoded) {
				t.Fatal("echo wire differs", err, currentErr)
			}
			for _, codec := range []benchmarkCodec{left, right} {
				value, err := codec.decode(wire)
				if err != nil || value != text {
					t.Fatal("echo decode differs", err)
				}
			}
			appendDecodePair(t, &pairs, mode, "echo/"+sample.Name+"/"+shape+"_decode", func() error {
				var err error
				decodedSink, err = left.decode(wire)
				return err
			}, func() error {
				var err error
				decodedSink, err = right.decode(wire)
				return err
			})
		}
	}
	growthSamples := growthFixtureSamples(t)
	for _, name := range strings.Split(os.Getenv("GSR_GROWTH_CASES"), ",") {
		var message any
		for _, sample := range growthSamples {
			if sample.Name == name {
				message = sample.Message
				break
			}
		}
		if message == nil {
			t.Fatal("missing growth fixture", name)
		}
		typ := reflect.TypeOf(message).Elem()
		factory := func() any { return reflect.New(typ).Interface() }
		pairs = append(pairs, decodeBodyOperations(t, mode, "growth/"+name, message, factory, false)...)
	}
	pairs = append(pairs, encodeControls...)
	if len(pairs) != 40 {
		t.Fatal("incomplete Decode matrix", len(pairs))
	}
	representativeSink, encodedSink, decodedSink = nil, nil, ""
	return pairs
}

// TestDecodeFixtureMatrix 保护新矩阵的40个固定入口均能处理同一wire并维持既有语义。
func TestDecodeFixtureMatrix(t *testing.T) {
	for _, pair := range pairOperations(t, "clone") {
		if err := pair.left(); err != nil {
			t.Fatalf("%s baseline: %v", pair.name, err)
		}
		if err := pair.right(); err != nil {
			t.Fatalf("%s candidate: %v", pair.name, err)
		}
	}
}
