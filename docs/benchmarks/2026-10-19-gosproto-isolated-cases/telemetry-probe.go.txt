package benchmarks

import (
	"encoding/json"
	"os"
	"runtime/metrics"
	"testing"
)

type isolatedSnapshot struct {
	Stage       string
	GC          uint64
	Limiter     uint64
	HeapObjects uint64
	HeapGoal    uint64
	HeapScan    uint64
	AllocBytes  uint64
	GOGC        uint64
	MemoryLimit uint64
}

func isolatedMetrics(stage string) isolatedSnapshot {
	samples := []metrics.Sample{
		{Name: "/gc/cycles/total:gc-cycles"},
		{Name: "/gc/limiter/last-enabled:gc-cycle"},
		{Name: "/memory/classes/heap/objects:bytes"},
		{Name: "/gc/heap/goal:bytes"},
		{Name: "/gc/scan/heap:bytes"},
		{Name: "/gc/heap/allocs:bytes"},
		{Name: "/gc/gogc:percent"},
		{Name: "/gc/gomemlimit:bytes"},
	}
	metrics.Read(samples)
	return isolatedSnapshot{stage, samples[0].Value.Uint64(), samples[1].Value.Uint64(),
		samples[2].Value.Uint64(), samples[3].Value.Uint64(), samples[4].Value.Uint64(),
		samples[5].Value.Uint64(), samples[6].Value.Uint64(), samples[7].Value.Uint64()}
}

// TestIsolatedMetrics 保护诊断所需计数可读且仍使用默认 GC 与无限内存软上限。
func TestIsolatedMetrics(t *testing.T) {
	snapshot := isolatedMetrics("check")
	if snapshot.GOGC != 100 || snapshot.MemoryLimit != 1<<63-1 || snapshot.HeapObjects == 0 {
		t.Fatalf("unexpected runtime settings: %+v", snapshot)
	}
}

// TestIsolatedCaseList 只列出原矩阵名称，驱动在任何正式采样前校验完整顺序。
func TestIsolatedCaseList(t *testing.T) {
	output := os.Getenv("GSR_ISOLATED_LIST")
	if output == "" {
		t.Skip("explicit case listing")
	}
	operations := pairOperations(t, "clone")
	var names []string
	for _, operation := range operations {
		names = append(names, operation.name)
	}
	contents, err := json.Marshal(names)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(output, append(contents, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
}
