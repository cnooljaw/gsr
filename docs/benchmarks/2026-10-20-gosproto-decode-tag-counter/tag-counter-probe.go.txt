package benchmarks

import (
 "encoding/binary"
 "errors"
 "testing"
 before "github.com/cnooljaw/gosproto-before"
 sproto "github.com/cnooljaw/gosproto"
)

type tagCounterTarget struct {
 Value *int64 `sproto:"integer,65535"`
}

// TestDecodeTagCounterBoundaries 保护非负累加的最大合法tag、越界实际字段、最大跳跃链及失败原值。
func TestDecodeTagCounterBoundaries(t *testing.T) {
 makeWire:=func(entries []uint16) []byte {
  out:=make([]byte,2+len(entries)*2)
  binary.LittleEndian.PutUint16(out,uint16(len(entries)))
  for i,v:=range entries {binary.LittleEndian.PutUint16(out[2+i*2:],v)}
  return out
 }
 maxSkips:=make([]uint16,65535)
 for i:=range maxSkips {maxSkips[i]=65535}
 lastActual:=append([]uint16(nil),maxSkips...);lastActual[len(lastActual)-1]=2
 cases:=[]struct{name string;entries []uint16;wantErr bool;present bool}{
  {"max_tag",[]uint16{65535,65533,2},false,true},
  {"max_unknown_tag",[]uint16{65535,65531,2},false,false},
  {"beyond_max_actual",[]uint16{65535,65535,2},true,false},
  {"max_skips",maxSkips,false,false},
  {"max_skips_then_actual",lastActual,true,false},
 }
 for _,c:=range cases {t.Run(c.name,func(t *testing.T){
  wire:=makeWire(c.entries)
  for _,budget:=range []int{len(c.entries),len(c.entries)+1} {
   original:=int64(99);left:=tagCounterTarget{Value:&original};right:=tagCounterTarget{Value:&original}
   usedA,errA:=before.DecodeWithLimits(wire,&left,before.Limits{MaxElements:budget})
   usedB,errB:=sproto.DecodeWithLimits(wire,&right,sproto.Limits{MaxElements:budget})
   if c.wantErr {
    if !errors.Is(errA,before.ErrDecode)||!errors.Is(errB,sproto.ErrDecode)||usedA!=0||usedB!=0||left.Value!=&original||right.Value!=&original {t.Fatalf("failure atomicity/errors: %d/%d %v/%v",usedA,usedB,errA,errB)}
   } else {
    if errA!=nil||errB!=nil||usedA!=len(wire)||usedB!=len(wire)||(left.Value!=nil)!=c.present||(right.Value!=nil)!=c.present {t.Fatalf("decode mismatch: %d/%d %v/%v present=%v/%v",usedA,usedB,errA,errB,left.Value!=nil,right.Value!=nil)}
    if c.present&&(*left.Value!=0||*right.Value!=0||left.Value==right.Value) {t.Fatal("value/ownership mismatch")}
   }
  }
 })}
}
