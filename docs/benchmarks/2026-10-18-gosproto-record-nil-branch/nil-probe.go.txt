package benchmarks

import (
    "bytes"
    "fmt"
    "math"
    "testing"
    sproto "github.com/cnooljaw/gosproto"
    before "github.com/cnooljaw/gosproto-before"
)

type nilProbeRecord struct {
    Ignored [19]byte
    First *int64 `sproto:"integer,0"`
    Second *int64 `sproto:"integer,2"`
    Third *int64 `sproto:"integer,4"`
    Last *int64 `sproto:"integer,6"`
}

type nilProbeDocument struct {
    Items []*nilProbeRecord `sproto:"struct,0,array"`
}

// 补充检查 nil 入口的潜在代价；原43项探针与生产候选保持冻结。
func pairOperations(t *testing.T, mode string) []pairOperation {
    t.Helper()
    var pairs []pairOperation
    for _, count := range []int{1, 32, 256, -256} {
        size := count
        label := "nil"
        if size < 0 {
            size = -size
            label = "mixed"
        }
        input := &nilProbeDocument{Items: make([]*nilProbeRecord, size)}
        first, second, third, last := int64(-1), int64(0), int64(42), int64(math.MaxInt64)
        if count < 0 {
            for index := range input.Items {
                switch index % 3 {
                case 1:
                    input.Items[index] = &nilProbeRecord{}
                case 2:
                    input.Items[index] = &nilProbeRecord{First: &first, Second: &second, Third: &third, Last: &last}
                }
            }
        }
        for _, packed := range []bool{false, true} {
            operation := "encode"
            if packed {
                operation = "pencode"
            }
            left := func() error {
                var err error
                if packed {
                    encodedSink, err = before.EncodePacked(input)
                } else {
                    encodedSink, err = before.Encode(input)
                }
                return err
            }
            right := func() error {
                var err error
                if packed {
                    encodedSink, err = sproto.EncodePacked(input)
                } else {
                    encodedSink, err = sproto.Encode(input)
                }
                return err
            }
            if err := left(); err != nil {
                t.Fatal(err)
            }
            want := bytes.Clone(encodedSink)
            capacity := cap(encodedSink)
            if err := right(); err != nil || !bytes.Equal(want, encodedSink) || cap(encodedSink) != capacity {
                t.Fatal("nil probe wire/cap changed", err)
            }
            if mode == "self" {
                left = right
            }
            pairs = append(pairs, pairOperation{fmt.Sprintf("%s%d/%s", label, size, operation), left, right})
        }
    }
    return pairs
}
