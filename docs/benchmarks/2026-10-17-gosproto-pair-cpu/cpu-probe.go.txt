package benchmarks

import (
	"encoding/json"
	"math"
	"os"
	"reflect"
	"runtime"
	"runtime/metrics"
	"syscall"
	"testing"
	"time"
)

type cpuProbeBlock struct {
	Arm          string `json:"arm"`
	Calls        int    `json:"calls"`
	WallNS       int64  `json:"wallNs"`
	UserNS       int64  `json:"userCpuNs"`
	SystemNS     int64  `json:"systemCpuNs"`
	CompletedGCs uint64 `json:"completedGcCycles"`
}

type cpuProbeQuartet struct {
	Round    int             `json:"round"`
	Index    int             `json:"index"`
	Blocks   []cpuProbeBlock `json:"blocks"`
	WallRate float64         `json:"wallRatio"`
	CPURate  float64         `json:"cpuRatio"`
}

type cpuProbeResult struct {
	Name         string            `json:"name"`
	SameFunction bool              `json:"sameFunction"`
	Calls        int               `json:"callsPerBlock"`
	Quartets     []cpuProbeQuartet `json:"quartets"`
	WallRounds   []float64         `json:"wallRoundRatios"`
	CPURounds    []float64         `json:"cpuRoundRatios"`
}

func usageNS(value syscall.Timeval) int64 {
	return int64(value.Sec)*int64(time.Second) + int64(value.Usec)*int64(time.Microsecond)
}

// sampleCPU 将 CPU 观测放在墙钟区间外；进程 CPU 包含 Runtime 线程，窗口略宽且有观测成本。
func sampleCPU(run func() error, calls int, counter []metrics.Sample) (cpuProbeBlock, error) {
	metrics.Read(counter)
	startGC := counter[0].Value.Uint64()
	var before, after syscall.Rusage
	if err := syscall.Getrusage(syscall.RUSAGE_SELF, &before); err != nil {
		return cpuProbeBlock{}, err
	}
	start := time.Now()
	for range calls {
		if err := run(); err != nil {
			return cpuProbeBlock{}, err
		}
	}
	elapsed := time.Since(start).Nanoseconds()
	if err := syscall.Getrusage(syscall.RUSAGE_SELF, &after); err != nil {
		return cpuProbeBlock{}, err
	}
	metrics.Read(counter)
	return cpuProbeBlock{
		Calls: calls, WallNS: elapsed,
		UserNS:       usageNS(after.Utime) - usageNS(before.Utime),
		SystemNS:     usageNS(after.Stime) - usageNS(before.Stime),
		CompletedGCs: counter[0].Value.Uint64() - startGC,
	}, nil
}

// TestCPUProbeCounters 保护秒/微秒换算、实际 CPU 计数单调和回调调用数；不要求 CPU 小于墙钟。
func TestCPUProbeCounters(t *testing.T) {
	if usageNS(syscall.Timeval{Sec: 2, Usec: 3}) != 2000003000 {
		t.Fatal("CPU time conversion changed")
	}
	calls := 0
	counter := []metrics.Sample{{Name: "/gc/cycles/total:gc-cycles"}}
	block, err := sampleCPU(func() error {
		calls++
		return nil
	}, 10000, counter)
	if err != nil || calls != 10000 || block.Calls != 10000 || block.WallNS <= 0 || block.UserNS < 0 || block.SystemNS < 0 {
		t.Fatalf("block=%+v calls=%d err=%v", block, calls, err)
	}
}

// TestCPUProbe 在同函数 A/A 中同时保留墙钟与进程 CPU 的全部块，不据此批准任何优化。
func TestCPUProbe(t *testing.T) {
	output := os.Getenv("GSR_CPU_PROBE")
	if output == "" {
		t.Skip("isolated diagnostic only")
	}
	if runtime.GOOS != "darwin" || runtime.GOMAXPROCS(0) != 1 {
		t.Fatal("this diagnostic requires macOS and GOMAXPROCS=1")
	}
	counter := []metrics.Sample{{Name: "/gc/cycles/total:gc-cycles"}, {Name: "/gc/gogc:percent"}}
	metrics.Read(counter)
	gcPercent := counter[1].Value.Uint64()
	if gcPercent == ^uint64(0) {
		t.Fatal("GC must remain enabled")
	}
	names := []string{"game_large/sproto_pack", "int32_edges/sproto_pack", "ascii4k/frame"}
	operations, err := selectPairs(pairOperations(t, "self"), names)
	if err != nil {
		t.Fatal(err)
	}
	var results []cpuProbeResult
	for _, operation := range operations {
		result := cpuProbeResult{
			Name:         operation.name,
			SameFunction: reflect.ValueOf(operation.left).Pointer() == reflect.ValueOf(operation.right).Pointer(),
		}
		if !result.SameFunction {
			t.Fatal("self requires the same function")
		}
		runtime.GC()
		warmup := time.Now()
		for time.Since(warmup) < 50*time.Millisecond {
			for range 64 {
				if err := operation.left(); err != nil {
					t.Fatal(err)
				}
				if err := operation.right(); err != nil {
					t.Fatal(err)
				}
			}
		}
		leftCalls, err := blockCalls(operation.left, 40*time.Millisecond, counter)
		if err != nil {
			t.Fatal(err)
		}
		rightCalls, err := blockCalls(operation.right, 40*time.Millisecond, counter)
		if err != nil {
			t.Fatal(err)
		}
		result.Calls = max(leftCalls, rightCalls)
		for round := range pairRounds {
			var wallLogs, cpuLogs float64
			for index := range pairQuartets {
				quartet := cpuProbeQuartet{Round: round, Index: index}
				var wallBlocks, cpuBlocks []pairBlock
				for _, arm := range quartetOrder(round, index) {
					run := operation.left
					if arm == 'B' {
						run = operation.right
					}
					block, err := sampleCPU(run, result.Calls, counter)
					if err != nil {
						t.Fatal(err)
					}
					block.Arm = string(arm)
					quartet.Blocks = append(quartet.Blocks, block)
					wallBlocks = append(wallBlocks, pairBlock{Arm: block.Arm, Calls: block.Calls, Nanoseconds: block.WallNS})
					cpuBlocks = append(cpuBlocks, pairBlock{Arm: block.Arm, Calls: block.Calls, Nanoseconds: block.UserNS + block.SystemNS})
				}
				quartet.WallRate, err = quartetRatio(wallBlocks)
				if err != nil {
					t.Fatal(err)
				}
				quartet.CPURate, err = quartetRatio(cpuBlocks)
				if err != nil {
					t.Fatal(err)
				}
				wallLogs += math.Log(quartet.WallRate)
				cpuLogs += math.Log(quartet.CPURate)
				result.Quartets = append(result.Quartets, quartet)
			}
			result.WallRounds = append(result.WallRounds, math.Exp(wallLogs/pairQuartets))
			result.CPURounds = append(result.CPURounds, math.Exp(cpuLogs/pairQuartets))
		}
		results = append(results, result)
		t.Logf("completed %s", operation.name)
	}
	contents, err := json.MarshalIndent(struct {
		GOGCPercent uint64           `json:"gogcPercent"`
		Results     []cpuProbeResult `json:"results"`
	}{gcPercent, results}, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(output, append(contents, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
}
