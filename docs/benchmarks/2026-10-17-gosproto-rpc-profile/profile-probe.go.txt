package benchmarks

import (
	"encoding/json"
	"reflect"
	"strings"
	"testing"

	sproto "github.com/cnooljaw/gosproto"
	"github.com/cnooljaw/gsr/examples/websocket-gate/gen/sproto/c2s"
)

func BenchmarkCurrentRPC(b *testing.B) {
	for _, name := range []string{"ascii32", "game_large", "int32_edges", "echo_ascii4k"} {
		b.Run(name, func(b *testing.B) {
			var message any
			if name == "ascii32" {
				text := strings.Repeat("A", 32)
				message = &c2s.EchoRequest{Text: &text}
			} else {
				for _, sample := range representativeCases(b) {
					if sample.Name == name {
						_, factory := representativeMessages(sample.Type)
						message = factory()
						if err := json.Unmarshal(sample.Data, message); err != nil {
							b.Fatal(err)
						}
					}
				}
			}
			if message == nil {
				b.Fatal("missing fixture")
			}
			rpc, err := sproto.NewRpc([]*sproto.Protocol{{Type: 1, Name: "frame", Request: reflect.TypeOf(message)}})
			if err != nil {
				b.Fatal(err)
			}
			b.ReportAllocs()
			b.ResetTimer()
			for b.Loop() {
				encodedSink, err = rpc.RequestEncode("frame", 0, message)
				if err != nil {
					b.Fatal(err)
				}
			}
		})
	}
}
