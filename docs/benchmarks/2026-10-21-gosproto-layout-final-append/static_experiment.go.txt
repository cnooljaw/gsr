func ExperimentStaticNormal(message interface{}, limits Limits, packResult bool, prefix []byte) ([]byte, error) {
	output, encoded, err := encodeBufferWithLimits(message, limits, packResult, prefix)
	if err != nil {
		return nil, err
	}
	if packResult {
		return packInto(output, encoded), nil
	}
	return output, nil
}

func ExperimentStaticBaseline(message interface{}, limits Limits, packResult bool, prefix []byte) ([]byte, error) {
	output, encoded, err := experimentBaselineBuffer(message, limits, packResult, prefix)
	if err != nil {
		return nil, err
	}
	if packResult {
		return packInto(output, encoded), nil
	}
	return output, nil
}

// experimentBaselineBuffer 完成独立 body 预检和写出，返回同一独占缓冲及其中的未压缩帧。
// reservePack 预留原 Pack 容量与读写间隔；RPC 请求可先做 session admission，再执行 Pack。
func experimentBaselineBuffer(message interface{}, limits Limits, reservePack bool, prefix []byte) ([]byte, []byte, error) {
	limits, err := normalizeLimits(limits)
	if err != nil {
		return nil, nil, err
	}
	typ, value, err := getbase(message)
	if err != nil {
		return nil, nil, err
	}
	schema, err := GetSprotoType(typ.Elem())
	if err != nil {
		return nil, nil, err
	}
	budget := &decodeBudget{limits: limits}
	var layout encodeLayout
	binding := knownEncodeType(schema)
	size, err := experimentBaselineSize(schema, value, budget, 1, &layout, binding)
	runtime.KeepAlive(message)
	if err != nil {
		return nil, nil, err
	}
	totalSize := len(prefix) + size
	capacity, start := totalSize, 0
	if reservePack {
		capacity = packedBufferSize(totalSize)
		// FF 头的最坏增长和八字节余量隔开读写游标，避免覆盖尚未消费的编码字节。
		start = capacity - totalSize
	}
	output := make([]byte, capacity)
	encoded := output[start : start+totalSize]
	if len(prefix) > 0 {
		copy(encoded, prefix)
	}
	experimentBaselineMessage(schema, value, encoded[len(prefix):], &layout, binding)
	runtime.KeepAlive(message)
	return output, encoded, nil
}

func experimentBaselineMessage(schema *SprotoType, value reflect.Value, output []byte, layout *encodeLayout, binding encodeBinding) int {
	count := layout.next()
	writeUint16(output, uint16(count))
	if value.IsNil() {
		return 2
	}
	record := value.Elem()
	pointer := messageEncodePointer(value, binding)
	headerOffset, bodyOffset, previous := 2, 2+count*2, -1
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		native := binding.native && field.nativeInteger(schema)
		var current reflect.Value
		var bits uint64
		if native {
			var present bool
			bits, present = pointer.integer(field.encOffset)
			if !present {
				continue
			}
		} else {
			current = record.Field(field.index[0])
			if omittedField(field, current) {
				continue
			}
		}
		if skip := skipTag(previous, field.Tag); skip > 0 {
			writeUint16(output[headerOffset:], skip)
			headerOffset += 2
		}
		var header uint16
		var length int
		if native {
			// 内联整数只占头部，非内联时才构造内容区切片。
			if bits <= MaxEmbeddedInt {
				header = uint16((bits + 1) * 2)
			} else {
				header, length = encodeIntegerInto(bits, output[bodyOffset:])
			}
		} else if field.Array || field.Wire == WireStructName {
			length = experimentBaselineField(field, current, output[bodyOffset+4:], layout, binding.exact && field.encOwner == schema && field.Array != field.pointer)
		} else {
			header, length = encodeScalarInto(field, current, output[bodyOffset:])
		}
		writeUint16(output[headerOffset:], header)
		headerOffset += 2
		previous = field.Tag
		if header == 0 {
			writeUint32(output[bodyOffset:], uint32(length))
			bodyOffset += 4 + length
		}
	}
	return bodyOffset
}

func experimentBaselineField(field *SprotoField, value reflect.Value, output []byte, layout *encodeLayout, trusted bool) int {
	if field.Array {
		if value.Len() == 0 {
			return 0
		}
		offset := 0
		switch field.Wire {
		case WireVarintName:
			width := layout.next()
			if field.int64Slice {
				return encodeInt64Slice(*value.Addr().Interface().(*[]int64), width, output)
			}
			output[0] = byte(width)
			for index := 0; index < value.Len(); index++ {
				number := integerBits(field, value.Index(index))
				if width == 4 {
					writeUint32(output[1+index*4:], uint32(number))
				} else {
					writeUint64(output[1+index*8:], number)
				}
			}
			offset = 1 + width*value.Len()
		case WireDoubleName:
			output[0] = 8
			offset = 1
			for index := 0; index < value.Len(); index++ {
				writeUint64(output[offset:], math.Float64bits(value.Index(index).Float()))
				offset += 8
			}
		case WireBooleanName:
			for index := 0; index < value.Len(); index++ {
				if value.Index(index).Bool() {
					output[index] = 1
				} else {
					output[index] = 0
				}
			}
			offset = value.Len()
		case WireBytesName:
			for index := 0; index < value.Len(); index++ {
				text := value.Index(index).String()
				writeUint32(output[offset:], uint32(len(text)))
				offset += 4 + copy(output[offset+4:], text)
			}
		case WireStructName:
			childBinding := bindEncodeChild(field, value, trusted)
			integers := childBinding.native && field.st.int64Record && field.st.nativeIntegerFields()
			var pointers []unsafe.Pointer
			if integers {
				pointers = integerRecordPointers(field.st, value)
			}
			if integers && value.Len() >= 8 {
				return experimentBaselineArray(field, value, output, layout, childBinding, pointers)
			}
			for index := 0; index < value.Len(); index++ {
				var length int
				if integers {
					var pointer encodePointer
					if pointers != nil {
						pointer = encodePointer{value: pointers[index]}
					} else {
						pointer = messageEncodePointer(value.Index(index), childBinding)
					}
					length = experimentBaselineRecord(field.st, pointer, output[offset+4:], layout.next())
				} else {
					length = experimentBaselineMessage(field.st, value.Index(index), output[offset+4:], layout, childBinding)
				}
				writeUint32(output[offset:], uint32(length))
				offset += 4 + length
			}
		}
		return offset
	}
	return experimentBaselineMessage(field.st, value, output, layout, bindEncodeChild(field, value, trusted))
}

func experimentBaselineSize(schema *SprotoType, value reflect.Value, budget *decodeBudget, depth int, layout *encodeLayout, binding encodeBinding) (int, error) {
	if depth > budget.limits.MaxDepth {
		return 0, ErrLimit
	}
	size, lastTag := 2, -1
	if size > budget.limits.MaxBytes {
		return 0, ErrLimit
	}
	layoutIndex := layout.add(0)
	if value.IsNil() {
		return size, nil
	}
	record := value.Elem()
	pointer := messageEncodePointer(value, binding)
	headerCount := 0
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		native := binding.native && field.nativeInteger(schema)
		var current reflect.Value
		var bits uint64
		if native {
			var present bool
			bits, present = pointer.integer(field.encOffset)
			if !present {
				continue
			}
		} else {
			current = record.Field(field.index[0])
			if omittedField(field, current) {
				continue
			}
		}
		if err := budget.add(1); err != nil {
			return 0, err
		}
		gap := field.Tag - lastTag - 1
		if gap > 32768 {
			return 0, fmt.Errorf("%w: tag gap", ErrEncode)
		}
		if gap > 0 {
			size += 2
			headerCount++
		}
		size += 2
		headerCount++
		lastTag = field.Tag
		var part int
		var inline bool
		var err error
		if native {
			part, inline = signedSize(bits)
		} else {
			part, inline, err = experimentBaselineFieldSize(field, current, budget, depth, layout, binding.exact && field.encOwner == schema && field.Array != field.pointer)
		}
		if err != nil {
			return 0, err
		}
		if !inline {
			size += 4 + part
		}
		if size > budget.limits.MaxBytes {
			return 0, ErrLimit
		}
	}
	layout.set(layoutIndex, headerCount)
	return size, nil
}

func experimentBaselineFieldSize(field *SprotoField, value reflect.Value, budget *decodeBudget, depth int, layout *encodeLayout, trusted bool) (int, bool, error) {
	if field.Array {
		if err := budget.add(value.Len()); err != nil {
			return 0, false, err
		}
		size := 0
		switch field.Wire {
		case WireVarintName:
			if field.int64Slice {
				values := *value.Addr().Interface().(*[]int64)
				width := 4
				for _, number := range values {
					if number < MinInt32 || number > MaxInt32 {
						width = 8
						break
					}
				}
				if len(values) > 0 {
					size = 1 + width*len(values)
					layout.add(width)
				}
				break
			}
			width := 4
			for i := 0; i < value.Len(); i++ {
				bits := integerBits(field, value.Index(i))
				if field.unsigned && bits > math.MaxInt64 {
					return 0, false, fmt.Errorf("%w: integer exceeds signed 64-bit range", ErrEncode)
				}
				if integerWidth(bits) > width {
					width = 8
				}
			}
			if value.Len() > 0 {
				size = 1 + width*value.Len()
				layout.add(width)
			}
		case WireDoubleName:
			if value.Len() > 0 {
				size = 1 + 8*value.Len()
			}
		case WireBooleanName:
			size = value.Len()
		case WireBytesName:
			for i := 0; i < value.Len(); i++ {
				size += 4 + value.Index(i).Len()
				if size > budget.limits.MaxBytes {
					return 0, false, ErrLimit
				}
			}
		case WireStructName:
			var childBinding encodeBinding
			if value.Len() > 0 {
				childBinding = bindEncodeChild(field, value, trusted)

			}
			integers := childBinding.native && field.st.int64Record && field.st.nativeIntegerFields()
			var pointers []unsafe.Pointer
			if integers {
				pointers = integerRecordPointers(field.st, value)
			}
			for i := 0; i < value.Len(); i++ {
				var part int
				var err error
				if integers {
					var pointer encodePointer
					if pointers != nil {
						pointer = encodePointer{value: pointers[i]}
					} else {
						pointer = messageEncodePointer(value.Index(i), childBinding)
					}
					part, err = experimentBaselineRecordSize(field.st, pointer, budget, depth+1, layout)
				} else {
					part, err = experimentBaselineSize(field.st, value.Index(i), budget, depth+1, layout, childBinding)
				}
				if err != nil {
					return 0, false, err
				}
				size += 4 + part
				if size > budget.limits.MaxBytes {
					return 0, false, ErrLimit
				}
			}
		}
		if size > budget.limits.MaxBytes {
			return 0, false, ErrLimit
		}
		return size, false, nil
	}
	if field.Wire == WireStructName {
		size, err := experimentBaselineSize(field.st, value, budget, depth+1, layout, bindEncodeChild(field, value, trusted))
		return size, false, err
	}
	if field.pointer {
		value = value.Elem()
	}
	switch field.Wire {
	case WireVarintName:
		bits := integerBits(field, value)
		if field.unsigned && bits > math.MaxInt64 {
			return 0, false, fmt.Errorf("%w: integer exceeds signed 64-bit range", ErrEncode)
		}
		return integerWidth(bits), bits <= MaxEmbeddedInt, nil
	case WireDoubleName:
		return 8, false, nil
	case WireBooleanName:
		return 0, true, nil
	case WireBytesName:
		return value.Len(), false, nil
	}
	return 0, false, ErrEncode
}

// Validate 检查显式预算；零值合法并在编解码时使用默认值。
func experimentBaselineRecordSize(schema *SprotoType, pointer encodePointer, budget *decodeBudget, depth int, layout *encodeLayout) (int, error) {
	if depth > budget.limits.MaxDepth || budget.limits.MaxBytes < 2 {
		return 0, ErrLimit
	}
	layoutIndex := layout.add(0)
	if pointer.value == nil {
		return 2, nil
	}
	size, headers, lastTag := 2, 0, -1
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		bits, present := pointer.integer(field.encOffset)
		if !present {
			continue
		}
		if err := budget.add(1); err != nil {
			return 0, err
		}
		gap := field.Tag - lastTag - 1
		if gap > 32768 {
			return 0, fmt.Errorf("%w: tag gap", ErrEncode)
		}
		if gap > 0 {
			size += 2
			headers++
		}
		size += 2
		headers++
		lastTag = field.Tag
		if bits > MaxEmbeddedInt {
			size += 4 + integerWidth(bits)
		}
		if size > budget.limits.MaxBytes {
			return 0, ErrLimit
		}
	}
	layout.set(layoutIndex, headers)
	return size, nil
}

// experimentBaselineRecord 复用同一次预检的布局，字段值仍逐个读取并写入独占输出。
func experimentBaselineRecord(schema *SprotoType, pointer encodePointer, output []byte, count int) int {
	writeUint16(output, uint16(count))
	if pointer.value == nil {
		return 2
	}
	headerOffset, bodyOffset, previous := 2, 2+count*2, -1
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		bits, present := pointer.integer(field.encOffset)
		if !present {
			continue
		}
		if skip := skipTag(previous, field.Tag); skip > 0 {
			writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], skip)
			headerOffset += 2
		}
		previous = field.Tag
		// inline 路径写完头部后立即继续，外置路径独立完成内容长度与写出。
		if bits <= MaxEmbeddedInt {
			writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], uint16((bits+1)*2))
			headerOffset += 2
			continue
		}
		body := output[bodyOffset:]
		_, length := encodeIntegerInto(bits, body)
		writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], 0)
		writeUint32(body, uint32(length))
		headerOffset += 2
		bodyOffset += 4 + length
	}
	return bodyOffset
}

// experimentBaselineArray 只消费本次预检形成的连续整数记录布局，不包含嵌套布局项。
func experimentBaselineArray(field *SprotoField, value reflect.Value, output []byte, layout *encodeLayout, binding encodeBinding, pointers []unsafe.Pointer) int {
	count := value.Len()
	index, offset := 0, 0
	for index < count {
		cursor := layout.cursor
		var headers []uint32
		if cursor < len(layout.local) {
			end := min(len(layout.local), cursor+count-index)
			headers = layout.local[cursor:end:end]
		} else {
			start := cursor - len(layout.local)
			end := start + count - index
			headers = layout.extra[start:end:end]
		}
		layout.cursor += len(headers)
		for _, header := range headers {
			var pointer encodePointer
			if pointers != nil {
				pointer = encodePointer{value: pointers[index]}
			} else {
				pointer = messageEncodePointer(value.Index(index), binding)
			}
			length := experimentBaselineRecord(field.st, pointer, output[offset+4:], int(header))
			writeUint32(output[offset:], uint32(length))
			offset += 4 + length
			index++
		}
	}
	return offset
}
func ExperimentStaticClone(message interface{}, limits Limits, packResult bool, prefix []byte) ([]byte, error) {

	if err != nil {

	}
	if packResult {
		return packInto(output, encoded), nil
	}
	return output, nil
}

		gap := field.Tag - lastTag - 1
		if gap > 32768 {
			return 0, fmt.Errorf("%w: tag gap", ErrEncode)
		}
		if gap > 0 {
			size += 2
			headerCount++
		}
		size += 2
		headerCount++
		lastTag = field.Tag
		var part int
		var inline bool
		var err error
		if native {
			part, inline = signedSize(bits)
		} else {
			part, inline, err = experimentCloneFieldSize(field, current, budget, depth, layout, binding.exact && field.encOwner == schema && field.Array != field.pointer)
		}
		if err != nil {
			return 0, err
		}
		if !inline {
			size += 4 + part
		}
		if size > budget.limits.MaxBytes {
			return 0, ErrLimit
		}
	}
	layout.set(layoutIndex, headerCount)
	return size, nil
}

func experimentCloneFieldSize(field *SprotoField, value reflect.Value, budget *decodeBudget, depth int, layout *encodeLayout, trusted bool) (int, bool, error) {
	if field.Array {
		if err := budget.add(value.Len()); err != nil {
			return 0, false, err
		}
		size := 0
		switch field.Wire {
		case WireVarintName:
			if field.int64Slice {
				values := *value.Addr().Interface().(*[]int64)
				width := 4
				for _, number := range values {
					if number < MinInt32 || number > MaxInt32 {
						width = 8
						break
					}
				}
				if len(values) > 0 {
					size = 1 + width*len(values)
					layout.add(width)
				}
				break
			}
			width := 4
			for i := 0; i < value.Len(); i++ {
				bits := integerBits(field, value.Index(i))
				if field.unsigned && bits > math.MaxInt64 {
					return 0, false, fmt.Errorf("%w: integer exceeds signed 64-bit range", ErrEncode)
				}
				if integerWidth(bits) > width {
					width = 8
				}
			}
			if value.Len() > 0 {
				size = 1 + width*value.Len()
				layout.add(width)
			}
		case WireDoubleName:
			if value.Len() > 0 {
				size = 1 + 8*value.Len()
			}
		case WireBooleanName:
			size = value.Len()
		case WireBytesName:
			for i := 0; i < value.Len(); i++ {
				size += 4 + value.Index(i).Len()
				if size > budget.limits.MaxBytes {
					return 0, false, ErrLimit
				}
			}
		case WireStructName:
			var childBinding encodeBinding
			if value.Len() > 0 {
				childBinding = bindEncodeChild(field, value, trusted)

			}
			integers := childBinding.native && field.st.int64Record && field.st.nativeIntegerFields()
			var pointers []unsafe.Pointer
			if integers {
				pointers = integerRecordPointers(field.st, value)
			}
			for i := 0; i < value.Len(); i++ {
				var part int
				var err error
				if integers {
					var pointer encodePointer
					if pointers != nil {
						pointer = encodePointer{value: pointers[i]}
					} else {
						pointer = messageEncodePointer(value.Index(i), childBinding)
					}
					part, err = experimentCloneRecordSize(field.st, pointer, budget, depth+1, layout)
				} else {
					part, err = experimentCloneSize(field.st, value.Index(i), budget, depth+1, layout, childBinding)
				}
				if err != nil {
					return 0, false, err
				}
				size += 4 + part
				if size > budget.limits.MaxBytes {
					return 0, false, ErrLimit
				}
			}
		}
		if size > budget.limits.MaxBytes {
			return 0, false, ErrLimit
		}
		return size, false, nil
	}
	if field.Wire == WireStructName {
		size, err := experimentCloneSize(field.st, value, budget, depth+1, layout, bindEncodeChild(field, value, trusted))
		return size, false, err
	}
	if field.pointer {
		value = value.Elem()
	}
	switch field.Wire {
	case WireVarintName:
		bits := integerBits(field, value)
		if field.unsigned && bits > math.MaxInt64 {
			return 0, false, fmt.Errorf("%w: integer exceeds signed 64-bit range", ErrEncode)
		}
		return integerWidth(bits), bits <= MaxEmbeddedInt, nil
	case WireDoubleName:
		return 8, false, nil
	case WireBooleanName:
		return 0, true, nil
	case WireBytesName:
		return value.Len(), false, nil
	}
	return 0, false, ErrEncode
}

// Validate 检查显式预算；零值合法并在编解码时使用默认值。
func experimentCloneRecordSize(schema *SprotoType, pointer encodePointer, budget *decodeBudget, depth int, layout *encodeLayout) (int, error) {
	if depth > budget.limits.MaxDepth || budget.limits.MaxBytes < 2 {
		return 0, ErrLimit
	}
	layoutIndex := layout.add(0)
	if pointer.value == nil {
		return 2, nil
	}
	size, headers, lastTag := 2, 0, -1
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		bits, present := pointer.integer(field.encOffset)
		if !present {
			continue
		}
		if err := budget.add(1); err != nil {
			return 0, err
		}
		gap := field.Tag - lastTag - 1
		if gap > 32768 {
			return 0, fmt.Errorf("%w: tag gap", ErrEncode)
		}
		if gap > 0 {
			size += 2
			headers++
		}
		size += 2
		headers++
		lastTag = field.Tag
		if bits > MaxEmbeddedInt {
			size += 4 + integerWidth(bits)
		}
		if size > budget.limits.MaxBytes {
			return 0, ErrLimit
		}
	}
	layout.set(layoutIndex, headers)
	return size, nil
}

// experimentCloneRecord 复用同一次预检的布局，字段值仍逐个读取并写入独占输出。
func experimentCloneRecord(schema *SprotoType, pointer encodePointer, output []byte, count int) int {
	writeUint16(output, uint16(count))
	if pointer.value == nil {
		return 2
	}
	headerOffset, bodyOffset, previous := 2, 2+count*2, -1
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		bits, present := pointer.integer(field.encOffset)
		if !present {
			continue
		}
		if skip := skipTag(previous, field.Tag); skip > 0 {
			writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], skip)
			headerOffset += 2
		}
		previous = field.Tag
		// inline 路径写完头部后立即继续，外置路径独立完成内容长度与写出。
		if bits <= MaxEmbeddedInt {
			writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], uint16((bits+1)*2))
			headerOffset += 2
			continue
		}
		body := output[bodyOffset:]
		_, length := encodeIntegerInto(bits, body)
		writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], 0)
		writeUint32(body, uint32(length))
		headerOffset += 2
		bodyOffset += 4 + length
	}
	return bodyOffset
}

// experimentCloneArray 只消费本次预检形成的连续整数记录布局，不包含嵌套布局项。
func experimentCloneArray(field *SprotoField, value reflect.Value, output []byte, layout *encodeLayout, binding encodeBinding, pointers []unsafe.Pointer) int {
	count := value.Len()
	index, offset := 0, 0
	for index < count {
		cursor := layout.cursor
		var headers []uint32
		if cursor < len(layout.local) {
			end := min(len(layout.local), cursor+count-index)
			headers = layout.local[cursor:end:end]
		} else {
			start := cursor - len(layout.local)
			end := start + count - index
			headers = layout.extra[start:end:end]
		}
		layout.cursor += len(headers)
		for _, header := range headers {
			var pointer encodePointer
			if pointers != nil {
				pointer = encodePointer{value: pointers[index]}
			} else {
				pointer = messageEncodePointer(value.Index(index), binding)
			}
			length := experimentCloneRecord(field.st, pointer, output[offset+4:], int(header))
			writeUint32(output[offset:], uint32(length))
			offset += 4 + length
			index++
		}
	}
	return offset
}
func ExperimentStaticCandidate(message interface{}, limits Limits, packResult bool, prefix []byte) ([]byte, error) {
	output, encoded, err := experimentCandidateBuffer(message, limits, packResult, prefix)
	if err != nil {
		return nil, err
	}
	if packResult {
		return packInto(output, encoded), nil
	}
	return output, nil
}

// experimentCandidateBuffer 完成独立 body 预检和写出，返回同一独占缓冲及其中的未压缩帧。
// reservePack 预留原 Pack 容量与读写间隔；RPC 请求可先做 session admission，再执行 Pack。
func experimentCandidateBuffer(message interface{}, limits Limits, reservePack bool, prefix []byte) ([]byte, []byte, error) {
	limits, err := normalizeLimits(limits)
	if err != nil {
		return nil, nil, err
	}
	typ, value, err := getbase(message)
	if err != nil {
		return nil, nil, err
	}
	schema, err := GetSprotoType(typ.Elem())
	if err != nil {
		return nil, nil, err
	}
	budget := &decodeBudget{limits: limits}
	var layout encodeLayout
	binding := knownEncodeType(schema)
	size, err := experimentCandidateSize(schema, value, budget, 1, &layout, binding)
	runtime.KeepAlive(message)
	if err != nil {
		return nil, nil, err
	}
	totalSize := len(prefix) + size
	capacity, start := totalSize, 0
	if reservePack {
		capacity = packedBufferSize(totalSize)
		// FF 头的最坏增长和八字节余量隔开读写游标，避免覆盖尚未消费的编码字节。
		start = capacity - totalSize
	}
	output := make([]byte, capacity)
	encoded := output[start : start+totalSize]
	if len(prefix) > 0 {
		copy(encoded, prefix)
	}
	experimentCandidateMessage(schema, value, encoded[len(prefix):], &layout, binding)
	runtime.KeepAlive(message)
	return output, encoded, nil
}

func experimentCandidateMessage(schema *SprotoType, value reflect.Value, output []byte, layout *encodeLayout, binding encodeBinding) int {
	count := layout.next()
	writeUint16(output, uint16(count))
	if value.IsNil() {
		return 2
	}
	record := value.Elem()
	pointer := messageEncodePointer(value, binding)
	headerOffset, bodyOffset, previous := 2, 2+count*2, -1
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		native := binding.native && field.nativeInteger(schema)
		var current reflect.Value
		var bits uint64
		if native {
			var present bool
			bits, present = pointer.integer(field.encOffset)
			if !present {
				continue
			}
		} else {
			current = record.Field(field.index[0])
			if omittedField(field, current) {
				continue
			}
		}
		if skip := skipTag(previous, field.Tag); skip > 0 {
			writeUint16(output[headerOffset:], skip)
			headerOffset += 2
		}
		var header uint16
		var length int
		if native {
			// 内联整数只占头部，非内联时才构造内容区切片。
			if bits <= MaxEmbeddedInt {
				header = uint16((bits + 1) * 2)
			} else {
				header, length = encodeIntegerInto(bits, output[bodyOffset:])
			}
		} else if field.Array || field.Wire == WireStructName {
			length = experimentCandidateField(field, current, output[bodyOffset+4:], layout, binding.exact && field.encOwner == schema && field.Array != field.pointer)
		} else {
			header, length = encodeScalarInto(field, current, output[bodyOffset:])
		}
		writeUint16(output[headerOffset:], header)
		headerOffset += 2
		previous = field.Tag
		if header == 0 {
			writeUint32(output[bodyOffset:], uint32(length))
			bodyOffset += 4 + length
		}
	}
	return bodyOffset
}

func experimentCandidateField(field *SprotoField, value reflect.Value, output []byte, layout *encodeLayout, trusted bool) int {
	if field.Array {
		if value.Len() == 0 {
			return 0
		}
		offset := 0
		switch field.Wire {
		case WireVarintName:
			width := layout.next()
			if field.int64Slice {
				return encodeInt64Slice(*value.Addr().Interface().(*[]int64), width, output)
			}
			output[0] = byte(width)
			for index := 0; index < value.Len(); index++ {
				number := integerBits(field, value.Index(index))
				if width == 4 {
					writeUint32(output[1+index*4:], uint32(number))
				} else {
					writeUint64(output[1+index*8:], number)
				}
			}
			offset = 1 + width*value.Len()
		case WireDoubleName:
			output[0] = 8
			offset = 1
			for index := 0; index < value.Len(); index++ {
				writeUint64(output[offset:], math.Float64bits(value.Index(index).Float()))
				offset += 8
			}
		case WireBooleanName:
			for index := 0; index < value.Len(); index++ {
				if value.Index(index).Bool() {
					output[index] = 1
				} else {
					output[index] = 0
				}
			}
			offset = value.Len()
		case WireBytesName:
			for index := 0; index < value.Len(); index++ {
				text := value.Index(index).String()
				writeUint32(output[offset:], uint32(len(text)))
				offset += 4 + copy(output[offset+4:], text)
			}
		case WireStructName:
			childBinding := bindEncodeChild(field, value, trusted)
			integers := childBinding.native && field.st.int64Record && field.st.nativeIntegerFields()
			var pointers []unsafe.Pointer
			if integers {
				pointers = integerRecordPointers(field.st, value)
			}
			if integers && value.Len() >= 8 {
				return experimentCandidateArray(field, value, output, layout, childBinding, pointers)
			}
			for index := 0; index < value.Len(); index++ {
				var length int
				if integers {
					var pointer encodePointer
					if pointers != nil {
						pointer = encodePointer{value: pointers[index]}
					} else {
						pointer = messageEncodePointer(value.Index(index), childBinding)
					}
					length = experimentCandidateRecord(field.st, pointer, output[offset+4:], layout.next())
				} else {
					length = experimentCandidateMessage(field.st, value.Index(index), output[offset+4:], layout, childBinding)
				}
				writeUint32(output[offset:], uint32(length))
				offset += 4 + length
			}
		}
		return offset
	}
	return experimentCandidateMessage(field.st, value, output, layout, bindEncodeChild(field, value, trusted))
}

			if pointers != nil {
				pointer = encodePointer{value: pointers[index]}
			} else {
				pointer = messageEncodePointer(value.Index(index), binding)
			}
			length := experimentCandidateRecord(field.st, pointer, output[offset+4:], int(header))
			writeUint32(output[offset:], uint32(length))
			offset += 4 + length
			index++
		}
	}
	return offset
}

var experimentStaticSelected func(interface{}, Limits, bool, []byte) ([]byte, error)

func init() { experimentStaticSelected = ExperimentStaticBaseline }

// ExperimentStringMode 只选择不接收layout指针的最外层入口，供既有契约逐模式验证。
func ExperimentStringMode(mode string) {
	switch mode {
	case "baseline":
		experimentStaticSelected = ExperimentStaticBaseline
	case "clone":
		experimentStaticSelected = ExperimentStaticClone
	case "candidate":
		experimentStaticSelected = ExperimentStaticCandidate
	default:
		panic("invalid mode")
	}
}
