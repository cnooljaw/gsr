package sproto

import (
	"bytes"
	"fmt"
	"reflect"
	"runtime"
	"runtime/debug"
	"testing"
)

// TestEncodeRecordFieldProjection 保护局部字段容量边界、乱序tag、nil元素、当前值及调用间描述替换。
func TestEncodeRecordFieldProjection(t *testing.T) {
	type named int64
	previous := debug.SetGCPercent(1)
	defer debug.SetGCPercent(previous)
	for _, count := range []int{1, 16, 17, 32} {
		nativeFields := make([]reflect.StructField, count)
		genericFields := make([]reflect.StructField, count)
		for i := 0; i < count; i++ {
			tag := reflect.StructTag(fmt.Sprintf("sproto:\"integer,%d\"", (count-1-i)*2))
			nativeFields[i] = reflect.StructField{Name: fmt.Sprintf("ProjectionValue%d", i), Type: reflect.TypeFor[*int64](), Tag: tag}
			genericFields[i] = reflect.StructField{Name: nativeFields[i].Name, Type: reflect.TypeFor[*named](), Tag: tag}
		}
		nativeType, genericType := reflect.StructOf(nativeFields), reflect.StructOf(genericFields)
		nativeSchema, err := GetSprotoType(nativeType)
		if err != nil {
			t.Fatal(err)
		}
		genericSchema, err := GetSprotoType(genericType)
		if err != nil {
			t.Fatal(err)
		}
		nativeOriginal, genericOriginal := nativeSchema.Fields[0], genericSchema.Fields[0]
		// 即使测试提前终止，也恢复仅属于本夹具的元信息。
		defer func() { nativeSchema.Fields[0] = nativeOriginal; genericSchema.Fields[0] = genericOriginal }()
		for _, length := range []int{1, 33} {
			nativeArray := reflect.MakeSlice(reflect.SliceOf(reflect.PointerTo(nativeType)), length, length)
			genericArray := reflect.MakeSlice(reflect.SliceOf(reflect.PointerTo(genericType)), length, length)
			var live []*int64
			var reference []*named
			for i := 0; i < length; i++ {
				if i%4 == 1 {
					continue
				}
				n, g := reflect.New(nativeType), reflect.New(genericType)
				for j := 0; j < count; j++ {
					if j%3 == 1 {
						continue
					}
					value := int64(i*100 + j)
					if j%3 == 2 {
						value = -1<<40 + value
					}
					a, b := new(int64), new(named)
					*a = value
					*b = named(value)
					n.Elem().Field(j).Set(reflect.ValueOf(a))
					g.Elem().Field(j).Set(reflect.ValueOf(b))
					live = append(live, a)
					reference = append(reference, b)
				}
				nativeArray.Index(i).Set(n)
				genericArray.Index(i).Set(g)
			}
			docType := reflect.StructOf([]reflect.StructField{{Name: "Rows", Type: nativeArray.Type(), Tag: "sproto:\"struct,0,array\""}})
			refType := reflect.StructOf([]reflect.StructField{{Name: "Rows", Type: genericArray.Type(), Tag: "sproto:\"struct,0,array\""}})
			doc, ref := reflect.New(docType), reflect.New(refType)
			doc.Elem().Field(0).Set(nativeArray)
			ref.Elem().Field(0).Set(genericArray)
			var original []byte
			for phase := 0; phase < 3; phase++ {
				if phase == 1 {
					a, b := *nativeOriginal, *genericOriginal
					a.Tag = -1
					b.Tag = -1
					nativeSchema.Fields[0], genericSchema.Fields[0] = &a, &b
				} else {
					nativeSchema.Fields[0], genericSchema.Fields[0] = nativeOriginal, genericOriginal
				}
				runtime.GC()
				got, err := Encode(doc.Interface())
				want, wantErr := Encode(ref.Interface())
				if err != nil || wantErr != nil || !bytes.Equal(got, want) || cap(got) != cap(want) {
					t.Fatalf("fields=%d rows=%d phase=%d: %v/%v", count, length, phase, err, wantErr)
				}
				packed, err := EncodePacked(doc.Interface())
				if err != nil || !bytes.Equal(packed, Pack(want)) {
					t.Fatal("packed differs", err)
				}
				if phase == 0 {
					original = bytes.Clone(got)
				}
				if phase == 2 && !bytes.Equal(original, got) {
					t.Fatal("metadata restoration did not restore wire")
				}
				result, err := EncodeWithLimits(doc.Interface(), Limits{MaxBytes: len(got) - 1})
				if err == nil || result != nil {
					t.Fatal("byte budget bypassed")
				}
				result, err = EncodeWithLimits(doc.Interface(), Limits{MaxBytes: len(got)})
				if err != nil || !bytes.Equal(result, got) {
					t.Fatal("exact byte budget rejected")
				}
				snapshot := bytes.Clone(got)
				for i := range live {
					*live[i]++
					*reference[i]++
				}
				changed, err := Encode(doc.Interface())
				expected, expectedErr := Encode(ref.Interface())
				if err != nil || expectedErr != nil || !bytes.Equal(changed, expected) || !bytes.Equal(snapshot, got) {
					t.Fatal("current values or output ownership changed")
				}
				for i := range live {
					*live[i]--
					*reference[i]--
				}
				runtime.KeepAlive(doc)
			}
		}
	}
}
