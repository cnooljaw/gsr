package benchmarks

import (
 "bytes"
 "crypto/sha256"
 "encoding/hex"
 "encoding/json"
 "math/bits"
 "os"
 "testing"
 sproto "github.com/cnooljaw/gosproto"
)

type packWordAudit struct {
 Name string
 Bytes, FullWords, TailBytes, PackedBytes int
 NonzeroHistogram [9]int
 ZeroMasks, SparseMasks, DenseMasks, FFMasks, FFSegments, FFConsumedBytes, OrdinaryWords int
 WireSHA256, PackedSHA256 string
}

// TestPackWordAudit 核对实际wire的mask分布与独立Pack重建，不采耗时或修改计时夹具。
func TestPackWordAudit(t *testing.T) {
 output:=os.Getenv("GSR_WIRE_AUDIT")
 if output=="" {t.Skip("explicit wire audit")}
 var result []packWordAudit
 for _,sample:=range representativeCases(t) {
  if sample.Name!="game_large"&&sample.Name!="int32_edges"&&sample.Name!="echo_ascii4k" {continue}
  _,wires:=representativeOperations(t,sample)
  wire,packed:=wires["sproto_encode"],wires["sproto_pack"]
  item:=packWordAudit{Name:sample.Name,Bytes:len(wire),FullWords:len(wire)/8,TailBytes:len(wire)%8,PackedBytes:len(packed)}
  wsum,psum:=sha256.Sum256(wire),sha256.Sum256(packed);item.WireSHA256,item.PackedSHA256=hex.EncodeToString(wsum[:]),hex.EncodeToString(psum[:])
  for offset:=0;offset+8<=len(wire);offset+=8 {
   count:=0
   for _,v:=range wire[offset:offset+8] {if v!=0 {count++}}
   item.NonzeroHistogram[count]++
   switch {case count==0:item.ZeroMasks++;case count<=2:item.SparseMasks++;case count==8:item.FFMasks++;default:item.DenseMasks++}
  }
  // 从packed流独立重建，区分实际普通mask与已合并进FF段的word。
  var restored []byte
  for i:=0;i<len(packed); {
   header:=packed[i];i++
   if header==255 {
    if i>=len(packed) {t.Fatal("truncated FF")}
    n:=(int(packed[i])+1)*8;i++
    if n>len(packed)-i {t.Fatal("truncated FF payload")}
    restored=append(restored,packed[i:i+n]...);i+=n
    item.FFSegments++;item.FFConsumedBytes+=n
   } else {
    item.OrdinaryWords++
    if bits.OnesCount8(header)>len(packed)-i {t.Fatal("truncated sparse word")}
    for bit:=uint(0);bit<8;bit++ {
     var v byte
     if header&(1<<bit)!=0 {v=packed[i];i++}
     restored=append(restored,v)
    }
   }
  }
  if len(restored)!=((len(wire)+7)&^7)||!bytes.Equal(restored[:len(wire)],wire) {t.Fatal("independent restore differs")}
  unpacked,err:=sproto.Unpack(packed);if err!=nil||!bytes.Equal(unpacked,restored) {t.Fatal("public unpack differs",err)}
  result=append(result,item)
 }
 if len(result)!=3 {t.Fatal("incomplete audit")}
 encoded,err:=json.MarshalIndent(result,"","  ");if err!=nil {t.Fatal(err)}
 if err=os.WriteFile(output,append(encoded,'\n'),0644);err!=nil {t.Fatal(err)}
}
