// TestNormalEncodeBlock 通过正常公开入口测量一个独立时间块，不在库中插入选择器。
func TestNormalEncodeBlock(t *testing.T) {
	if runtime.GOMAXPROCS(0) != 1 {
		t.Fatal("GOMAXPROCS1 required")
	}
	name := os.Getenv("GSR_CASE")
	var source any
	var wire, packed []byte
	var newMessage func() any
	for _, sample := range representativeCases(t) {
		if sample.Name != name {
			continue
		}
		_, wires := representativeOperations(t, sample)
		checkRepresentativeWire(t, sample, wires)
		wire, packed = wires["sproto_encode"], wires["sproto_pack"]
		_, newMessage = representativeMessages(sample.Type)
		source = newMessage()
		if err := json.Unmarshal(sample.Data, source); err != nil {
			t.Fatal(err)
		}
	}
	if wire == nil {
		var message any
		count := 33
		switch name {
		case "records7":
			count = 7
		case "records8":
			count = 8
		case "records33":
			count = 33
		case "records128":
			count = 128
		case "prefix31":
		default:
			t.Fatal("unknown case")
		}
		rows := &experimentLayoutRows{Items: experimentLayoutItems(count), After: experimentLayoutItems(7)}
		if name == "prefix31" {
			rows.Before = experimentLayoutItems(31)
		}
		label := "tail/中文"
		rows.Tail = &label
		message = rows
		source = message
		newMessage = func() any { return new(experimentLayoutRows) }
		var err error
		wire, err = sp.Encode(message)
		if err != nil {
			t.Fatal(err)
		}
		packed = sp.Pack(wire)
	}

	_ = newMessage
	op := os.Getenv("GSR_OP")
	calls, err := strconv.Atoi(os.Getenv("GSR_CALLS"))
	if err != nil || calls < 1 {
		t.Fatal("calls required")
	}
	if op != "encode" && op != "pencode" {
		t.Fatal("invalid op")
	}
	body, err := sp.Encode(source)
	if err != nil || !bytes.Equal(body, wire) {
		t.Fatal("body preflight", err)
	}
	frame, err := sp.EncodePacked(source)
	if err != nil || !bytes.Equal(frame, packed) {
		t.Fatal("frame preflight", err)
	}
	run := func() {
		var output []byte
		var err error
		if op == "encode" {
			output, err = sp.Encode(source)
		} else {
			output, err = sp.EncodePacked(source)
		}
		if err != nil {
			t.Fatal(err)
		}
		encodedSink = output
	}
	encodedSink, representativeSink, decodedSink = nil, nil, ""
	runtime.GC()
	start := time.Now()
	for time.Since(start) < 20*time.Millisecond {
		for range 64 {
			run()
		}
	}
	var before, after runtime.MemStats
	runtime.ReadMemStats(&before)
	start = time.Now()
	for range calls {
		run()
	}
	ns := time.Since(start).Nanoseconds()
	runtime.ReadMemStats(&after)
	bodyAfter, err := sp.Encode(source)
	if err != nil || !bytes.Equal(bodyAfter, wire) {
		t.Fatal("source changed", err)
	}
	hash := func(data []byte) string { sum := sha256.Sum256(data); return hex.EncodeToString(sum[:]) }
	result := struct {
		Name, Operation, BodyHash, PackedHash string
		Calls                                 int
		Nanoseconds                           int64
		Bytes, Allocs                         uint64
		GC                                    uint32
		PauseNs                               uint64
		HeapBefore, HeapAfter                 uint64
	}{name, op, hash(wire), hash(packed), calls, ns, after.TotalAlloc - before.TotalAlloc, after.Mallocs - before.Mallocs, after.NumGC - before.NumGC, after.PauseTotalNs - before.PauseTotalNs, before.HeapAlloc, after.HeapAlloc}
	raw, err := json.Marshal(result)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("GSR_OUTPUT"), append(raw, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
	runtime.KeepAlive(source)
}

type experimentText string
type experimentTextRecord struct {
	First *string `sproto:"string,0"`
	Last  *string `sproto:"string,1"`
}
type experimentTextRows struct {
	Items []*experimentTextRecord `sproto:"struct,0,array"`
}
type experimentNamedTextRecord struct {
	First *experimentText `sproto:"string,0"`
	Last  *experimentText `sproto:"string,1"`
}
type experimentNamedTextRows struct {
	Items []*experimentNamedTextRecord `sproto:"struct,0,array"`
}

type experimentTextPointer *experimentTextRecord
type experimentPointerRows struct {
	Items []experimentTextPointer `sproto:"struct,0,array"`
}
type experimentIntRecord struct {
	First *int64 `sproto:"integer,0"`
	Last  *int64 `sproto:"integer,1"`
}
type experimentIntPointer *experimentIntRecord
type experimentIntRows struct {
	Items []experimentIntPointer `sproto:"struct,0,array"`
}

type experimentLayoutRecord struct {
	First *int64 `sproto:"integer,0"`
	Last  *int64 `sproto:"integer,3"`
}
type experimentLayoutRows struct {
	Before []*experimentLayoutRecord `sproto:"struct,0,array"`
	Items  []*experimentLayoutRecord `sproto:"struct,1,array"`
	After  []*experimentLayoutRecord `sproto:"struct,2,array"`
	Tail   *string                   `sproto:"string,3"`
}

func experimentLayoutItems(n int) []*experimentLayoutRecord {
	rows := make([]*experimentLayoutRecord, n)
	for i := range rows {
		if i%17 == 0 {
			continue
		}
		a, b := int64(i+32766), -int64(i+1)
		rows[i] = &experimentLayoutRecord{First: &a, Last: &b}
		if i%5 == 0 {
			rows[i].Last = nil
		}
	}
	return rows
}
