package packingprobe

import (
	"bytes"
	sproto "github.com/cnooljaw/gosproto"
	before "github.com/cnooljaw/gosproto-before"
	"testing"
)

// 对原始切片偏移、尾块和 FF 256-word 邻域做旧新差分，输入及旧输出不得被下一次 Pack 修改。
func TestPackDifferential(t *testing.T) {
	cases := 0
	check := func(raw []byte) {
		t.Helper()
		input := bytes.Clone(raw)
		old := before.Pack(raw)
		got := sproto.Pack(raw)
		if !bytes.Equal(old, got) || !bytes.Equal(raw, input) {
			t.Fatalf("wire/input length=%d old=%x new=%x", len(raw), old, got)
		}
		snapshot := bytes.Clone(got)
		sproto.Pack([]byte{7, 0, 8})
		if !bytes.Equal(snapshot, got) {
			t.Fatal("output ownership")
		}
		cases++
	}
	for _, words := range []int{1, 255, 256, 257} {
		for mask := 0; mask < 256; mask++ {
			for _, value := range []byte{1, 0x7f, 0x80, 0xff} {
				for _, tail := range []int{0, 7} {
					raw := bytes.Repeat([]byte{value}, words*8)
					for i := 0; i < 8; i++ {
						v := byte(0)
						if mask&(1<<i) != 0 {
							v = value
						}
						raw = append(raw, v)
					}
					raw = append(raw, 1, 2, 3, 4, 5, 0, 0, 0)
					raw = append(raw, bytes.Repeat([]byte{0xff}, tail)...)
					check(raw)
					check(append(bytes.Clone(raw), make([]byte, 256)...))
				}
			}
		}
	}
	for _, size := range []int{0, 1, 7, 8, 9, 31, 32, 33, 63, 64, 65, 127, 128, 129, 255, 256, 257, 2046, 2047, 2048, 2049, 4094, 4095, 4096, 4097} {
		for offset := 0; offset < 8; offset++ {
			for kind := 0; kind < 8; kind++ {
				storage := bytes.Repeat([]byte{0xee}, offset+size+8)
				raw := storage[offset : offset+size]
				state := uint32(size + kind + 1)
				for i := range raw {
					state ^= state << 13
					state ^= state >> 17
					state ^= state << 5
					v := byte(state)
					if kind < 7 && i%8 >= kind {
						v = 0
					}
					raw[i] = v
				}
				check(raw)
				if !bytes.Equal(storage[:offset], bytes.Repeat([]byte{0xee}, offset)) || !bytes.Equal(storage[offset+size:], bytes.Repeat([]byte{0xee}, 8)) {
					t.Fatal("slice boundary")
				}
			}
		}
	}
	t.Logf("%d differential inputs", cases)
}
