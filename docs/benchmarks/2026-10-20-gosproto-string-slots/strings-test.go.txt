package sproto

import (
 "bytes"
 "encoding/binary"
 "errors"
 "reflect"
 "runtime"
 "runtime/debug"
 "testing"
)

type slotStrings []string
type slotString string
type slotNamedStrings []slotString
type slotNativeDocument struct { Labels []string `sproto:"string,0,array"` }
type slotOuterDocument struct { Labels slotStrings `sproto:"string,0,array"` }
type slotElementDocument struct { Labels slotNamedStrings `sproto:"string,0,array"` }

// 验证字符串数组的实际元素类型、空值、预算、失败原子性及输入/跨调用独占。
func TestStringArraySlotOwnership(t *testing.T) {
 previous:=debug.SetGCPercent(1)
 defer debug.SetGCPercent(previous)
 for _, count:=range []int{0,1,3,16,128} {
  names:=make([]string,count)
  for i:=range names {names[i]=string([]byte{byte(i),0,255,'x'})}
  named:=make(slotNamedStrings,count)
  for i,s:=range names {named[i]=slotString(s)}
  cases:=[]struct{ message any; fresh func()any }{
   {&slotNativeDocument{Labels:names},func()any{return new(slotNativeDocument)}},
   {&slotOuterDocument{Labels:slotStrings(names)},func()any{return new(slotOuterDocument)}},
   {&slotElementDocument{Labels:named},func()any{return new(slotElementDocument)}},
  }
  for _, sample:=range cases {
   data,err:=Encode(sample.message)
   if err!=nil {t.Fatal(err)}
   first:=sample.fresh()
   if used,err:=DecodeWithLimits(data,first,Limits{MaxElements:count+1});err!=nil || used!=len(data) {t.Fatal(used,err)}
   snapshot:=sample.fresh()
   if _,err:=Decode(data,snapshot);err!=nil {t.Fatal(err)}
   second:=sample.fresh()
   if _,err:=Decode(data,second);err!=nil || !reflect.DeepEqual(second,snapshot) {t.Fatal(err)}
   if count>0 {
    // 修改后一次结果，不能影响先前结果；输入也不成为字符串的可变底层缓冲。
    reflect.ValueOf(second).Elem().Field(0).Index(0).SetString("changed")
    if !reflect.DeepEqual(first,snapshot) {t.Fatal("results alias")}
    bad:=append([]byte{},data...)
    last:=8
    for i:=0;i<count-1;i++ {last+=4+int(binary.LittleEndian.Uint32(bad[last:]))}
    binary.LittleEndian.PutUint32(bad[last:],uint32(len(names[count-1])+1))
    if _,err:=Decode(bad,first);!errors.Is(err,ErrDecode)||!reflect.DeepEqual(first,snapshot) {t.Fatal("bad tail changed destination",err)}
    if _,err:=DecodeWithLimits(data,first,Limits{MaxElements:count});!errors.Is(err,ErrLimit)||!reflect.DeepEqual(first,snapshot) {t.Fatal("budget changed destination",err)}
   }
   wireBefore:=append([]byte{},data...)
   for i:=range data {data[i]^=255}
   runtime.GC()
   if !reflect.DeepEqual(first,snapshot) {t.Fatal("input aliases decoded strings")}
   wire,err:=Encode(first)
   if err!=nil || !bytes.Equal(wire,wireBefore) {t.Fatal("GC or input mutation changed result",err)}
  }
 }
}
