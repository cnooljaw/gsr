package sproto

import (
 "reflect"
 "testing"
)

type stringSlotForeignSource struct {
 Padding *int64 `sproto:"integer,0"`
 Labels []string `sproto:"string,1,array"`
}

type stringSlotForeignTarget struct {
 Padding *int64 `sproto:"integer,0"`
 Labels []slotString `sproto:"string,1,array"`
}

// 验证外来原生字符串描述不能绕过目标实际命名元素类型的反射回退。
func TestStringArrayForeignDescriptorFallback(t *testing.T) {
 source:=&stringSlotForeignTarget{Labels:[]slotString{"first","\x00\xff"}}
 data,err:=Encode(source)
 if err!=nil {t.Fatal(err)}
 targetSchema,err:=GetSprotoType(reflect.TypeFor[stringSlotForeignTarget]())
 if err!=nil {t.Fatal(err)}
 foreignSchema,err:=GetSprotoType(reflect.TypeFor[stringSlotForeignSource]())
 if err!=nil {t.Fatal(err)}
 original:=targetSchema.Fields[1]
 targetSchema.Fields[1]=foreignSchema.Fields[1]
 defer func(){targetSchema.Fields[1]=original}()
 var decoded stringSlotForeignTarget
 if _,err:=Decode(data,&decoded);err!=nil || !reflect.DeepEqual(source,&decoded) {t.Fatal("foreign descriptor changed element semantics",err)}
}
