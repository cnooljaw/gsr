package sproto

import (
	"reflect"
	"unsafe"
)

func experimentStringClone(_ uint16, _ bool, data []byte, _ *SprotoField, value reflect.Value, budget *decodeBudget) error {
	count, err := countChunks(data, budget)
	if err != nil {
		return err
	}
	result := reflect.MakeSlice(value.Type(), count, count)
	for index := 0; index < count; index++ {
		used, content, err := readChunk(data)
		if err != nil {
			return err
		}
		result.Index(index).SetString(string(content))
		data = data[used:]
	}
	value.Set(result)
	return nil
}

func experimentStringCandidate(_ uint16, _ bool, data []byte, _ *SprotoField, value reflect.Value, budget *decodeBudget) error {
	count, err := countChunks(data, budget)
	if err != nil {
		return err
	}
	result := reflect.MakeSlice(value.Type(), count, count)
	var slots []string
	if count > 0 && value.Type().Elem() == reflect.TypeFor[string]() {
		slots = unsafe.Slice((*string)(result.UnsafePointer()), count)
	}
	for index := 0; index < count; index++ {
		used, content, err := readChunk(data)
		if err != nil {
			return err
		}
		if slots != nil {
			slots[index] = string(content)
		} else {
			result.Index(index).SetString(string(content))
		}
		data = data[used:]
	}
	value.Set(result)
	return nil
}

// ExperimentStringSelector 仅在实验归档中切换解码器，选择动作不计入时间块。
func ExperimentStringSelector(message any) func(string) {
	schema, err := GetSprotoType(reflect.TypeOf(message).Elem())
	if err != nil {
		panic(err)
	}
	seen := make(map[*SprotoType]bool)
	fields := []*SprotoField{}
	var visit func(*SprotoType)
	visit = func(st *SprotoType) {
		if st == nil || seen[st] {
			return
		}
		seen[st] = true
		for _, f := range st.Fields {
			if f.Array && f.Wire == WireBytesName && f.valueType.Kind() == reflect.Slice && f.valueType.Elem().Kind() == reflect.String {
				fields = append(fields, f)
			}
			visit(f.st)
		}
	}
	visit(schema)
	return func(mode string) {
		fn := decoder(decodeStringSlice)
		switch mode {
		case "baseline":
		case "clone":
			fn = experimentStringClone
		case "candidate":
			fn = experimentStringCandidate
		default:
			panic("invalid experiment decoder")
		}
		for _, f := range fields {
			f.dec = fn
		}
	}
}
