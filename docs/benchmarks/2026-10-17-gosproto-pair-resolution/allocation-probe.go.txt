package benchmarks

import (
	"encoding/json"
	"os"
	"runtime"
	"testing"

	sproto "github.com/cnooljaw/gosproto"
	"github.com/cnooljaw/gsr/examples/websocket-gate/gen/sproto/c2s"
)

type allocationProbe struct {
	Name          string `json:"name"`
	Calls         int    `json:"calls"`
	GCBeforeIndex int    `json:"gcBeforeIndex"`
	Bytes         uint64 `json:"bytes"`
	Mallocs       uint64 `json:"mallocs"`
	GCs           uint32 `json:"completedGcCycles"`
}

// TestAllocationProbe 仅在隔离模块观测 tiny 块与 GC 边界，不作为协议或计时门禁。
func TestAllocationProbe(t *testing.T) {
	if runtime.GOMAXPROCS(0) != 1 {
		t.Fatal("requires one P")
	}
	typeID, session := int64(1), int64(1)
	prefix, err := sproto.Encode(&c2s.Package{Type: &typeID, Session: &session})
	if err != nil || len(prefix) != 6 {
		t.Fatalf("prefix len=%d err=%v", len(prefix), err)
	}
	for range 5 {
		runtime.GC()
	}
	var rows []allocationProbe
	for repeat := 0; repeat < 3; repeat++ {
		for _, mid := range []int{-1, 63, 64} {
			for _, allocate := range []bool{false, true} {
				runtime.GC()
				var start, end runtime.MemStats
				runtime.ReadMemStats(&start)
				for index := 0; index < 128; index++ {
					if index == mid {
						runtime.GC()
					}
					if allocate {
						encodedSink = make([]byte, 6)
					}
				}
				runtime.ReadMemStats(&end)
				name := "empty"
				if allocate {
					name = "prefix6"
				}
				rows = append(rows, allocationProbe{Name: name, Calls: 128, GCBeforeIndex: mid, Bytes: end.TotalAlloc - start.TotalAlloc, Mallocs: end.Mallocs - start.Mallocs, GCs: end.NumGC - start.NumGC})
			}
		}
	}
	contents, err := json.MarshalIndent(rows, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("GSR_ALLOCATION_PROBE"), append(contents, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
}
