package unpackprobe

import (
	"bytes"
	"errors"
	sproto "github.com/cnooljaw/gosproto"
	before "github.com/cnooljaw/gosproto-before"
	"testing"
)

func oldClass(err error) int {
	switch {
	case err == nil:
		return 0
	case errors.Is(err, before.ErrInvalidLimits):
		return 1
	case errors.Is(err, before.ErrLimit):
		return 2
	case errors.Is(err, before.ErrUnpack):
		return 3
	default:
		return 4
	}
}
func newClass(err error) int {
	switch {
	case err == nil:
		return 0
	case errors.Is(err, sproto.ErrInvalidLimits):
		return 1
	case errors.Is(err, sproto.ErrLimit):
		return 2
	case errors.Is(err, sproto.ErrUnpack):
		return 3
	default:
		return 5
	}
}

// 锁定预算和迟到坏帧的错误顺序，并检查独占输出及源切片不变。
func TestUnpackBudgetDifferential(t *testing.T) {
	cases := 0
	check := func(raw []byte, limit int) {
		t.Helper()
		snapshot := bytes.Clone(raw)
		want, oldErr := before.UnpackWithLimit(raw, limit)
		got, err := sproto.UnpackWithLimit(raw, limit)
		if oldClass(oldErr) != newClass(err) || !bytes.Equal(want, got) || (want == nil) != (got == nil) || !bytes.Equal(raw, snapshot) {
			t.Fatalf("input=%x limit=%d old=%x/%v new=%x/%v", raw, limit, want, oldErr, got, err)
		}
		saved := bytes.Clone(got)
		sproto.Unpack([]byte{0})
		if !bytes.Equal(saved, got) {
			t.Fatal("output ownership")
		}
		cases++
	}
	tails := [][]byte{nil, {1, 0x80}, {3, 0x7f}, {0xff}, {0xff, 0}, {0xff, 0, 1, 2, 3, 4, 5, 6, 7, 8}, {0}, {0, 0xff}}
	for _, words := range []int{0, 1, 2, 3, 7, 8, 9, 15, 16, 17, 63, 64, 65, 511, 512, 513} {
		for _, tail := range tails {
			for offset := 0; offset < 8; offset++ {
				wire := append(make([]byte, words), tail...)
				backing := bytes.Repeat([]byte{0xee}, offset+len(wire)+8)
				copy(backing[offset:], wire)
				raw := backing[offset : offset+len(wire)]
				for _, limit := range []int{-1, 0, 1, 7, 8, 15, 16, 31, 32, words*8 - 1, words * 8, words*8 + 8, int(^uint(0) >> 1)} {
					check(raw, limit)
				}
				if !bytes.Equal(backing[:offset], bytes.Repeat([]byte{0xee}, offset)) || !bytes.Equal(backing[offset+len(wire):], bytes.Repeat([]byte{0xee}, 8)) {
					t.Fatal("slice boundary")
				}
			}
		}
	}
	for mask := 0; mask < 256; mask++ {
		for _, value := range []byte{1, 0x7f, 0x80, 0xff} {
			wire := []byte{byte(mask)}
			if mask == 255 {
				wire = append(wire, 0)
			}
			for bit := 0; bit < 8; bit++ {
				if mask&(1<<bit) != 0 {
					wire = append(wire, value)
				}
			}
			for _, words := range []int{0, 1, 2, 7, 8, 9, 63} {
				raw := append(make([]byte, words), wire...)
				need := (words + 1) * 8
				for _, limit := range []int{1, need - 1, need, need + 1} {
					check(raw, limit)
				}
			}
		}
	}
	for _, size := range []int{0, 1, 2, 7, 8, 9, 31, 32, 33, 63, 64, 65, 127, 128, 129, 255, 256, 257, 2048, 4096} {
		for seed := 0; seed < 64; seed++ {
			raw := make([]byte, (seed%9)*2+size)
			state := uint32(seed + size + 1)
			for i := len(raw) - size; i < len(raw); i++ {
				state ^= state << 13
				state ^= state >> 17
				state ^= state << 5
				raw[i] = byte(state)
			}
			for _, limit := range []int{1, 64, 4096} {
				check(raw, limit)
			}
		}
	}
	t.Logf("%d input/budget differential cases", cases)
}
