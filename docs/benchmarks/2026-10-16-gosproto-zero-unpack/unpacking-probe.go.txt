// 同一进程分别比较 Unpack、PDecode 与冻结的纯 Decode，不相加分段耗时。
func BenchmarkPairedUnpacking(b *testing.B) {
	for _, sample := range representativeCases(b) {
		ops, wires := representativeOperations(b, sample)
		checkRepresentativeWire(b, sample, wires)
		_, factory := representativeMessages(sample.Type)
		raw := wires["sproto_encode"]
		packed := wires["sproto_pack"]
		pair := []representativeOperation{
			{"before_unpack", func() error { var err error; encodedSink, err = before.Unpack(packed); return err }},
			{"before_pdecode", func() error {
				message := factory()
				err := before.DecodePacked(packed, message)
				representativeSink = message
				return err
			}},
			{"before_decode", func() error {
				message := factory()
				_, err := before.Decode(raw, message)
				representativeSink = message
				return err
			}},
		}
		for _, op := range ops {
			if op.name == "sproto_unpack" || op.name == "sproto_pdecode" || op.name == "sproto_decode" || op.name == "protobuf_decode" {
				pair = append(pair, op)
			}
		}
		runPaired(b, sample.Name, pair)
	}
}

// Echo 保留原手工 frame 解码入口，逐项与旧库及 PB 配对。
func BenchmarkPairedUnpackEcho(b *testing.B) {
	contents, err := os.ReadFile("cases.json")
	if err != nil {
		b.Fatal(err)
	}
	var cases []benchmarkCase
	if err := json.Unmarshal(contents, &cases); err != nil {
		b.Fatal(err)
	}
	for _, sample := range cases {
		text := strings.Repeat(sample.Pattern, sample.Repeat)
		old, now := beforeEchoCodecs(text), echoCodecs(text)
		var pair []representativeOperation
		for _, shape := range []string{"payload", "frame"} {
			left, right, pb := old["sproto_"+shape], now["sproto_"+shape], now["protobuf_"+shape]
			wire, err := left.encode()
			if err != nil {
				b.Fatal(err)
			}
			same, err := right.encode()
			if err != nil || !bytes.Equal(wire, same) {
				b.Fatal("wire", err)
			}
			pbwire, err := pb.encode()
			if err != nil {
				b.Fatal(err)
			}
			for _, op := range []struct {
				name   string
				decode func([]byte) (string, error)
				wire   []byte
			}{{"before_" + shape, left.decode, wire}, {"sproto_" + shape, right.decode, wire}, {"protobuf_" + shape, pb.decode, pbwire}} {
				pair = append(pair, representativeOperation{op.name, func() error { var err error; decodedSink, err = op.decode(op.wire); return err }})
			}
		}
		runPaired(b, sample.Name, pair)
	}
}

// 零段邻域和短零/FF 交替形态覆盖批量扫描的边界；既有稀疏形态继续保留。
func BenchmarkPairedUnpackShapes(b *testing.B) {
	shapes := unpackShapes()
	for _, words := range []int{1, 2, 3, 7, 8, 9, 15, 16, 17, 63, 64, 65} {
		shapes = append(shapes, unpackShape{fmt.Sprintf("zeroWords%d", words), make([]byte, words*8)})
	}
	for _, words := range []int{1, 2, 3, 7, 9} {
		raw := bytes.Repeat(append(make([]byte, words*8), bytes.Repeat([]byte{0x7f}, 8)...), 64)
		shapes = append(shapes, unpackShape{fmt.Sprintf("zeroRun%dx64", words), raw})
	}
	for _, shape := range shapes {
		packed := before.Pack(shape.raw)
		a, err := before.Unpack(packed)
		if err != nil {
			b.Fatal(err)
		}
		z, err := sproto.Unpack(packed)
		if err != nil || !bytes.Equal(a, z) || !bytes.Equal(z, shape.raw) {
			b.Fatal(shape.name, err)
		}
		runPaired(b, shape.name, []representativeOperation{
			{"before_unpack", func() error { var err error; encodedSink, err = before.Unpack(packed); return err }},
			{"sproto_unpack", func() error { var err error; encodedSink, err = sproto.Unpack(packed); return err }},
		})
	}
}
