package sproto

import (
	"bytes"
	"encoding/binary"
	"math/rand"
	"reflect"
	"testing"
	"unsafe"
)

type experimentScalarRecord struct {
	Padding [32]byte
	First   *int64 `sproto:"integer,0"`
	Last    *int64 `sproto:"integer,2"`
}

// 验证直接读取与旧路径在合法值、未知字段、截断、错误顺序及预算状态上保持一致。
func TestRecordScalarDifferential(t *testing.T) {
	schema, err := GetSprotoType(reflect.TypeFor[experimentScalarRecord]())
	if err != nil {
		t.Fatal(err)
	}
	check := func(input []byte) {
		t.Helper()
		before := append([]byte(nil), input...)
		for _, limit := range []Limits{{MaxDepth: 64, MaxElements: 100000}, {MaxDepth: 1, MaxElements: 1}} {
			for _, depth := range []int{0, 1} {
				expected := new(experimentScalarRecord)
				left := decodeBudget{limits: limit, depth: depth}
				used, wantErr := experimentStringBaseline(input, schema, decodePointer{value: unsafe.Pointer(expected)}, &left)
				for _, fn := range []func([]byte, *SprotoType, decodePointer, *decodeBudget) (int, error){experimentStringClone, experimentStringCandidate} {
					actual := new(experimentScalarRecord)
					right := decodeBudget{limits: limit, depth: depth}
					got, gotErr := fn(input, schema, decodePointer{value: unsafe.Pointer(actual)}, &right)
					if got != used || gotErr != wantErr || left != right || !reflect.DeepEqual(expected, actual) {
						t.Fatalf("wire=%x used=%d/%d error=%v/%v budget=%+v/%+v result=%+v/%+v", input, used, got, wantErr, gotErr, left, right, expected, actual)
					}
				}
			}
		}
		if !bytes.Equal(input, before) {
			t.Fatal("input modified")
		}
	}
	for _, header := range []uint16{0, 1, 2, 4, 65534, 65535} {
		for _, size := range []uint32{0, 1, 3, 4, 5, 7, 8, 9, 0xffffffff} {
			for _, count := range []uint16{0, 1, 2, 3} {
				wire := make([]byte, 28)
				binary.LittleEndian.PutUint16(wire, count)
				for i := 0; i < int(count); i++ {
					binary.LittleEndian.PutUint16(wire[2+i*2:], header)
				}
				binary.LittleEndian.PutUint32(wire[2+int(count)*2:], size)
				for i := 6 + int(count)*2; i < len(wire); i++ {
					wire[i] = byte(i * 17)
				}
				for end := 0; end <= len(wire); end++ {
					check(wire[:end])
				}
			}
		}
	}
	// 随机输入仍以固定种子复现，并同时覆盖深度/集合超限时的返回顺序。
	rng := rand.New(rand.NewSource(20261021))
	for range 5000 {
		input := make([]byte, rng.Intn(96))
		rng.Read(input)
		check(input)
	}
	// 官方wire允许未知字段任意合法长度，已知integer只允许4或8字节。
	for _, tag := range []uint16{0, 1, 2, 3} {
		for _, value := range []int64{0, 32766, 32767, -1, -2147483648, 2147483647, 1 << 40, -1 << 40} {
			n := int64(value)
			message := &experimentScalarRecord{First: &n, Last: &n}
			wire, err := Encode(message)
			if err != nil {
				t.Fatal(err)
			}
			check(wire)
			for end := 0; end < len(wire); end++ {
				check(wire[:end])
			}
		}
		wire := make([]byte, 15)
		binary.LittleEndian.PutUint16(wire, 2)
		binary.LittleEndian.PutUint16(wire[2:], tag*2+1)
		binary.LittleEndian.PutUint32(wire[6:], 5)
		check(wire)
	}
	// 既有可变描述允许nil，当前字段路由映射仍需被相同方式解释。
	saved := schema.Fields[0]
	schema.Fields[0] = nil
	check([]byte{1, 0, 0, 0, 1, 0, 0, 0, 9})
	schema.Fields[0] = saved
}
