type unpackBlock struct {
 Arm string `json:"arm"`
 Calls int `json:"calls"`
 Nanoseconds int64 `json:"nanoseconds"`
 Bytes, Allocs uint64
 GC uint32
 PauseNs uint64
 HeapBefore, HeapAfter uint64
}

type unpackQuartet struct {
 Round, Index int
 Order string
 Blocks []unpackBlock
 Ratio float64
}

// TestStringInProcessExperiment 在同一模块/进程交错消息头parser，保留全部GC、分配与时间块。
func TestStringInProcessExperiment(t *testing.T) {
 if runtime.GOMAXPROCS(0)!=1 { t.Fatal("GOMAXPROCS1 required") }
 name, mode := os.Getenv("GSR_CASE"), os.Getenv("GSR_MODE")
 var source any
 var wire,packed []byte
 var newMessage func() any
 for _, sample := range representativeCases(t) {
  if sample.Name!=name {continue}
  _,wires:=representativeOperations(t,sample)
  checkRepresentativeWire(t,sample,wires)
  wire,packed=wires["sproto_encode"],wires["sproto_pack"]
  _,newMessage=representativeMessages(sample.Type)
 source=newMessage();if err:=json.Unmarshal(sample.Data,source);err!=nil {t.Fatal(err)}
 }
 if wire==nil {
  var message any
  count:=33
  switch name {case "records7":count=7;case "records8":count=8;case "records33":count=33;case "records128":count=128;case "prefix31":default:t.Fatal("unknown case")}
  rows:=&experimentLayoutRows{Items:experimentLayoutItems(count),After:experimentLayoutItems(7)}
  if name=="prefix31" {rows.Before=experimentLayoutItems(31)}
  label:="tail/中文";rows.Tail=&label
  message=rows;source=message;newMessage=func()any{return new(experimentLayoutRows)}
  var err error
  wire,err=sp.Encode(message)
  if err!=nil {t.Fatal(err)}
  packed=sp.Pack(wire)
 }
 operation:=os.Getenv("GSR_OP")
 if operation!="encode" && operation!="pencode" {t.Fatal("invalid operation")}
 selectMode:=sp.ExperimentStringMode
 defer selectMode("baseline")
 right:="candidate"
 switch mode {
 case "self": right="baseline"
 case "clone": right="clone"
 case "pass1","pass2":
 default:t.Fatal("unknown mode")
 }
 selections:=map[string]string{"A":"baseline","B":right}
 beforeInput:=append([]byte(nil),packed...)
 expected:=newMessage()
 selectMode("baseline")
 if _,err:=sp.Decode(wire,expected);err!=nil {t.Fatal(err)}
 for _,variant:=range []string{"baseline",right} {
  selectMode(variant)
  encoded,err:=sp.Encode(source);if err!=nil || !bytes.Equal(encoded,wire) {t.Fatal("body mismatch",err)}
  frame,err:=sp.EncodePacked(source);if err!=nil || !bytes.Equal(frame,packed) {t.Fatal("packed mismatch",err)}
 }
 if os.Getenv("GSR_PREFLIGHT")=="1" {return}
 entries:=map[string]func(interface{},sp.Limits,bool,[]byte)([]byte,error){"baseline":sp.ExperimentStaticBaseline,"clone":sp.ExperimentStaticClone,"candidate":sp.ExperimentStaticCandidate}
 runs:=map[string]func(){}
 for _,arm:=range []string{"A","B"} {
  entry:=entries[selections[arm]]
  runs[arm]=func(){output,err:=entry(source,sp.Limits{},operation=="pencode",nil);if err!=nil {t.Fatal(err)};encodedSink=output}
 }
 runtime.GC()
 for _,arm:=range []string{"A","B"} {
  selectMode(selections[arm])
  start:=time.Now()
  for time.Since(start)<50*time.Millisecond {for range 64 {runs[arm]()}}
 }
 selectMode("baseline")
 start:=time.Now()
 for range 256 {runs["A"]()}
 calls:=int(float64(20*time.Millisecond)/float64(time.Since(start))*256)
 calls=max(1,min(calls,1<<20))
 blocks:=[]unpackQuartet{}
 for round:=0; round<6; round++ {
  for quartet:=0; quartet<4; quartet++ {
   order:="ABBA"
   if (round+quartet)%2!=0 {order="BAAB"}
   q:=unpackQuartet{Round:round,Index:quartet,Order:order}
   logRatio:=0.0
   for _,label:=range order {
    arm:=string(label)
    var before,after runtime.MemStats
    selectMode(selections[arm])
    runtime.ReadMemStats(&before)
    run:=runs[arm]
    start:=time.Now()
    for range calls {run()}
    ns:=time.Since(start).Nanoseconds()
    runtime.ReadMemStats(&after)
    block:=unpackBlock{Arm:arm,Calls:calls,Nanoseconds:ns,Bytes:after.TotalAlloc-before.TotalAlloc,Allocs:after.Mallocs-before.Mallocs,GC:after.NumGC-before.NumGC,PauseNs:after.PauseTotalNs-before.PauseTotalNs,HeapBefore:before.HeapAlloc,HeapAfter:after.HeapAlloc}
    q.Blocks=append(q.Blocks,block)
    sign:=1.0
    if arm=="A" {sign=-1}
    logRatio+=sign*math.Log(float64(ns)/float64(calls))
   }
   q.Ratio=math.Exp(logRatio/2)
   blocks=append(blocks,q)
  }
 }
 selectMode("baseline");afterWire,afterErr:=sp.Encode(source);if afterErr!=nil || !bytes.Equal(afterWire,wire) {t.Fatal("source modified",afterErr)}
 if !bytes.Equal(packed,beforeInput) {t.Fatal("input modified")}
 raw,err:=json.MarshalIndent(struct{Name,Operation,Mode string;CallsPerBlock int;Quartets []unpackQuartet}{name,operation,mode,calls,blocks},"","  ")
 if err!=nil {t.Fatal(err)}
 if err:=os.WriteFile(os.Getenv("GSR_OUTPUT"),append(raw,'\n'),0644);err!=nil {t.Fatal(err)}
 runtime.KeepAlive(runs)
}

type experimentText string
type experimentTextRecord struct {First *string `sproto:"string,0"`;Last *string `sproto:"string,1"`}
type experimentTextRows struct {Items []*experimentTextRecord `sproto:"struct,0,array"`}
type experimentNamedTextRecord struct {First *experimentText `sproto:"string,0"`;Last *experimentText `sproto:"string,1"`}
type experimentNamedTextRows struct {Items []*experimentNamedTextRecord `sproto:"struct,0,array"`}

type experimentTextPointer *experimentTextRecord
type experimentPointerRows struct {Items []experimentTextPointer `sproto:"struct,0,array"`}
type experimentIntRecord struct {First *int64 `sproto:"integer,0"`;Last *int64 `sproto:"integer,1"`}
type experimentIntPointer *experimentIntRecord
type experimentIntRows struct {Items []experimentIntPointer `sproto:"struct,0,array"`}

type experimentLayoutRecord struct {First *int64 `sproto:"integer,0"`;Last *int64 `sproto:"integer,3"`}
type experimentLayoutRows struct {Before []*experimentLayoutRecord `sproto:"struct,0,array"`;Items []*experimentLayoutRecord `sproto:"struct,1,array"`;After []*experimentLayoutRecord `sproto:"struct,2,array"`;Tail *string `sproto:"string,3"`}
func experimentLayoutItems(n int) []*experimentLayoutRecord {
 rows:=make([]*experimentLayoutRecord,n)
 for i:=range rows {if i%17==0 {continue};a,b:=int64(i+32766),-int64(i+1);rows[i]=&experimentLayoutRecord{First:&a,Last:&b};if i%5==0 {rows[i].Last=nil}}
 return rows
}

// TestStaticAllocationParity 保护静态调用树与未包装正常入口拥有相同分配数量。
func TestStaticAllocationParity(t *testing.T) {
 for _,sample:=range representativeCases(t) {
  _,fresh:=representativeMessages(sample.Type);source:=fresh();if err:=json.Unmarshal(sample.Data,source);err!=nil {t.Fatal(err)}
  for _,packed:=range []bool{false,true} {
   entries:=[]struct{name string;entry func(interface{},sp.Limits,bool,[]byte)([]byte,error)}{{"normal",sp.ExperimentStaticNormal},{"baseline",sp.ExperimentStaticBaseline},{"clone",sp.ExperimentStaticClone},{"candidate",sp.ExperimentStaticCandidate}}
   expected:=-1.0
   for _,e:=range entries {
    n:=testing.AllocsPerRun(64,func(){output,err:=e.entry(source,sp.Limits{},packed,nil);if err!=nil {t.Fatal(err)};encodedSink=output})
    if expected<0 {expected=n}else if n!=expected {t.Fatal(sample.Name,packed,e.name,n,expected)}
    t.Logf("%s packed=%v %s allocs=%g",sample.Name,packed,e.name,n)
   }
  }
 }
}
