package sproto

import (
	"fmt"
	"reflect"
	"unsafe"
)

// encodedIntegerRecordSize 只用于数组入口已核对实际 Go 类型及全部字段 owner/形状的记录。
// 保留与通用结构编码相同的深度、字段数、tag 间隔和字节预算，不缓存任何标量值。
func encodedIntegerRecordSize(schema *SprotoType, pointer encodePointer, budget *decodeBudget, depth int, layout *encodeLayout) (int, error) {
	if depth > budget.limits.MaxDepth || budget.limits.MaxBytes < 2 {
		return 0, ErrLimit
	}
	layoutIndex := layout.add(0)
	if pointer.value == nil {
		return 2, nil
	}
	elements := budget.elements
	size, headers, lastTag := 2, 0, -1
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		bits, present := pointer.integer(field.encOffset)
		if !present {
			continue
		}
		if 1 > budget.limits.MaxElements-elements {
			budget.elements = elements
			return 0, ErrLimit
		}
		elements++
		gap := field.Tag - lastTag - 1
		if gap > 32768 {
			budget.elements = elements
			return 0, fmt.Errorf("%w: tag gap", ErrEncode)
		}
		if gap > 0 {
			size += 2
			headers++
		}
		size += 2
		headers++
		lastTag = field.Tag
		if bits > MaxEmbeddedInt {
			size += 4 + integerWidth(bits)
		}
		if size > budget.limits.MaxBytes {
			budget.elements = elements
			return 0, ErrLimit
		}
	}
	budget.elements = elements
	layout.set(layoutIndex, headers)
	return size, nil
}

// encodeIntegerRecordInto 复用同一次预检的布局，字段值仍逐个读取并写入独占输出。
func encodeIntegerRecordInto(schema *SprotoType, pointer encodePointer, output []byte, count int) int {
	writeUint16(output, uint16(count))
	if pointer.value == nil {
		return 2
	}
	headerOffset, bodyOffset, previous := 2, 2+count*2, -1
	for _, index := range schema.order {
		field := schema.Fields[index]
		if field.Tag < 0 {
			continue
		}
		bits, present := pointer.integer(field.encOffset)
		if !present {
			continue
		}
		if skip := skipTag(previous, field.Tag); skip > 0 {
			writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], skip)
			headerOffset += 2
		}
		previous = field.Tag
		// inline 路径写完头部后立即继续，外置路径独立完成内容长度与写出。
		if bits <= MaxEmbeddedInt {
			writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], uint16((bits+1)*2))
			headerOffset += 2
			continue
		}
		body := output[bodyOffset:]
		_, length := encodeIntegerInto(bits, body)
		writeUint16(output[headerOffset:headerOffset+2:headerOffset+2], 0)
		writeUint32(body, uint32(length))
		headerOffset += 2
		bodyOffset += 4 + length
	}
	return bodyOffset
}

// encodeIntegerRecordArrayInto 只消费本次预检形成的连续整数记录布局，不包含嵌套布局项。
func encodeIntegerRecordArrayInto(field *SprotoField, value reflect.Value, output []byte, layout *encodeLayout, binding encodeBinding, pointers []unsafe.Pointer) int {
	count := value.Len()
	index, offset := 0, 0
	for index < count {
		cursor := layout.cursor
		var headers []uint32
		if cursor < len(layout.local) {
			end := min(len(layout.local), cursor+count-index)
			headers = layout.local[cursor:end:end]
		} else {
			start := cursor - len(layout.local)
			end := start + count - index
			headers = layout.extra[start:end:end]
		}
		layout.cursor += len(headers)
		for _, header := range headers {
			var pointer encodePointer
			if pointers != nil {
				pointer = encodePointer{value: pointers[index]}
			} else {
				pointer = messageEncodePointer(value.Index(index), binding)
			}
			length := encodeIntegerRecordInto(field.st, pointer, output[offset+4:], int(header))
			writeUint32(output[offset:], uint32(length))
			offset += 4 + length
			index++
		}
	}
	return offset
}
