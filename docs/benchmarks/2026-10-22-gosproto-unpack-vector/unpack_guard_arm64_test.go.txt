func guardedVectorBuffer(t *testing.T, size int) ([]byte, []byte) {
	t.Helper()
	page := os.Getpagesize()
	if size > page {
		t.Fatal("guarded fixture too large")
	}
	mapping, err := syscall.Mmap(-1, 0, 3*page, syscall.PROT_READ|syscall.PROT_WRITE, syscall.MAP_ANON|syscall.MAP_PRIVATE)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		if err := syscall.Munmap(mapping); err != nil {
			t.Error(err)
		}
	})
	if err := syscall.Mprotect(mapping[:page], syscall.PROT_NONE); err != nil {
		t.Fatal(err)
	}
	if err := syscall.Mprotect(mapping[2*page:], syscall.PROT_NONE); err != nil {
		t.Fatal(err)
	}
	middle := mapping[page : 2*page : 2*page]
	for i := range middle[:page-size] {
		middle[i] = 0xa5
	}
	return middle[page-size : page : page], middle
}

// 保护页验证所有mask的源尾/目标尾不会被向量窗口越过，并覆盖短尾回退及连续零/FF段。
func TestUnpackVectorGuardPages(t *testing.T) {
	for mask := 0; mask < 256; mask++ {
		for _, count := range []int{1, 127, 128, 129} {
			t.Run(fmt.Sprintf("%d/%d", mask, count), func(t *testing.T) {
				var word [8]byte
				packedWord := []byte{byte(mask)}
				if mask == 255 {
					packedWord = append(packedWord, 0)
				}
				for i := range word {
					if mask&(1<<i) != 0 {
						word[i] = byte(0x81 + i)
						packedWord = append(packedWord, word[i])
					}
				}
				packed := bytes.Repeat(packedWord, count)
				want := bytes.Repeat(word[:], count)
				input, inputPage := guardedVectorBuffer(t, len(packed))
				copy(input, packed)
				if err := syscall.Mprotect(inputPage, syscall.PROT_READ); err != nil {
					t.Fatal(err)
				}
				got, err := UnpackWithLimit(input, len(want))
				if err != nil || !bytes.Equal(got, want) {
					t.Fatalf("public result: %v", err)
				}
				output, outputPage := guardedVectorBuffer(t, len(want))
				if !unpackVector(output, input) || !bytes.Equal(output, want) {
					t.Fatal("vector result")
				}
				for _, b := range outputPage[:len(outputPage)-len(output)] {
					if b != 0xa5 {
						t.Fatal("write before output")
					}
				}
			})
		}
	}
}
