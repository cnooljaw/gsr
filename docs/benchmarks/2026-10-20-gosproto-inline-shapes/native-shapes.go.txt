var nativeShapes = []string{"zero", "inline_max", "external32", "negative32", "external64", "mixed", "half_missing", "all_missing"}

func nativeShapeValue(count int, shape string) any {
 var message any = &nativeShape8{}
 if count == 64 { message = &nativeShape64{} }
 record := reflect.ValueOf(message).Elem()
 for i := 0; i < count; i++ {
  var number int64
  switch shape {
  case "zero": number = 0
  case "inline_max": number = 32766
  case "external32": number = 32767
  case "negative32": number = -1
  case "external64": number = 1<<48
  case "mixed": number = []int64{0,32766,32767,-1,1<<48,math.MinInt64,math.MaxInt64,1}[i%8]
  case "half_missing":
   if i%2==0 {continue}; number=1
  case "all_missing": continue
  default: panic("unknown shape")
  }
  value:=new(int64);*value=number
  record.Field(i).Set(reflect.ValueOf(value))
 }
 return message
}

func checkNativeShape(t *testing.T, count int, shape string, message any) {
 t.Helper()
 left,err:=before.Encode(message);if err!=nil {t.Fatal(err)}
 right,err:=sproto.Encode(message);if err!=nil || !bytes.Equal(left,right) {t.Fatalf("wire %d/%s: %v",count,shape,err)}
 packed,err:=sproto.EncodePacked(message);if err!=nil || !bytes.Equal(packed,before.Pack(left)) {t.Fatalf("packed %d/%s: %v",count,shape,err)}
 // 两边逐字段恢复相同原值与presence；数组专用路径不会被这些根结构触发。
 oldValue:=reflect.New(reflect.TypeOf(message).Elem()).Interface()
 newValue:=reflect.New(reflect.TypeOf(message).Elem()).Interface()
 used,err:=before.Decode(left,oldValue);if err!=nil || used!=len(left) || !reflect.DeepEqual(oldValue,message) {t.Fatalf("baseline decode %d/%s: %v",count,shape,err)}
 used,err=sproto.Decode(right,newValue);if err!=nil || used!=len(right) || !reflect.DeepEqual(newValue,message) {t.Fatalf("candidate decode %d/%s: %v",count,shape,err)}
 // 精确字节预算必须接纳，少1字节必须失败；检查失败不篡改借用输入。
 oldBefore:=reflect.ValueOf(message).Elem().Field(0).Interface()
 for _,limit:=range []int{len(left)-1,len(left)} {
  a,ae:=before.EncodeWithLimits(message,before.Limits{MaxBytes:limit})
  b,be:=sproto.EncodeWithLimits(message,sproto.Limits{MaxBytes:limit})
  if (ae==nil)!=(be==nil) || !bytes.Equal(a,b) || (limit<len(left) && (ae==nil || a!=nil || b!=nil)) {t.Fatalf("budget %d/%s/%d: %v/%v",count,shape,limit,ae,be)}
 }
 if !reflect.DeepEqual(reflect.ValueOf(message).Elem().Field(0).Interface(),oldBefore) {t.Fatal("input changed")}
}

// TestNativeShapeFixtures 保护合成根结构的wire、presence、恢复值和精确字节预算对照。
func TestNativeShapeFixtures(t *testing.T) {
 for _,count:=range []int{8,64} {for _,shape:=range nativeShapes {
  message:=nativeShapeValue(count,shape)
  checkNativeShape(t,count,shape,message)
 }}
}

func pairOperations(t *testing.T, mode string) []pairOperation {
 t.Helper()
 var result []pairOperation
 for _,count:=range []int{8,64} {for _,shape:=range nativeShapes {
  message:=nativeShapeValue(count,shape)
  checkNativeShape(t,count,shape,message)
  for _,op:=range []string{"encode","pencode"} {
   var left,right func()error
   if op=="encode" {
    left=func()error {var err error;encodedSink,err=before.Encode(message);return err}
    right=func()error {var err error;encodedSink,err=sproto.Encode(message);return err}
   } else {
    left=func()error {var err error;encodedSink,err=before.EncodePacked(message);return err}
    right=func()error {var err error;encodedSink,err=sproto.EncodePacked(message);return err}
   }
   if mode=="self" {left=right}
   result=append(result,pairOperation{fmt.Sprintf("native/%d/%s/%s",count,shape,op),left,right})
  }
 }}
 return result
}
