 "bytes"
 "testing"

// TestPreflightCapComparison 保护预检切片容量变化不改变任意预算下的结果与错误优先级。
func TestPreflightCapComparison(t *testing.T) {
 check := func(input []byte, limit int) {
  t.Helper()
  source := append([]byte(nil), input...)
  a,ae:=ExperimentUnpackOriginal(input,limit)
  b,be:=UnpackWithLimit(input,limit)
  if ae!=be || !bytes.Equal(a,b) || (a==nil)!=(b==nil) || cap(a)!=cap(b) || !bytes.Equal(input,source) {
   t.Fatalf("input=%x limit=%d baseline=%x/%v candidate=%x/%v",input,limit,a,ae,b,be)
  }
 }
 for mask:=0;mask<256;mask++ {
  raw:=make([]byte,64)
  for i:=range raw {if mask&(1<<(i%8))!=0 {raw[i]=byte(128+i)}}
  packed:=Pack(raw)
  for cut:=0;cut<=len(packed);cut++ {
   backing:=make([]byte,cut+37);copy(backing[3:],packed[:cut]);input:=backing[3:3+cut]
   for _,limit:=range []int{-1,0,1,7,8,9,15,16,63,64,65,int(^uint(0)>>1)} {check(input,limit)}
  }
 }
 state:=uint32(0x1729)
 for round:=0;round<4096;round++ {
  state=state*1664525+1013904223
  wire:=make([]byte,int(state%257));for i:=range wire {state=state*1664525+1013904223;wire[i]=byte(state>>24)}
  for _,limit:=range []int{1,8,16,64,2048,8<<20} {check(wire,limit)}
 }
}
