package sproto

func experimentEncodeWithLimits(message interface{}, limits Limits, packResult bool, prefix []byte) ([]byte, error) {
	output, encoded, err := encodeBufferWithLimits(message, limits, packResult, prefix)
	if err != nil {
		return nil, err
	}
	if packResult {
		return ExperimentPackInto(output, encoded), nil
	}
	return output, nil
}

func ExperimentEncodePacked(message interface{}) ([]byte, error) {
	return experimentEncodeWithLimits(message, Limits{}, true, nil)
}
func ExperimentPack(src []byte) []byte {
	return ExperimentPackInto(make([]byte, packedBufferSize(len(src))), src)
}
