package sproto
import "math/bits"
func ExperimentUnpackClone(src []byte, maxBytes int) ([]byte, error) {
	if maxBytes < 1 {
		return nil, ErrInvalidLimits
	}
	// 首次只扫描 tag/连续块长度，在分配前确认完整边界及有限的确切输出长度。
	size, err, handled := experimentSizeHelper(src, maxBytes)
	if err != nil {
		return nil, err
	}
	if !handled {
		for remaining := src; len(remaining) > 0; {
			header := remaining[0]
			consumed, produced := 1, 8
			if header == 0 {
				if len(remaining) >= 2 && remaining[1] == 0 {
					words := (maxBytes - size) / 8
					if words < 2 {
						return nil, ErrLimit
					}
					// 多检查一个零 mask 即可判定超额，不为小预算扫描整段大输入。
					end := min(len(remaining), words+1)
					run := 2 + zeroHeaderPrefix(remaining[2:end])
					if run > words {
						return nil, ErrLimit
					}
					consumed, produced = run, run*8
				}
			} else if header == 0xff {
				if len(remaining) < 2 {
					return nil, ErrUnpack
				}
				produced = (int(remaining[1]) + 1) * 8
				consumed = 2 + produced
			} else {
				consumed += bits.OnesCount8(header)
			}
			if consumed > len(remaining) {
				return nil, ErrUnpack
			}
			if produced > maxBytes-size {
				return nil, ErrLimit
			}
			size += produced
			remaining = remaining[consumed:]
		}
	}
	unpacked := make([]byte, size)
	// 经完整预检后恢复；限制单次汇编循环规模，小包和更大输入沿用Go路径。
	if len(src) >= 128 && len(src) <= 1<<20 && unpackVector(unpacked, src) {
		return unpacked, nil
	}
	offset := 0
	for len(src) > 0 {
		header := src[0]
		if header == 0 {
			run := 1
			if len(src) >= 2 && src[1] == 0 {
				run = 2 + zeroHeaderPrefix(src[2:])
			}
			// 输出由本次 make 清零，已预检的零段只需推进游标。
			offset += run * 8
			src = src[run:]
			continue
		}
		src = src[1:]
		if header == 0xff {
			length := (int(src[0]) + 1) * 8
			copy(unpacked[offset:], src[1:1+length])
			offset += length
			src = src[1+length:]
			continue
		}
		// 零 mask 已在上方消费，普通 mask 至少有一个有效字节。
		// 首遍已证明完整 word 的输出长度，固定视图让目标索引限于 0～7。
		dst := unpacked[offset : offset+8 : offset+8]
		used := 0
		for mask := header; ; {
			index := bits.TrailingZeros8(mask) & 7
			dst[index] = src[used]
			used++
			mask &= mask - 1
			if mask == 0 {
				break
			}
		}
		src = src[used:]
		offset += 8
	}
	return unpacked, nil
}
