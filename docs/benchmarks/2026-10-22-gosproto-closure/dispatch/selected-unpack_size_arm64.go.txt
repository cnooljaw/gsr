//go:build arm64 && !purego

package sproto

import "math/bits"

// 预检只需要字节数；固定256B表在初始化后只读，不携带恢复用的置换索引。
var unpackSizeCounts = func() (counts [256]uint8) {
	for header := range counts {
		counts[header] = uint8(bits.OnesCount8(uint8(header)))
	}
	return counts
}()

// unpackSizeFast 仅由公开入口在限定的输入范围内调用，返回尺寸与私有状态码。
func unpackSizeFast(src []byte, maxBytes int) (int, uint64) {
	return unpackedSizeARM64(src, maxBytes, &unpackSizeCounts)
}
