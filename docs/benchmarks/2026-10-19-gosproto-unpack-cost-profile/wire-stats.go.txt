package benchmarks

import (
 "bytes"
 "crypto/sha256"
 "encoding/hex"
 "encoding/json"
 "fmt"
 "math/bits"
 "os"
 "testing"

 sproto "github.com/cnooljaw/gosproto"
 before "github.com/cnooljaw/gosproto-before"
)

// TestUnpackCostWireStats 核实代价夹具实际经过的mask分类，不将源码未改等同于零成本。
func TestUnpackCostWireStats(t *testing.T) {
 type statistics struct {
  Name string
  RawBytes, PackedBytes, ProducedBytes int
  ZeroMasks, FFRuns, FFBytes, OrdinaryWords, OrdinaryPayload int
  Headers map[string]int
  RawSHA256, PackedSHA256 string
 }
 var output []statistics
 for _,shape:=range unpackShapes() {
  if shape.name!="FF512-sparse4096" && shape.name!="zeroRun8x64" && shape.name!="nonzero2x256" {continue}
  wire:=sproto.Pack(shape.raw)
  if !bytes.Equal(wire,before.Pack(shape.raw)) {t.Fatal("Pack wire differs",shape.name)}
  decoded,err:=sproto.Unpack(wire)
  if err!=nil || !bytes.Equal(decoded,shape.raw) {t.Fatal("unpack differs",shape.name,err)}
  rawHash,packedHash:=sha256.Sum256(shape.raw),sha256.Sum256(wire)
  row:=statistics{Name:shape.name,RawBytes:len(shape.raw),PackedBytes:len(wire),Headers:make(map[string]int),RawSHA256:hex.EncodeToString(rawHash[:]),PackedSHA256:hex.EncodeToString(packedHash[:])}
  for position:=0;position<len(wire); {
   header:=wire[position];position++
   row.Headers[fmt.Sprintf("%02x",header)]++
   if header==255 {
    if position>=len(wire) {t.Fatal("missing FF count")}
    length:=(int(wire[position])+1)*8;position++
    if length>len(wire)-position {t.Fatal("truncated FF")}
    row.FFRuns++;row.FFBytes+=length;row.ProducedBytes+=length;position+=length
   } else {
    count:=bits.OnesCount8(header)
    if count>len(wire)-position {t.Fatal("truncated mask")}
    if header==0 {row.ZeroMasks++} else {row.OrdinaryWords++;row.OrdinaryPayload+=count}
    position+=count;row.ProducedBytes+=8
   }
  }
  if row.ProducedBytes!=len(shape.raw) {t.Fatal("produced length differs")}
  output=append(output,row)
 }
 if len(output)!=3 {t.Fatal("incomplete selected shapes")}
 data,err:=json.MarshalIndent(output,"","  ")
 if err!=nil {t.Fatal(err)}
 if err:=os.WriteFile(os.Getenv("GSR_WIRE_STATS"),append(data,'\n'),0644);err!=nil {t.Fatal(err)}
}
