package sproto

import (
	"bytes"
	"errors"
	"reflect"
	"runtime"
	"runtime/debug"
	"testing"
)

type stringPointerRecord struct {
	Padding [64]byte
	First   *string `sproto:"string,0"`
	Number  *int64  `sproto:"integer,1"`
	Last    *string `sproto:"string,2"`
	Marker  int
}
type stringPointerPure struct {
	First *string `sproto:"string,0"`
	Last  *string `sproto:"string,1"`
}
type stringPointerText string
type stringPointerNamed struct {
	First *stringPointerText `sproto:"string,0"`
	Last  *stringPointerText `sproto:"string,1"`
}
type stringPointerRows struct {
	Items []*stringPointerPure `sproto:"struct,0,array"`
}
type stringPointerForeign struct {
	Small   byte
	Value   *string `sproto:"string,0"`
	Padding [64]byte
}

// 验证字符串受控赋值保持复制、独占、presence、低GC、描述回退与失败原子性。
func checkStringPointerOwnership(t *testing.T) {
	oldGC := debug.SetGCPercent(1)
	defer debug.SetGCPercent(oldGC)
	text, empty, n := "中文🙂/text", "", int64(-1)
	namedText, namedEmpty := stringPointerText(text), stringPointerText(empty)
	samples := []any{&stringPointerRecord{First: &text, Last: &empty, Number: &n}, &stringPointerPure{First: &text, Last: &empty}, &stringPointerPure{}, &stringPointerNamed{First: &namedText, Last: &namedEmpty}, &stringPointerRows{Items: []*stringPointerPure{{First: &text, Last: &empty}, {First: &text, Last: &empty}}}}
	var collect func(reflect.Value) []reflect.Value
	collect = func(v reflect.Value) []reflect.Value {
		if v.Kind() == reflect.Ptr {
			if v.IsNil() {
				return nil
			}
			if v.Type().Elem().Kind() == reflect.String {
				return []reflect.Value{v}
			}
			return collect(v.Elem())
		}
		var values []reflect.Value
		if v.Kind() == reflect.Struct {
			for i := 0; i < v.NumField(); i++ {
				values = append(values, collect(v.Field(i))...)
			}
		}
		if v.Kind() == reflect.Slice {
			for i := 0; i < v.Len(); i++ {
				values = append(values, collect(v.Index(i))...)
			}
		}
		return values
	}
	for _, input := range samples {
		wire, err := Encode(input)
		if err != nil {
			t.Fatal(err)
		}
		a, b := reflect.New(reflect.TypeOf(input).Elem()).Interface(), reflect.New(reflect.TypeOf(input).Elem()).Interface()
		if used, err := Decode(wire, a); err != nil || used != len(wire) {
			t.Fatal(used, err)
		}
		if _, err := Decode(wire, b); err != nil {
			t.Fatal(err)
		}
		if !reflect.DeepEqual(a, b) {
			t.Fatal("different decoded values")
		}
		before, err := Encode(a)
		if err != nil {
			t.Fatal(err)
		}
		left, right := collect(reflect.ValueOf(a)), collect(reflect.ValueOf(b))
		seen := map[uintptr]bool{}
		for i, v := range left {
			if v.Pointer() == right[i].Pointer() || seen[v.Pointer()] {
				t.Fatal("string pointers alias")
			}
			seen[v.Pointer()] = true
			right[i].Elem().SetString("changed")
		}
		for i := range wire {
			wire[i] ^= 255
		}
		runtime.GC()
		after, err := Encode(a)
		if err != nil || !bytes.Equal(before, after) {
			t.Fatal("input, later decode or GC changed strings", err)
		}
	}
	wire, err := Encode(&stringPointerRecord{First: &text, Last: &empty, Number: &n})
	if err != nil {
		t.Fatal(err)
	}
	sentinel := stringPointerRecord{First: &text, Marker: 31}
	if _, err := Decode(wire[:len(wire)-1], &sentinel); !errors.Is(err, ErrDecode) || sentinel.Marker != 31 || sentinel.First != &text {
		t.Fatal("bad tail changed caller", err)
	}
	schema, err := GetSprotoType(reflect.TypeFor[stringPointerRecord]())
	if err != nil {
		t.Fatal(err)
	}
	original := append([]*SprotoField(nil), schema.Fields...)
	defer func() { schema.Fields = original }()
	field := schema.Fields[1]
	oldWire, oldArray := field.Wire, field.Array
	defer func() { field.Wire, field.Array = oldWire, oldArray }()
	inline := []byte{1, 0, 4, 0}
	sentinel = stringPointerRecord{Marker: 31}
	if _, err := Decode(inline, &sentinel); !errors.Is(err, ErrDecode) || sentinel.Marker != 31 {
		t.Fatal("inline string accepted", err)
	}
	// 改Wire后原decoder仍是string，保持既有反射语义，不借用未经证明的offset。
	field.Wire = WireVarintName
	var output stringPointerRecord
	if _, err := Decode(inline, &output); err != nil || output.First == nil || *output.First != "" {
		t.Fatal("changed wire semantics", err)
	}
	field.Wire = oldWire
	field.Array = true
	if _, err := Decode(wire, &output); err != nil || output.First == nil || *output.First != text {
		t.Fatal("changed array semantics", err)
	}
	field.Array = oldArray
	foreign, err := GetSprotoType(reflect.TypeFor[stringPointerForeign]())
	if err != nil {
		t.Fatal(err)
	}
	schema.Fields = append([]*SprotoField(nil), original...)
	schema.Fields[3] = foreign.Fields[1]
	output = stringPointerRecord{}
	if _, err := Decode(wire, &output); err != nil || output.First == nil || *output.First != "" || output.Last != nil || output.Number == nil || *output.Number != n {
		t.Fatal("foreign owner index fallback", err)
	}
	schema.Fields = append([]*SprotoField(nil), original...)
	schema.Fields[1] = nil
	output = stringPointerRecord{}
	if _, err := Decode(wire, &output); err != nil || output.First != nil || output.Last == nil {
		t.Fatal("nil field semantics", err)
	}
}

// 验证三种递归parser均保持字符串独占、元信息回退和失败原子性。
func TestStringExperiment(t *testing.T) {
	defer ExperimentStringMode("baseline")
	for _, mode := range []string{"baseline", "clone", "candidate"} {
		ExperimentStringMode(mode)
		t.Run(mode, checkStringPointerOwnership)
	}
}
