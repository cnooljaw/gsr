package benchmarks

import (
 "encoding/json"
 "math/bits"
 "os"
 "testing"
)

type unpackPattern struct {
 Name string
 RawBytes, PackedBytes, Words, ZeroHeaders, OrdinaryHeaders, FFHeaders, FFWords int
 OrdinaryBytes int
 OrdinaryBits [9]int
 Headers [256]int
 MaxZeroRun, MaxFFWords int
}

// TestUnpackPatterns 分配和profile之外统计真实wire，不影响计时路径。
func TestUnpackPatterns(t *testing.T) {
 var records []unpackPattern
 for _,sample:=range representativeCases(t) {
  if sample.Name!="int32_edges" && sample.Name!="game_large" && sample.Name!="addressbook" {continue}
  _,wires:=representativeOperations(t,sample);checkRepresentativeWire(t,sample,wires)
  packed:=wires["sproto_pack"];r:=unpackPattern{Name:sample.Name,RawBytes:len(wires["sproto_encode"]),PackedBytes:len(packed)};zeroRun:=0
  for len(packed)>0 {
   header:=packed[0];packed=packed[1:];r.Headers[header]++
   if header==0 {r.ZeroHeaders++;r.Words++;zeroRun++;r.MaxZeroRun=max(r.MaxZeroRun,zeroRun);continue};zeroRun=0
   if header==255 {if len(packed)<1 {t.Fatal("truncated ff")};n:=int(packed[0])+1;if len(packed)<1+n*8 {t.Fatal("truncated ff body")};r.FFHeaders++;r.FFWords+=n;r.Words+=n;r.MaxFFWords=max(r.MaxFFWords,n);packed=packed[1+n*8:];continue}
   n:=bits.OnesCount8(header);if len(packed)<n {t.Fatal("truncated ordinary")};r.OrdinaryHeaders++;r.OrdinaryBits[n]++;r.OrdinaryBytes+=n;r.Words++;packed=packed[n:]
  }
  if r.Words*8 < r.RawBytes || r.Words*8-r.RawBytes>7 {t.Fatal("padding")}
  records=append(records,r)
 }
 raw,err:=json.MarshalIndent(records,"","  ");if err!=nil{t.Fatal(err)}
 if err:=os.WriteFile(os.Getenv("GSR_PATTERN_OUTPUT"),append(raw,'\n'),0644);err!=nil{t.Fatal(err)}
}
