package inlinediff

import (
	now "github.com/cnooljaw/gosproto"
	old "github.com/cnooljaw/gosproto-before"
	"math"
	"reflect"
	"runtime"
	"testing"
)

// 不同对齐和非零 offset 的指针写入在 GC 后仍有效，记录和调用之间保持独占。
func TestTypedStorePaddingAndGC(t *testing.T) {
	type record struct {
		Before [127]byte
		First  *int64 `sproto:"integer,0"`
		Gap    [13]byte
		Last   *int64 `sproto:"integer,30000"`
	}
	type root struct {
		Items []*record `sproto:"struct,0,array"`
	}
	var retained []*root
	for i := 0; i < 48; i++ {
		a, b := int64(i), int64(math.MaxInt64-int64(i))
		input := &root{Items: []*record{{First: &a, Last: &b}, {First: &b, Last: &a}}}
		wire, err := old.Encode(input)
		if err != nil {
			t.Fatal(err)
		}
		left, right := new(root), new(root)
		_, e1 := old.Decode(wire, left)
		_, e2 := now.Decode(wire, right)
		if e1 != nil || e2 != nil || !reflect.DeepEqual(left, right) {
			t.Fatalf("decode padding %v/%v", e1, e2)
		}
		if right.Items[0].First == right.Items[1].Last || right.Items[0].Last == right.Items[1].First {
			t.Fatal("shared pointer")
		}
		retained = append(retained, right)
		if i%4 == 0 {
			runtime.GC()
		}
	}
	runtime.GC()
	for i, output := range retained {
		if *output.Items[0].First != int64(i) || *output.Items[0].Last != math.MaxInt64-int64(i) {
			t.Fatal("GC damaged pointer values")
		}
	}
	*retained[0].Items[0].First = -1
	if *retained[0].Items[1].Last != 0 || *retained[1].Items[0].First != 1 {
		t.Fatal("mutation leaked")
	}
}

// 字段描述来自不同布局时，保留原索引语义且不能按外来 offset 写入。
func TestTypedStoreForeignOffsetFallback(t *testing.T) {
	type record struct {
		Padding [64]byte
		First   *int64 `sproto:"integer,0"`
		Last    *int64 `sproto:"integer,1"`
	}
	type document struct {
		Items []*record `sproto:"struct,0,array"`
	}
	type other struct {
		Small   byte
		Value   *int64 `sproto:"integer,0"`
		Padding [64]byte
	}
	first, last := int64(1), int64(2)
	wire, err := now.Encode(&document{Items: []*record{{First: &first, Last: &last}, {}}})
	if err != nil {
		t.Fatal(err)
	}
	schema, err := now.GetSprotoType(reflect.TypeFor[record]())
	if err != nil {
		t.Fatal(err)
	}
	replacement, err := now.GetSprotoType(reflect.TypeFor[other]())
	if err != nil {
		t.Fatal(err)
	}
	original := schema.Fields[2]
	schema.Fields[2] = replacement.Fields[1]
	t.Cleanup(func() { schema.Fields[2] = original })
	var output document
	if _, err := now.Decode(wire, &output); err != nil || len(output.Items) != 2 ||
		output.Items[0].First == nil || *output.Items[0].First != last || output.Items[0].Last != nil {
		t.Fatalf("replacement index semantics: %+v %v", output, err)
	}
}
