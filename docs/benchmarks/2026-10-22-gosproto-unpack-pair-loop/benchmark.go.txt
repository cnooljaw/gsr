package benchmarks

import (
 "bytes"
 "fmt"
 "testing"
 sproto "github.com/cnooljaw/gosproto"
)

// BenchmarkPairUnpackReview 验证普通mask展开在真实组合、不同有效位数量与短包上的成本。
func BenchmarkPairUnpackReview(b *testing.B) {
 for _,sample:=range representativeCases(b) {
  if sample.Name!="addressbook" && sample.Name!="game_large" && sample.Name!="int32_edges" {continue}
  operations,wires:=representativeOperations(b,sample);checkRepresentativeWire(b,sample,wires)
  for _,op:=range operations {
   if op.name!="sproto_unpack" && op.name!="sproto_pdecode" {continue}
   b.Run(sample.Name+"/"+op.name,func(b *testing.B){b.ReportAllocs();b.ResetTimer();for range b.N {if err:=op.run();err!=nil{b.Fatal(err)}}})
  }
 }
 var shapes []unpackShape
 for _,shape:=range unpackShapes() {
  if shape.name=="zero32" || shape.name=="zero4096" || shape.name=="ascii32" || shape.name=="ascii4096" || (len(shape.name)>7 && shape.name[:7]=="nonzero") {shapes=append(shapes,shape)}
 }
 for _,count:=range []int{1,3,7} {
  word:=make([]byte,8)
  for i:=0;i<count;i++ {word[(i*3)%8]=byte(i+1)}
  shapes=append(shapes,unpackShape{fmt.Sprintf("single%d",count),word})
 }
 for _,shape:=range shapes {
  packed:=sproto.Pack(shape.raw);output,err:=sproto.Unpack(packed)
  if err!=nil || !bytes.Equal(output,shape.raw) || cap(output)!=len(shape.raw){b.Fatal("shape/wire/cap",shape.name,err)}
  b.Run(shape.name,func(b *testing.B){b.ReportAllocs();b.ResetTimer();for range b.N {out,err:=sproto.Unpack(packed);if err!=nil{b.Fatal(err)};encodedSink=out}})
 }
}
