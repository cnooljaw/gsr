package sproto

import (
	"bytes"
	"encoding/binary"
	"math/bits"
)

func packSegment(dst []byte, src []byte, ffN int) int {

	var header uint8 = 0
	notzero := 0
	for i, v := range src {
		if v != 0 {
			notzero++
			header |= (1 << uint(i))
			dst[notzero] = v
		}
	}

	if ffN > 0 {
		if notzero >= 6 {
			return 8
		}
	}

	if notzero == 8 {
		return 10
	}

	dst[0] = header
	return notzero + 1
}

func writeFF(dst, src []byte, n int) {
	// 最后一个连续块可能不足八字节，不能读取切片长度之外的底层容量。
	n = min(n, len(src))
	align8 := (n + 7) & (^7)
	dst[0] = 0xff
	dst[1] = uint8(align8/8 - 1)
	copy(dst[2:], src[:n])
	for i := n; i < align8; i++ {
		dst[2+i] = 0
	}
}

// packFFMergeable 判断当前 word 是否可按官方规则继续 FF 段。
func packFFMergeable(header byte) bool {
	zero := ^header
	remaining := zero & (zero - 1)
	return remaining&(remaining-1) == 0
}

// followingFFWords 只在已确认的 FF 起点后搜索，最多接纳 255 个完整非零 word。
// 含零 word 和不足八字节的尾部不计入，继续交给普通 FF 合并逻辑。
func followingFFWords(src []byte) int {
	available := min(2040, len(src)) &^ 7
	if zero := bytes.IndexByte(src[:available], 0); zero >= 0 {
		return zero / 8
	}
	return available / 8
}

// ffRunBytes 仅在已确认的完整 FF word 起点调用；最多接纳 256 个 word。
// 后续完整 word 至少六个非零字节才合并，短尾块按实际长度计数。
func ffRunBytes(src []byte) int {
	limit := min(len(src), 2048)
	consumed := 8
	if len(src) >= 256 {
		consumed += followingFFWords(src[8:]) * 8
	}
	for consumed+8 <= limit {
		word := binary.LittleEndian.Uint64(src[consumed:])
		if (word-0x0101010101010101)&^word&0x8080808080808080 != 0 {
			nonzero := (((word & 0x7f7f7f7f7f7f7f7f) + 0x7f7f7f7f7f7f7f7f) | word) & 0x8080808080808080
			header := uint8(((nonzero >> 7) * 0x0102040810204080) >> 56)
			if !packFFMergeable(header) {
				return consumed
			}
		}
		consumed += 8
	}
	if consumed < limit {
		nonzero := 0
		for _, value := range src[consumed:limit] {
			if value != 0 {
				nonzero++
			}
		}
		if nonzero >= 6 {
			consumed = limit
		}
	}
	return consumed
}

func packedBufferSize(size int) int {
	// 每 256 个 word 最多增加一个双字节 FF 头，另保留末尾补齐空间。
	return size + ((size+2047)/2048)*2 + 8
}

// Pack 按 Sproto 规则压缩字节，返回独占缓冲且不修改输入。
func Pack(src []byte) []byte {
	return packInto(make([]byte, packedBufferSize(len(src))), src)
}

// packInto 只接收本次调用独占且按 packedBufferSize 预留的输出。
// EncodePacked 允许 src 位于同一缓冲后部，必须保留完整增长额度和八字节写出余量。
func packInto(packed, src []byte) []byte {
	offset := 0
	for len(src) >= 8 {
		word := binary.LittleEndian.Uint64(src)
		nonzero := (((word & 0x7f7f7f7f7f7f7f7f) + 0x7f7f7f7f7f7f7f7f) | word) & 0x8080808080808080
		header := uint8(((nonzero >> 7) * 0x0102040810204080) >> 56)
		if header == 0xff {
			consumed := ffRunBytes(src)
			writeFF(packed[offset:], src, consumed)
			offset += 2 + ((consumed + 7) &^ 7)
			src = src[consumed:]
			continue
		}
		// 去掉至多两个置位后是否为零，比普通路径完整计数更直接。
		remaining := header & (header - 1)
		if remaining&(remaining-1) == 0 {
			// 长度与容量均固定且非空，避免每个 word 重算动态切片的指针掩码。
			dst := packed[offset : offset+3 : offset+3]
			// 已确认至多两个非零字节，直接写固定位置，避免通用循环和可变目标索引。
			_ = dst[2]
			dst[0] = header
			index := 1
			if header != 0 {
				shift := bits.TrailingZeros8(header) * 8
				dst[1] = byte(word >> shift)
				index = 2
				if remaining != 0 {
					shift = bits.TrailingZeros8(remaining) * 8
					dst[2] = byte(word >> shift)
					index = 3
				}
			}
			offset += index
		} else {
			dst := packed[offset : offset+9 : offset+9]
			// Pack 预留八字节尾空间，零字节暂写的位置会被下一个有效字节覆盖，不进入返回切片。
			_ = dst[8]
			// 固定八字节视图越过 mask，非零字节位置直接使用 0～7 的前缀计数。
			payload := dst[1:9]
			// 0/1 字节标志相乘得到前缀和，每个和至多为 8，不跨字节进位。
			// FF 已在前面处理；各目标位置只依赖对应前缀，不串行累加 index。
			ranks := (nonzero >> 7) * 0x0101010101010101
			payload[0] = byte(word)
			payload[int(ranks&7)] = byte(word >> 8)
			payload[int((ranks>>8)&7)] = byte(word >> 16)
			payload[int((ranks>>16)&7)] = byte(word >> 24)
			payload[int((ranks>>24)&7)] = byte(word >> 32)
			payload[int((ranks>>32)&7)] = byte(word >> 40)
			payload[int((ranks>>40)&7)] = byte(word >> 48)
			payload[int((ranks>>48)&7)] = byte(word >> 56)
			dst[0] = header
			offset += 1 + int(ranks>>56)
		}
		src = src[8:]
	}
	if len(src) > 0 {
		offset += packSegment(packed[offset:], src, 0)
	}
	return packed[:offset]
}

// zeroHeaderPrefix 按有效切片长度统计连续零 mask，遇到首个非零字节立即停止。
// 调用方只在已确认的零段后使用；预检传入的范围另受剩余输出预算限制。
func zeroHeaderPrefix(src []byte) int {
	index := 0
	for len(src)-index >= 8 {
		word := binary.LittleEndian.Uint64(src[index:])
		if word != 0 {
			return index + bits.TrailingZeros64(word)/8
		}
		index += 8
	}
	for index < len(src) && src[index] == 0 {
		index++
	}
	return index
}

// Unpack 使用默认解包大小预算。
func Unpack(src []byte) ([]byte, error) {
	return UnpackWithLimit(src, 8<<20)
}

// UnpackWithLimit 在扩容前限制解包后含填充的长度；packed 输入大小由调用方限制。
func UnpackWithLimit(src []byte, maxBytes int) ([]byte, error) {
	if maxBytes < 1 {
		return nil, ErrInvalidLimits
	}
	// 首次只扫描 tag/连续块长度，在分配前确认完整边界及有限的确切输出长度。
	size := 0
	for remaining := src; len(remaining) > 0; {
		header := remaining[0]
		consumed, produced := 1, 8
		if header == 0 {
			if len(remaining) >= 2 && remaining[1] == 0 {
				words := (maxBytes - size) / 8
				if words < 2 {
					return nil, ErrLimit
				}
				// 多检查一个零 mask 即可判定超额，不为小预算扫描整段大输入。
				end := min(len(remaining), words+1)
				run := 2 + zeroHeaderPrefix(remaining[2:end])
				if run > words {
					return nil, ErrLimit
				}
				consumed, produced = run, run*8
			}
		} else if header == 0xff {
			if len(remaining) < 2 {
				return nil, ErrUnpack
			}
			produced = (int(remaining[1]) + 1) * 8
			consumed = 2 + produced
		} else {
			consumed += bits.OnesCount8(header)
		}
		if consumed > len(remaining) {
			return nil, ErrUnpack
		}
		if produced > maxBytes-size {
			return nil, ErrLimit
		}
		size += produced
		remaining = remaining[consumed:]
	}
	unpacked := make([]byte, size)
	offset := 0
	for len(src) > 0 {
		header := src[0]
		if header == 0 {
			run := 1
			if len(src) >= 2 && src[1] == 0 {
				run = 2 + zeroHeaderPrefix(src[2:])
			}
			// 输出由本次 make 清零，已预检的零段只需推进游标。
			offset += run * 8
			src = src[run:]
			continue
		}
		src = src[1:]
		if header == 0xff {
			length := (int(src[0]) + 1) * 8
			copy(unpacked[offset:], src[1:1+length])
			offset += length
			src = src[1+length:]
			continue
		}
		// 零 mask 已在上方消费，普通 mask 至少有一个有效字节。
		// 首遍已证明完整 word 的输出长度，固定视图让目标索引限于 0～7。
		dst := unpacked[offset : offset+8 : offset+8]
		used := 0
		for mask := header; ; {
			index := bits.TrailingZeros8(mask) & 7
			dst[index] = src[used]
			used++
			mask &= mask - 1
			if mask == 0 {
				break
			}
			index = bits.TrailingZeros8(mask) & 7
			dst[index] = src[used]
			used++
			mask &= mask - 1
			if mask == 0 {
				break
			}
		}
		src = src[used:]
		offset += 8
	}
	return unpacked, nil
}
